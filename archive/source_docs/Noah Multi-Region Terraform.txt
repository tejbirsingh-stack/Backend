# modules/noah-infrastructure/main.tf
resource "aws_vpc" "main" {
  for_each = var.vpc_config
  
  cidr_block           = each.value.cidr
  enable_dns_hostnames = true
  enable_dns_support   = true
  
  tags = merge(var.tags, {
    Name = "noah-${var.environment}-${each.key}"
    Region = each.key
  })
}

resource "aws_subnet" "private" {
  for_each = {
    for subnet in flatten([
      for region, config in var.vpc_config : [
        for idx, cidr in config.private_subnets : {
          key    = "${region}-private-${idx}"
          region = region
          cidr   = cidr
          az     = config.azs[idx]
          vpc_id = aws_vpc.main[region].id
        }
      ]
    ]) : subnet.key => subnet
  }
  
  vpc_id                  = each.value.vpc_id
  cidr_block              = each.value.cidr
  availability_zone       = each.value.az
  map_public_ip_on_launch = false
  
  tags = merge(var.tags, {
    Name = "noah-${var.environment}-${each.key}"
    Type = "Private"
  })
}
