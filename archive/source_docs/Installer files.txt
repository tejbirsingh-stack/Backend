<!-- Package build script - build-installer.sh -->
#!/bin/bash

# Build macOS installer package
echo "Building macOS installer..."

# Create package structure
mkdir -p build/packages

# Build Premiere Pro package
pkgbuild --root premiere-pro/com.noah.premiere \
         --identifier com.noah.premiere \
         --version 1.0.0 \
         --scripts premiere-pro/scripts \
         --install-location "/Library/Application Support/Adobe/CEP/extensions/com.noah.premiere" \
         build/packages/premiere.pkg

# Build Final Cut Pro package
pkgbuild --root final-cut-pro/build/Release \
         --identifier com.noah.fcp \
         --version 1.0.0 \
         --install-location "/Library/Application Support/Final Cut Pro/Workflow Extensions" \
         build/packages/fcp.pkg

# Build DaVinci Resolve package
pkgbuild --root davinci-resolve/scripts \
         --identifier com.noah.resolve \
         --version 1.0.0 \
         --install-location "/Library/Application Support/Blackmagic Design/DaVinci Resolve/Fusion/Scripts/Comp/Noah" \
         build/packages/resolve.pkg

# Build distribution package
productbuild --distribution distribution.xml \
             --package-path build/packages \
             --resources resources \
             "NoahIntegration-1.0.0.pkg"

# Sign the package
productsign --sign "Developer ID Installer: Noah Systems Inc" \
            NoahIntegration-1.0.0.pkg \
            NoahIntegration-1.0.0-signed.pkg

# Notarize
xcrun altool --notarize-app \
             --primary-bundle-id "com.noah.installer" \
             --username "$APPLE_ID" \
             --password "$APPLE_APP_PASSWORD" \
             --file NoahIntegration-1.0.0-signed.pkg

echo "macOS installer built successfully!"

# Video Editor Plugin Certification Requirements

## Adobe Creative Cloud

### Technical Requirements
- **CEP Version**: 9.0 or higher (CC 2019+)
- **Host Apps**: Premiere Pro 14.0+, After Effects 17.0+
- **Node.js**: Embedded Node.js 12.x
- **Signing**: Required for distribution

### Submission Process
1. Create Adobe Developer account
2. Generate signing certificate
3. Sign the ZXP package:
   ```bash
   ZXPSignCmd -sign com.noah.premiere dist/noah-premiere.zxp certificate.p12 password
   ```
4. Submit via Adobe Developer Console
5. Complete metadata and screenshots
6. Submit for review (5-10 business days)
