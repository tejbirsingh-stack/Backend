// VideoCompressorModule.kt
package com.yourcompany.instagram.modules

import com.facebook.react.bridge.*
import com.arthenica.mobileffmpeg.Config
import com.arthenica.mobileffmpeg.FFmpeg
import kotlinx.coroutines.*
import java.io.File
import android.media.MediaMetadataRetriever
import android.net.Uri

class VideoCompressorModule(reactContext: ReactApplicationContext) : 
    ReactContextBaseJavaModule(reactContext) {
    
    override fun getName(): String = "VideoCompressor"
    
    @ReactMethod
    fun compressVideo(
        inputPath: String,
        outputPath: String,
        quality: Float,
        promise: Promise
    ) {
        GlobalScope.launch(Dispatchers.IO) {
            try {
                val inputFile = File(inputPath)
                if (!inputFile.exists()) {
                    promise.reject("FILE_NOT_FOUND", "Input file does not exist")
                    return@launch
                }
                
                // Get video metadata
                val retriever = MediaMetadataRetriever()
                retriever.setDataSource(inputPath)
                val duration = retriever.extractMetadata(
                    MediaMetadataRetriever.METADATA_KEY_DURATION
                )?.toLong() ?: 0
                val width = retriever.extractMetadata(
                    MediaMetadataRetriever.METADATA_KEY_VIDEO_WIDTH
                )?.toInt() ?: 1920
                val height = retriever.extractMetadata(
                    MediaMetadataRetriever.METADATA_KEY_VIDEO_HEIGHT
                )?.toInt() ?: 1080
                retriever.release()
                
                // Calculate compression parameters
                val bitrate = calculateBitrate(quality, width, height)
                val scale = calculateScale(quality, width, height)
                
                // FFmpeg command for compression
                val command = arrayOf(
                    "-i", inputPath,
                    "-c:v", "libx264",
                    "-preset", getPreset(quality),
                    "-crf", getCRF(quality),
                    "-vf", "scale=$scale",
                    "-c:a", "aac",
                    "-b:a", "128k",
                    "-movflags", "+faststart",
                    "-y",
                    outputPath
                )
                
                val rc = FFmpeg.execute(command)
                
                if (rc == Config.RETURN_CODE_SUCCESS) {
                    val outputFile = File(outputPath)
                    val result = WritableNativeMap().apply {
                        putString("path", outputPath)
                        putDouble("size", outputFile.length().toDouble())
                        putDouble("duration", duration.toDouble() / 1000)
                    }
                    promise.resolve(result)
                } else {
                    promise.reject(
                        "COMPRESSION_FAILED",
                        "FFmpeg execution failed with rc=$rc"
                    )
                }
            } catch (e: Exception) {
                promise.reject("COMPRESSION_ERROR", e.message, e)
            }
        }
    }
    
    private fun calculateBitrate(quality: Float, width: Int, height: Int): String {
        val baseBitrate = when {
            quality < 0.3 -> 1000000 // 1 Mbps
            quality < 0.7 -> 2500000 // 2.5 Mbps
            else -> 5000000 // 5 Mbps
        }
        
        val pixels = width * height
        val hdPixels = 1920 * 1080
        val scaleFactor = pixels.toFloat() / hdPixels
        
        return "${(baseBitrate * scaleFactor).toInt()}"
    }
    
    private fun calculateScale(quality: Float, width: Int, height: Int): String {
        val maxDimension = when {
            quality < 0.3 -> 854
            quality < 0.7 -> 1280
            else -> 1920
        }
        
        return if (width > height) {
            "w=$maxDimension:h=-2"
        } else {
            "w=-2:h=$maxDimension"
        }
    }
    
    private fun getPreset(quality: Float): String = when {
        quality < 0.3 -> "ultrafast"
        quality < 0.7 -> "fast"
        else -> "medium"
    }
    
    private fun getCRF(quality: Float): String = when {
        quality < 0.3 -> "28"
        quality < 0.7 -> "23"
        else -> "18"
    }
}

// ImageProcessorModule.kt
package com.yourcompany.instagram.modules

import com.facebook.react.bridge.*
import android.graphics.*
import android.graphics.ColorMatrix
import android.graphics.ColorMatrixColorFilter
import android.media.ExifInterface
import kotlinx.coroutines.*
import java.io.File
import java.io.FileOutputStream
import java.util.UUID

class ImageProcessorModule(reactContext: ReactApplicationContext) : 
    ReactContextBaseJavaModule(reactContext) {
    
    override fun getName(): String = "ImageProcessor"
    
    @ReactMethod
    fun applyFilter(
        imagePath: String,
        filter: String,
        promise: Promise
    ) {
        GlobalScope.launch(Dispatchers.IO) {
            try {
                val bitmap = loadBitmap(imagePath)
                val filteredBitmap = when (filter) {
                    "vintage" -> applyVintageFilter(bitmap)
                    "noir" -> applyNoirFilter(bitmap)
                    "chrome" -> applyChromeFilter(bitmap)
                    "fade" -> applyFadeFilter(bitmap)
                    "mono" -> applyMonoFilter(bitmap)
                    else -> bitmap
                }
                
                val outputPath = saveBitmap(filteredBitmap)
                bitmap.recycle()
                filteredBitmap.recycle()
                
                promise.resolve(outputPath)
            } catch (e: Exception) {
                promise.reject("FILTER_ERROR", e.message, e)
            }
        }
    }
    
    @ReactMethod
    fun cropImage(
        imagePath: String,
        cropData: ReadableMap,
        promise: Promise
    ) {
        GlobalScope.launch(Dispatchers.IO) {
            try {
                val bitmap = loadBitmap(imagePath)
                
                val x = cropData.getInt("x")
                val y = cropData.getInt("y")
                val width = cropData.getInt("width")
                val height = cropData.getInt("height")
                
                val croppedBitmap = Bitmap.createBitmap(bitmap, x, y, width, height)
                val outputPath = saveBitmap(croppedBitmap)
                
                bitmap.recycle()
                croppedBitmap.recycle()
                
                promise.resolve(outputPath)
            } catch (e: Exception) {
                promise.reject("CROP_ERROR", e.message, e)
            }
        }
    }
    
    @ReactMethod
    fun adjustImage(
        imagePath: String,
        adjustments: ReadableMap,
        promise: Promise
    ) {
        GlobalScope.launch(Dispatchers.IO) {
            try {
                val bitmap = loadBitmap(imagePath)
                val adjustedBitmap = bitmap.copy(bitmap.config, true)
                
                val canvas = Canvas(adjustedBitmap)
                val paint = Paint()
                
                val colorMatrix = ColorMatrix()
                
                // Apply brightness
                adjustments.getDouble("brightness").let { brightness ->
                    if (!brightness.isNaN()) {
                        colorMatrix.postConcat(ColorMatrix().apply {
                            setScale(
                                brightness.toFloat(),
                                brightness.toFloat(),
                                brightness.toFloat(),
                                1f
                            )
                        })
                    }
                }
                
                // Apply contrast
                adjustments.getDouble("contrast").let { contrast ->
                    if (!contrast.isNaN()) {
                        val scale = contrast.toFloat()
                        val translate = (1f - scale) / 2f * 255f
                        colorMatrix.postConcat(ColorMatrix(floatArrayOf(
                            scale, 0f, 0f, 0f, translate,
                            0f, scale, 0f, 0f, translate,
                            0f, 0f, scale, 0f, translate,
                            0f, 0f, 0f, 1f, 0f
                        )))
                    }
                }
                
                // Apply saturation
                adjustments.getDouble("saturation").let { saturation ->
                    if (!saturation.isNaN()) {
                        colorMatrix.setSaturation(saturation.toFloat())
                    }
                }
                
                paint.colorFilter = ColorMatrixColorFilter(colorMatrix)
                canvas.drawBitmap(bitmap, 0f, 0f, paint)
                
                val outputPath = saveBitmap(adjustedBitmap)
                
                bitmap.recycle()
                adjustedBitmap.recycle()
                
                promise.resolve(outputPath)
            } catch (e: Exception) {
                promise.reject("ADJUST_ERROR", e.message, e)
            }
        }
    }
    
    private fun loadBitmap(path: String): Bitmap {
        val options = BitmapFactory.Options().apply {
            inPreferredConfig = Bitmap.Config.ARGB_8888
        }
        
        val bitmap = BitmapFactory.decodeFile(path, options)
            ?: throw Exception("Failed to load bitmap")
        
        // Handle EXIF orientation
        val exif = ExifInterface(path)
        val orientation = exif.getAttributeInt(
            ExifInterface.TAG_ORIENTATION,
            ExifInterface.ORIENTATION_NORMAL
        )
        
        return when (orientation) {
            ExifInterface.ORIENTATION_ROTATE_90 -> rotateBitmap(bitmap, 90f)
            ExifInterface.ORIENTATION_ROTATE_180 -> rotateBitmap(bitmap, 180f)
            ExifInterface.ORIENTATION_ROTATE_270 -> rotateBitmap(bitmap, 270f)
            else -> bitmap
        }
    }
    
    private fun rotateBitmap(bitmap: Bitmap, degrees: Float): Bitmap {
        val matrix = Matrix().apply { postRotate(degrees) }
        return Bitmap.createBitmap(
            bitmap, 0, 0, bitmap.width, bitmap.height, matrix, true
        )
    }
    
    private fun saveBitmap(bitmap: Bitmap): String {
        val outputDir = reactApplicationContext.cacheDir
        val outputFile = File(outputDir, "processed_${UUID.randomUUID()}.jpg")
        
        FileOutputStream(outputFile).use { out ->
            bitmap.compress(Bitmap.CompressFormat.JPEG, 90, out)
        }
        
        return outputFile.absolutePath
    }
    
    private fun applyVintageFilter(bitmap: Bitmap): Bitmap {
        val matrix = ColorMatrix().apply {
            setSaturation(0.5f)
            postConcat(ColorMatrix(floatArrayOf(
                1.2f, 0.1f, 0.1f, 0f, 0f,
                0.1f, 1.0f, 0.1f, 0f, 0f,
                0.1f, 0.1f, 0.8f, 0f, 0f,
                0f, 0f, 0f, 1f, 0f
            )))
        }
        return applyColorMatrix(bitmap, matrix)
    }
    
    private fun applyNoirFilter(bitmap: Bitmap): Bitmap {
        val matrix = ColorMatrix().apply {
            setSaturation(0f)
            postConcat(ColorMatrix(floatArrayOf(
                1.5f, 0f, 0f, 0f, -50f,
                0f, 1.5f, 0f, 0f, -50f,
                0f, 0f, 1.5f, 0f, -50f,
                0f, 0f, 0f, 1f, 0f
            )))
        }
        return applyColorMatrix(bitmap, matrix)
    }
    
    private fun applyChromeFilter(bitmap: Bitmap): Bitmap {
        val matrix = ColorMatrix(floatArrayOf(
            1.5f, 0f, 0f, 0f, 0f,
            0f, 1.3f, 0f, 0f, 0f,
            0f, 0f, 1.0f, 0f, 0f,
            0f, 0f, 0f, 1f, 0f
        ))
        return applyColorMatrix(bitmap, matrix)
    }
    
    private fun applyFadeFilter(bitmap: Bitmap): Bitmap {
        val matrix = ColorMatrix().apply {
            setScale(1.2f, 1.2f, 1.2f, 0.8f)
        }
        return applyColorMatrix(bitmap, matrix)
    }
    
    private fun applyMonoFilter(bitmap: Bitmap): Bitmap {
        val matrix = ColorMatrix().apply {
            setSaturation(0f)
        }
        return applyColorMatrix(bitmap, matrix)
    }
    
    private fun applyColorMatrix(bitmap: Bitmap, colorMatrix: ColorMatrix): Bitmap {
        val result = bitmap.copy(bitmap.config, true)
        val canvas = Canvas(result)
        val paint = Paint().apply {
            colorFilter = ColorMatrixColorFilter(colorMatrix)
        }
        canvas.drawBitmap(bitmap, 0f, 0f, paint)
        return result
    }
}
