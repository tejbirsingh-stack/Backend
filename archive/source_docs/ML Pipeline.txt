    def _recognize_faces(self, video_path: str) -> Dict:

        cap = cv2.VideoCapture(video_path)
        fps = cap.get(cv2.CAP_PROP_FPS)

        frame_idx = 0
        
        while cap.isOpened():
            ret, frame = cap.read()

            className={`px-6 py-3 rounded-lg transition-colors ${

                ? 'bg-purple-600 text-white' 
                : 'bg-gray-800 text-gray-300 hover:bg-gray-700'
            }`}
          >
            <Monitor className="w-4 h-4 inline mr-2" />

          </button>

            onClick={() => setActiveTab('ab-testing')}
            className={`px-6 py-3 rounded-lg transition-colors ${
              activeTab === 'ab-testing' 
                ? 'bg-purple-600 text-white' 
                : 'bg-gray-800 text-gray-300 hover:bg-gray-700'

            }`}

            <DollarSign className="w-4 h-4 inline mr-2" />

              </div>
              
              <div className="space-y-4">
                <div className="bg-gray-800 rounded-lg p-4">
                  <h4 className="text-white font-semibold mb-2">Model Information</h4>
                  <div className="grid grid-cols-2 gap-4 text-sm">
                    <div>
                      <span className="text-gray-400">Architecture:</span>
                      <div className="text-white">
                        {selectedModel === 'classification' && 'Swin Transformer Large'}
                        {selectedModel === 'detection' && 'DETR ResNet-101'}
                        {selectedModel === 'segmentation' && 'SegFormer B5'}
                        {selectedModel === 'captioning' && 'BLIP Large'}
                      </div>
                    </div>
                    <div>
                      <span className="text-gray-400">Parameters:</span>
                      <div className="text-white">
                        {selectedModel === 'classification' && '197M'}
                        {selectedModel === 'detection' && '60M'}
                        {selectedModel === 'segmentation' && '82M'}
                        {selectedModel === 'captioning' && '447M'}
                      </div>
                    </div>
                  </div>
                </div>
                
                <div className="bg-gray-800 rounded-lg p-4">
                  <h4 className="text-white font-semibold mb-2">Performance Metrics</h4>
                  <div className="space-y-2">
                    <div className="flex justify-between">
                      <span className="text-gray-400">Accuracy</span>
                      <span className="text-white">{(modelMetrics[selectedModel].accuracy * 100).toFixed(1)}%</span>
                    </div>
                    <div className="flex justify-between">

                      <span className="text-white">{modelMetrics[selectedModel].latency}ms</span>
                    </div>
                    <div className="flex justify-between">

                <div className="bg-gray-800 rounded-lg p-4">
                  <h4 className="text-white font-semibold mb-2">Deployment Status</h4>

                    <div className="flex items-center justify-between">

                      </div>
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-gray-400">Edge</span>
                      <div className="flex items-center gap-2">
                        <div className="w-2 h-2 bg-yellow-500 rounded-full"></div>
                        <span className="text-yellow-400 text-sm">Rolling out</span>
                      </div>
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-gray-400">Version</span>
                      <span className="text-white">{modelMetrics[selectedModel].version}</span>
                    </div>
                  </div>
                </div>
                
                <div className="flex gap-3">
                  <button className="flex-1 px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors">
                    View Metrics
                  </button>
                  <button className="flex-1 px-4 py-2 bg-gray-700 text-gray-300 rounded-lg hover:bg-gray-600 transition-colors">
                    Update Model
                  </button>
                </div>
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  );
};
