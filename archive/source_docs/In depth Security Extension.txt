export class SecurityMiddleware {
  static configure(app: Express): void {
    // Rate limiting with Redis
    app.use(new RateLimiter({
      store: new RedisStore({
        client: redis,
        prefix: 'rl:'
      }),
      points: 100, // requests
      duration: 60, // per minute
      blockDuration: 300, // 5 min block
      keyGenerator: (req) => {
        return req.user?.id || req.ip;
      }
    }).middleware());
    
    // Advanced security headers
    app.use(helmet({
      contentSecurityPolicy: {
        directives: {
          defaultSrc: ["'self'"],
          scriptSrc: ["'self'", "'unsafe-inline'", "https://cdn.jsdelivr.net"],
          styleSrc: ["'self'", "'unsafe-inline'"],
          imgSrc: ["'self'", "data:", "https:"],
          connectSrc: ["'self'", "wss:", "https:"],
          fontSrc: ["'self'"],
          objectSrc: ["'none'"],
          mediaSrc: ["'self'", "blob:"],
          frameSrc: ["'none'"],
        },
      },
      hsts: {
        maxAge: 31536000,
        includeSubDomains: true,
        preload: true
      }
    }));
    
    // Request sanitization
    app.use(mongoSanitize());
    app.use(xss());
    
    // SQL injection prevention
    app.use((req, res, next) => {
      const suspicious = /(\b(union|select|insert|update|delete|drop)\b)/i;
      const checkValue = (value: any): boolean => {
        if (typeof value === 'string') {
          return suspicious.test(value);
        }
        return false;
      };
      
      const hasSQLInjection = Object.values(req.query).some(checkValue) ||
                              Object.values(req.body || {}).some(checkValue);
      
      if (hasSQLInjection) {
        return res.status(400).json({ error: 'Invalid input detected' });
      }
      
      next();
    });
  }
}

// Encryption service
export class EncryptionService {
  private readonly algorithm = 'aes-256-gcm';
  private readonly keyDerivationIterations = 100000;
  
  async encryptFile(
    file: Buffer,
    password: string
  ): Promise<EncryptedFile> {
    const salt = crypto.randomBytes(32);
    const key = await this.deriveKey(password, salt);
    const iv = crypto.randomBytes(16);
    
    const cipher = crypto.createCipheriv(this.algorithm, key, iv);
    
    const encrypted = Buffer.concat([
      cipher.update(file),
      cipher.final()
    ]);
    
    const authTag = cipher.getAuthTag();
    
    return {
      encrypted,
      salt: salt.toString('base64'),
      iv: iv.toString('base64'),
      authTag: authTag.toString('base64')
    };
  }
  
  private async deriveKey(
    password: string,
    salt: Buffer
  ): Promise<Buffer> {
    return new Promise((resolve, reject) => {
      crypto.pbkdf2(
        password,
        salt,
        this.keyDerivationIterations,
        32,
        'sha256',
        (err, derivedKey) => {
          if (err) reject(err);
          else resolve(derivedKey);
        }
      );
    });
  }
}

// Audit logging with immutability
export class AuditLogger {
  private readonly blockchain: HyperledgerFabric;
  
  async log(event: AuditEvent): Promise<void> {
    const entry: AuditEntry = {
      id: uuid(),
      timestamp: new Date().toISOString(),
      actor: event.userId,
      action: event.action,
      resource: event.resourceId,
      result: event.result,
      metadata: event.metadata,
      hash: ''
    };
    
    // Calculate hash including previous entry
    const previousEntry = await this.getLastEntry();
    entry.hash = this.calculateHash(entry, previousEntry?.hash);
    
    // Store in immutable ledger
    await this.blockchain.invoke('createAuditEntry', entry);
    
    // Also store in database for querying
    await this.db.auditLogs.create({ data: entry });
  }
  
  private calculateHash(
    entry: AuditEntry,
    previousHash?: string
  ): string {
    const content = JSON.stringify({
      ...entry,
      previousHash: previousHash || '0'
    });
    
    return crypto
      .createHash('sha256')
      .update(content)
      .digest('hex');
  }
}

export class GDPRCompliance {
  async handleDataExportRequest(userId: string): Promise<void> {
    const exportId = uuid();
    
    await this.queue.add('gdpr-export', {
      userId,
      exportId,
      requestedAt: new Date()
    });
    
    // Process in background
    return exportId;
  }
  
  async processDataExport(job: Job): Promise<void> {
    const { userId, exportId } = job.data;
    
    // Collect all user data
    const userData = await this.collectUserData(userId);
    
    // Create encrypted archive
    const archive = await this.createArchive(userData);
    
    // Upload to secure location
    const downloadUrl = await this.uploadArchive(archive, exportId);
    
    // Notify user
    await this.notifyUser(userId, downloadUrl);
  }
  
  private async collectUserData(userId: string): Promise<UserData> {
    const [
      profile,
      assets,
      activities,
      permissions,
      auditLogs
    ] = await Promise.all([
      this.db.users.findUnique({ where: { id: userId } }),
      this.db.assets.findMany({ where: { uploadedBy: userId } }),
      this.db.activities.findMany({ where: { userId } }),
      this.db.permissions.findMany({ where: { userId } }),
      this.db.auditLogs.findMany({ where: { actor: userId } })
    ]);
    
    return {
      profile,
      assets,
      activities,
      permissions,
      auditLogs
    };
  }
}

export class SOC2Monitor {
  private readonly controls = {
    CC1_1: this.monitorAccessControl,
    CC1_2: this.monitorDataProtection,
    CC1_3: this.monitorIncidentResponse,
    CC1_4: this.monitorChangeManagement,
    CC1_5: this.monitorRiskAssessment
  };
  
  async runComplianceCheck(): Promise<ComplianceReport> {
    const results = await Promise.all(
      Object.entries(this.controls).map(async ([control, monitor]) => ({
        control,
        result: await monitor.call(this),
        timestamp: new Date()
      }))
    );
    
    return {
      overallCompliance: this.calculateCompliance(results),
      controls: results,
      recommendations: this.generateRecommendations(results)
    };
  }
}
