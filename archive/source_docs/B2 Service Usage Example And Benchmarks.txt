// examples/usage.ts
import { B2 } from '@backblaze/b2';
import { createLogger } from 'pino';
import Redis from 'ioredis';
import { B2Service, StorageTierManager, CostOptimizer } from '../b2-service';
import { MonitoringDashboard } from '../monitoring/dashboard';
import { productionConfig } from '../config/production';
import fs from 'fs';
import { pipeline } from 'stream/promises';
import crypto from 'crypto';

// Initialize services
async function initializeServices() {
  // Initialize B2 client
  const b2 = new B2({
    applicationKeyId: productionConfig.b2.applicationKeyId,
    applicationKey: productionConfig.b2.applicationKey
  });
  
  await b2.authorize();
  
  // Initialize logger
  const logger = createLogger({
    level: 'info',
    transport: {
      target: 'pino-pretty',
      options: {
        colorize: true
      }
    }
  });
  
  // Initialize Redis
  const redis = new Redis(productionConfig.redis);
  
  // Initialize services
  const b2Service = new B2Service(b2, logger, redis);
  const tierManager = new StorageTierManager(b2, redis, logger);
  const costOptimizer = new CostOptimizer(redis, logger);
  
  // Initialize monitoring
  const dashboard = new MonitoringDashboard(b2Service, tierManager, costOptimizer);
  
  return { b2Service, tierManager, costOptimizer, dashboard, logger };
}

// Example 1: Upload a large file with progress tracking
async function uploadLargeFileExample(b2Service: B2Service, logger: any) {
  logger.info('Starting large file upload example...');
  
  const fileName = 'large-dataset.bin';
  const fileSize = 1024 * 1024 * 1024 * 5; // 5GB
  
  // Create a readable stream for the file
  const fileStream = fs.createReadStream('/path/to/large/file');
  
  // Track upload progress
  b2Service.on('upload:progress', (progress) => {
    logger.info({
      fileName: progress.fileName,
      progress: `${progress.progress.toFixed(2)}%`,
      uploaded: `${progress.uploadedChunks}/${progress.totalChunks} chunks`
    }, 'Upload progress');
  });
  
  // Handle upload completion
  b2Service.on('upload:complete', (result) => {
    logger.info({
      fileId: result.fileId,
      size: `${(result.size / (1024 ** 3)).toFixed(2)}GB`,
      cost: `$${result.cost.toFixed(4)}`,
      tier: result.tier
    }, 'Upload completed');
  });
  
  try {
    const result = await b2Service.uploadFile(fileStream, fileName, {
      mimeType: 'application/octet-stream',
      size: fileSize.toString(),
      checksum: 'sha256:' + crypto.randomBytes(32).toString('hex'),
      uploadedBy: 'data-pipeline',
      department: 'analytics'
    });
    
    return result;
  } catch (error) {
    logger.error({ error }, 'Upload failed');
    throw error;
  }
}

// Example 2: Concurrent uploads for maximum throughput
async function concurrentUploadsExample(b2Service: B2Service, logger: any) {
  logger.info('Starting concurrent uploads example...');
  
  const files = Array.from({ length: 20 }, (_, i) => ({
    name: `data-chunk-${i}.parquet`,
    size: 100 * 1024 * 1024, // 100MB each
    data: Buffer.alloc(100 * 1024 * 1024, i) // Dummy data
  }));
  
  const startTime = Date.now();
  
  // Upload all files concurrently
  const uploadPromises = files.map(file => 
    b2Service.uploadFile(file.data, file.name, {
      mimeType: 'application/parquet',
      batchId: 'batch-001',
      index: file.name
    })
  );
  
  const results = await Promise.all(uploadPromises);
  
  const duration = (Date.now() - startTime) / 1000;
  const totalSize = files.reduce((sum, f) => sum + f.size, 0);
  const throughput = (totalSize / (1024 ** 2)) / duration; // MB/s
  
  logger.info({
    filesUploaded: results.length,
    totalSize: `${(totalSize / (1024 ** 3)).toFixed(2)}GB`,
    duration: `${duration.toFixed(2)}s`,
    throughput: `${throughput.toFixed(2)}MB/s`
  }, 'Concurrent uploads completed');
  
  return { results, throughput };
}

// Example 3: Storage tier management
async function storageTierExample(tierManager: StorageTierManager, logger: any) {
  logger.info('Starting storage tier management example...');
  
  // Simulate file access patterns
  const assets = [
    { id: 'hot-file-001', accessCount: 50, lastAccessed: new Date() },
    { id: 'warm-file-001', accessCount: 3, lastAccessed: new Date(Date.now() - 45 * 24 * 60 * 60 * 1000) },
    { id: 'cold-file-001', accessCount: 1, lastAccessed: new Date(Date.now() - 120 * 24 * 60 * 60 * 1000) },
    { id: 'archive-file-001', accessCount: 0, lastAccessed: new Date(Date.now() - 400 * 24 * 60 * 60 * 1000) }
  ];
  
  // Record access patterns
  for (const asset of assets) {
    await tierManager.recordAccess(asset.id, 1024 * 1024); // 1MB transfer
  }
  
  // Evaluate and move assets
  for (const asset of assets) {
    await tierManager.evaluateAndMoveAsset(asset.id);
    
    const pattern = await tierManager.getAccessPattern(asset.id);
    logger.info({
      assetId: asset.id,
      tier: pattern.tier,
      lastAccessed: pattern.lastAccessed,
      accessCount: pattern.accessCount
    }, 'Asset tier evaluation');
  }
  
  // Batch evaluation for production use
  logger.info('Running batch tier evaluation...');
  await tierManager.batchEvaluateAssets(100);
}

// Example 5: Circuit breaker resilience testing
async function resilienceTestExample(b2Service: B2Service, logger: any) {
  logger.info('Starting resilience test example...');
  
  // Monitor circuit breaker events
  (b2Service as any).breaker.on('open', () => {
    logger.warn('Circuit breaker opened - B2 API issues detected');
  });
  
  (b2Service as any).breaker.on('halfOpen', () => {
    logger.info('Circuit breaker half-open - testing B2 API recovery');
  });
  
  // Simulate failures by uploading with invalid credentials
  const testFile = Buffer.from('test data');
  const attempts = 20;
  
  for (let i = 0; i < attempts; i++) {
    try {
      await b2Service.uploadFile(testFile, `test-${i}.txt`);
      logger.info(`Upload ${i + 1} succeeded`);
    } catch (error) {
      logger.error(`Upload ${i + 1} failed - circuit breaker state: ${(b2Service as any).breaker.state}`);
    }
    
    // Small delay between attempts
    await new Promise(resolve => setTimeout(resolve, 100));
  }
}

// Benchmark suite
async function runBenchmarks(b2Service: B2Service, logger: any) {
  logger.info('Starting performance benchmarks...');
  
  const benchmarks = {
    smallFiles: { count: 1000, size: 1024 * 1024 }, // 1MB files
    mediumFiles: { count: 100, size: 100 * 1024 * 1024 }, // 100MB files
    largeFiles: { count: 10, size: 1024 * 1024 * 1024 }, // 1GB files
  };
  
  const results: any = {};
  
  for (const [category, config] of Object.entries(benchmarks)) {
    logger.info(`Running ${category} benchmark...`);
    
    const startTime = Date.now();
    const uploads = [];
    
    for (let i = 0; i < config.count; i++) {
      const data = Buffer.alloc(config.size);
      uploads.push(
        b2Service.uploadFile(data, `benchmark-${category}-${i}.bin`, {
          mimeType: 'application/octet-stream',
          benchmark: category
        })
      );
      
      // Limit concurrency for fair testing
      if (uploads.length >= 10) {
        await Promise.race(uploads);
      }
    }
    
    await Promise.all(uploads);
    
    const duration = (Date.now() - startTime) / 1000;
    const totalSize = config.count * config.size;
    const throughput = (totalSize / (1024 ** 2)) / duration;
    
    results[category] = {
      filesUploaded: config.count,
      fileSize: `${(config.size / (1024 ** 2)).toFixed(2)}MB`,
      totalSize: `${(totalSize / (1024 ** 3)).toFixed(2)}GB`,
      duration: `${duration.toFixed(2)}s`,
      throughput: `${throughput.toFixed(2)}MB/s`,
      avgFileTime: `${(duration / config.count).toFixed(2)}s`
    };
    
    logger.info(results[category], `${category} benchmark completed`);
  }
  
  return results;
}

// Main execution
async function main() {
  const { b2Service, tierManager, costOptimizer, dashboard, logger } = await initializeServices();
  
  // Start monitoring dashboard
  dashboard.listen(productionConfig.monitoring.port);
  logger.info(`Monitoring dashboard started on port ${productionConfig.monitoring.port}`);
  
  try {
    // Run examples
    await uploadLargeFileExample(b2Service, logger);
    await concurrentUploadsExample(b2Service, logger);
    await storageTierExample(tierManager, logger);
    await costOptimizationExample(costOptimizer, logger);
    await resilienceTestExample(b2Service, logger);
    
    // Run benchmarks
    const benchmarkResults = await runBenchmarks(b2Service, logger);
    logger.info({ benchmarkResults }, 'All benchmarks completed');
    
  } catch (error) {
    logger.error({ error }, 'Example execution failed');
  }
}

  // Pattern 1: Data pipeline integration

    sourceStream: NodeJS.ReadableStream,

      pipelineId: string;

      timestamp: Date;

    }

      const result = await this.b2Service.uploadFile(sourceStream, fileName, {
        mimeType: this.getMimeType(metadata.format),
        pipelineId: metadata.pipelineId,
        stage: metadata.stage,
        processedAt: metadata.timestamp.toISOString()
      });
      
      this.logger.info({
        fileId: result.fileId,
        pipeline: metadata.pipelineId,
        stage: metadata.stage
      }, 'Pipeline data uploaded');
      
      // Schedule for tier evaluation after 30 days
      setTimeout(() => {
        this.tierManager.evaluateAndMoveAsset(result.fileId);
      }, 30 * 24 * 60 * 60 * 1000);
      
    } catch (error) {
      this.logger.error({
        error,
        pipeline: metadata.pipelineId,
        stage: metadata.stage
      }, 'Pipeline upload failed');
      throw error;
    }
  }
  
  // Pattern 2: Backup with deduplication
  async handleBackupWithDedup(
    data: Buffer,
    backupId: string
  ): Promise<void> {
    const hash = crypto.createHash('sha256').update(data).digest('hex');
    const existingFile = await this.checkDuplicate(hash);
    
    if (existingFile) {
      this.logger.info({
        backupId,
        existingFileId: existingFile,
        savedBytes: data.length
      }, 'Duplicate file detected, skipping upload');
      
      // Just create a reference
      await this.createBackupReference(backupId, existingFile);
      return;
    }
    
    // Upload new file
    const result = await this.b2Service.uploadFile(data, `backup/${backupId}/${hash}`, {
      mimeType: 'application/octet-stream',
      backupId,
      sha256: hash,
      createdAt: new Date().toISOString()
    });
    
    // Store hash for future dedup
    await this.storeFileHash(hash, result.fileId);
  }
  
  // Pattern 3: Multi-region replication

    regions: string[]
  ): Promise<void> {

    const replicaRegions = regions.slice(1);
    
    // Upload to primary region
    const primaryResult = await this.b2Service.uploadFile(data, fileName, {
      mimeType: 'application/octet-stream',
      region: primaryRegion,
      replicated: 'true'
    });
    
    // Async replication to other regions
    const replicationPromises = replicaRegions.map(region => 
      this.replicateToRegion(primaryResult.fileId, data, fileName, region)
    );
    
    // Fire and forget replication
    Promise.all(replicationPromises).catch(error => {
      this.logger.error({ error, fileName }, 'Replication failed');
    });
  }
