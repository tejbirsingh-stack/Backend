# Horizontal Pod Autoscaler
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: noah-api-hpa
  namespace: noah-production
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: noah-api
  minReplicas: 10
  maxReplicas: 100
  metrics:
  - type: Resource
    resource:
      name: cpu
      target:
        type: Utilization
        averageUtilization: 70
  - type: Resource
    resource:
      name: memory
      target:
        type: Utilization
        averageUtilization: 80
  - type: Pods
    pods:
      metric:
        name: http_requests_per_second
      target:
        type: AverageValue
        averageValue: "1000"
  - type: Object
    object:
      metric:
        name: nginx_ingress_controller_requests_rate
      describedObject:
        apiVersion: v1
        kind: Service
        name: noah-api
      target:
        type: Value
        value: "10000"
  behavior:
    scaleDown:
      stabilizationWindowSeconds: 300
      policies:
      - type: Percent
        value: 50
        periodSeconds: 60
      - type: Pods
        value: 5
        periodSeconds: 60
      selectPolicy: Min
    scaleUp:
      stabilizationWindowSeconds: 60
      policies:
      - type: Percent
        value: 100
        periodSeconds: 30
      - type: Pods
        value: 10
        periodSeconds: 30
      selectPolicy: Max

# PrometheusRule for Alerts
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: noah-api-alerts
  namespace: noah-production
spec:
  groups:
  - name: noah-api
    interval: 30s
    rules:
    - alert: NoahAPIHighErrorRate
      expr: |
        (
          sum(rate(noah_http_requests_total{status=~"5.."}[5m])) by (job)
          /
          sum(rate(noah_http_requests_total[5m])) by (job)
        ) > 0.05
      for: 5m
      labels:
        severity: critical
        service: noah-api
      annotations:
        summary: "High error rate on Noah API"
        description: "Error rate is {{ $value | humanizePercentage }} for job {{ $labels.job }}"
        runbook_url: "https://runbooks.noah.app/api-high-error-rate"
    
    - alert: NoahAPIHighLatency
      expr: |
        histogram_quantile(0.95,
          sum(rate(noah_http_request_duration_seconds_bucket[5m])) by (le, job)
        ) > 1
      for: 5m
      labels:
        severity: warning
        service: noah-api
      annotations:
        summary: "High latency on Noah API"
        description: "95th percentile latency is {{ $value }}s for job {{ $labels.job }}"
        runbook_url: "https://runbooks.noah.app/api-high-latency"
    
    - alert: NoahAPIPodCrashLooping
      expr: |
        rate(kube_pod_container_status_restarts_total{namespace="noah-production",pod=~"noah-api-.*"}[5m]) > 0
      for: 5m
      labels:
        severity: critical
        service: noah-api
      annotations:
        summary: "Noah API pod is crash looping"
        description: "Pod {{ $labels.pod }} has restarted {{ $value }} times in the last 5 minutes"
        runbook_url: "https://runbooks.noah.app/pod-crash-looping"
    
    - alert: NoahCompressionQueueBacklog
      expr: |
        noah_compression_queue_size > 1000
      for: 15m
      labels:
        severity: warning
        service: noah-worker
      annotations:
        summary: "Compression queue backlog detected"
        description: "Compression queue has {{ $value }} items pending"
        runbook_url: "https://runbooks.noah.app/compression-queue-backlog"
    
    - alert: NoahStorageUsageHigh
      expr: |
        (noah_storage_bytes_used / noah_storage_bytes_quota) > 0.9
      for: 30m
      labels:
        severity: warning
        service: noah-storage
      annotations:
        summary: "Storage usage is high"
        description: "Storage usage is at {{ $value | humanizePercentage }} of quota"
        runbook_url: "https://runbooks.noah.app/storage-usage-high"
