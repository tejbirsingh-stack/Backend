// Query Result Streaming
export class StreamingSearchService {
  async *searchStream(query: SearchQuery): AsyncGenerator<MediaAsset> {
    const scrollTimeout = '1m';
    let scrollId = null;
    
    try {
      const initialResponse = await this.client.search({
        index: 'media_assets',
        scroll: scrollTimeout,
        size: 100,
        body: this.buildQuery(query)
      });
      
      scrollId = initialResponse._scroll_id;
      yield* initialResponse.hits.hits.map(hit => hit._source);
      
      while (true) {
        const scrollResponse = await this.client.scroll({
          scroll_id: scrollId,
          scroll: scrollTimeout
        });
        
        if (scrollResponse.hits.hits.length === 0) break;
        
        yield* scrollResponse.hits.hits.map(hit => hit._source);
      }
    } finally {
      if (scrollId) {
        await this.client.clearScroll({ scroll_id: scrollId });
      }
    }
  }
}`;
