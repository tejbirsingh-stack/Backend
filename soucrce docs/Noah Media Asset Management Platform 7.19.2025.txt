-- Create monthly partitions for the next 2 years
DO $$
DECLARE
    start_date DATE := '2024-01-01';
    end_date DATE := '2026-01-01';
    partition_date DATE;
    partition_name TEXT;
BEGIN
    partition_date := start_date;
    WHILE partition_date < end_date LOOP
        partition_name := 'media_assets_' || to_char(partition_date, 'YYYY_MM');
        EXECUTE format('CREATE TABLE %I PARTITION OF media_assets FOR VALUES FROM (%L) TO (%L)',
            partition_name,
            partition_date,
            partition_date + INTERVAL '1 month'
        );
        partition_date := partition_date + INTERVAL '1 month';
    END LOOP;
END $$;

-- Storage quota enforcement function
CREATE OR REPLACE FUNCTION check_storage_quota()
RETURNS TRIGGER AS $$
DECLARE
    current_usage BIGINT;
    quota BIGINT;
BEGIN
    SELECT storage_used_bytes, storage_quota_bytes INTO current_usage, quota
    FROM organizations WHERE id = NEW.org_id;
    
    IF current_usage + NEW.file_size > quota THEN
        RAISE EXCEPTION 'Storage quota exceeded';
    END IF;
    
    -- Update organization storage usage
    UPDATE organizations 
    SET storage_used_bytes = storage_used_bytes + NEW.file_size
    WHERE id = NEW.org_id;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER enforce_storage_quota
    BEFORE INSERT ON media_assets
    FOR EACH ROW
    EXECUTE FUNCTION check_storage_quota();

-- Materialized view for asset statistics
CREATE MATERIALIZED VIEW asset_statistics AS
SELECT 
    org_id,
    COUNT(*) as total_assets,
    SUM(file_size) as total_size_bytes,
    SUM(original_size - file_size) as total_saved_bytes,
    AVG(compression_ratio) as avg_compression_ratio,
    COUNT(DISTINCT mime_type) as unique_mime_types,
    jsonb_object_agg(
        mime_type, 
        COUNT(*)
    ) as assets_by_type
FROM media_assets
WHERE deleted_at IS NULL
GROUP BY org_id;

CREATE UNIQUE INDEX ON asset_statistics(org_id);

-- Refresh materialized view every hour
CREATE OR REPLACE FUNCTION refresh_asset_statistics()
RETURNS void AS $$
BEGIN
    REFRESH MATERIALIZED VIEW CONCURRENTLY asset_statistics;
END;
$$ LANGUAGE plpgsql;

3. Storage Architecture
Backblaze B2 Integration
python# storage_service.py
import asyncio
import hashlib
from typing import Optional, Dict, Any
import boto3
from botocore.client import Config
import aiofiles
from fastapi import FastAPI, UploadFile, HTTPException
from pydantic import BaseModel
import redis.asyncio as redis

class StorageConfig(BaseModel):
    b2_application_key_id: str
    b2_application_key: str
    b2_bucket_name: str
    b2_endpoint: str = "https://s3.us-west-001.backblazeb2.com"
    cdn_base_url: str
    chunk_size: int = 5 * 1024 * 1024  # 5MB chunks

class StorageService:
    def __init__(self, config: StorageConfig):
        self.config = config
        self.s3_client = boto3.client(
            's3',
            endpoint_url=config.b2_endpoint,
            aws_access_key_id=config.b2_application_key_id,
            aws_secret_access_key=config.b2_application_key,
            config=Config(signature_version='s3v4')
        )
        self.redis_client = redis.Redis(
            host='redis-cluster.noah.svc.cluster.local',
            decode_responses=True
        )
    
    async def upload_file(
        self, 
        file: UploadFile, 
        org_id: str,
        user_id: str
    ) -> Dict[str, Any]:
        """
        Upload file with intelligent tiering and compression
        """
        file_key = f"{org_id}/{user_id}/{file.filename}"
        
        # Calculate checksum
        sha256_hash = hashlib.sha256()
        temp_file_path = f"/tmp/{file.filename}"
        
        async with aiofiles.open(temp_file_path, 'wb') as f:
            while chunk := await file.read(self.config.chunk_size):
                await f.write(chunk)
                sha256_hash.update(chunk)
        
        checksum = sha256_hash.hexdigest()
        
        # Check for duplicate
        existing = await self.redis_client.get(f"checksum:{checksum}")
        if existing:
            return {"duplicate": True, "file_id": existing}
        
        # Multipart upload for large files
        if file.size > 100 * 1024 * 1024:  # 100MB
            upload_id = self._init_multipart_upload(file_key)
            parts = await self._upload_parts(temp_file_path, file_key, upload_id)
            response = self._complete_multipart_upload(file_key, upload_id, parts)
        else:
            response = await self._simple_upload(temp_file_path, file_key)
        
        # Store metadata in Redis
        await self.redis_client.setex(
            f"checksum:{checksum}",
            86400 * 30,  # 30 days
            response['file_id']
        )
        
        return {
            "file_id": response['file_id'],
            "file_key": file_key,
            "checksum": checksum,
            "cdn_url": f"{self.config.cdn_base_url}/{file_key}",
            "size": file.size
        }
    
    async def set_lifecycle_policy(self, file_key: str, days_until_archive: int = 90):
        """
        Set intelligent tiering policy
        """
        lifecycle_config = {
            'Rules': [{
                'ID': f'archive-{file_key}',
                'Status': 'Enabled',
                'Transitions': [
                    {
                        'Days': days_until_archive,
                        'StorageClass': 'GLACIER'
                    }
                ]
            }]
        }
        
        self.s3_client.put_bucket_lifecycle_configuration(
            Bucket=self.config.b2_bucket_name,
            LifecycleConfiguration=lifecycle_config
        )

// Custom span example
function uploadAsset(file, orgId) {
  return tracer.trace('asset.upload', { 
    resource: file.name,
    type: 'web',
    meta: {
      'org.id': orgId,
      'file.size': file.size,
      'file.type': file.mimetype
    }
  }, async (span) => {
    try {
      // Set custom tags
      span.setTag('file.size', file.size);
      span.setTag('compression.enabled', true);
      
      // Trace sub-operations
      const compressionSpan = tracer.startSpan('asset.compress', {
        childOf: span
      });
      
      const compressed = await compressFile(file);
      compressionSpan.setTag('compression.ratio', compressed.ratio);
      compressionSpan.finish();
      
      // Upload to storage
      const uploadSpan = tracer.startSpan('storage.upload', {
        childOf: span
      });
      
      const result = await storageService.upload(compressed);
      uploadSpan.setTag('storage.provider', 'backblaze');
      uploadSpan.setTag('cdn.enabled', true);
      uploadSpan.finish();
      
      return result;
    } catch (error) {
      span.setTag('error', true);
      span.setTag('error.message', error.message);
      throw error;
    }
  });
}
