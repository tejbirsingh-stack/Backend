// Integration Test Suite

import { describe, it, expect, beforeAll, afterAll } from '@jest/globals';
import request from 'supertest';
import { app } from '../src/app';
import { stripe } from '../src/services/stripe';
import { redis } from '../src/services/redis';

describe('Enterprise Integration Tests', () => {
  let authToken: string;
  let organizationId: string;
  let customerId: string;
  let subscriptionId: string;
  
  beforeAll(async () => {
    // Setup test data
    const response = await request(app)
      .post('/auth/login')
      .send({
        email: 'test@example.com',
        password: 'testpassword'
      });
    
    authToken = response.body.token;
    organizationId = response.body.user.organizationId;
    
    // Create Stripe customer
    const customer = await stripe.customers.create({
      email: 'test@example.com',
      metadata: { organizationId }
    });
    customerId = customer.id;
  });
  
  afterAll(async () => {
    // Cleanup
    if (subscriptionId) {
      await stripe.subscriptions.cancel(subscriptionId);
    }
    if (customerId) {
      await stripe.customers.del(customerId);
    }
    await redis.quit();
  });
  
  describe('Billing Integration', () => {
    it('should create a subscription', async () => {
      const response = await request(app)
        .post('/billing/subscriptions')
        .set('Authorization', `Bearer ${authToken}`)
        .send({
          customerId,
          planId: 'price_test_plan',
          trialDays: 14,
          metadata: { organizationId }
        });
      
      expect(response.status).toBe(201);
      expect(response.body).toHaveProperty('id');
      expect(response.body.status).toBe('trialing');
      
      subscriptionId = response.body.id;
    });
    
    it('should record usage', async () => {
      const response = await request(app)
        .post('/billing/usage')
        .set('Authorization', `Bearer ${authToken}`)
        .send({
          subscriptionItemId: 'si_test_item',
          quantity: 1024,
          metadata: {
            resource: 'storage',
            unit: 'mb'
          }
        });
      
      expect(response.status).toBe(201);
    });
    
    it('should generate usage report', async () => {
      const response = await request(app)
        .get('/billing/usage-report')
        .set('Authorization', `Bearer ${authToken}`)
        .query({
          organizationId,
          startDate: '2024-01-01',
          endDate: '2024-01-31'
        });
      
      expect(response.status).toBe(200);
      expect(response.body).toHaveProperty('usage');
      expect(response.body).toHaveProperty('totalCost');
    });
    
    it('should handle payment method updates', async () => {
      // Create payment method
      const paymentMethod = await stripe.paymentMethods.create({
        type: 'card',
        card: {
          token: 'tok_visa'
        }
      });
      
      const response = await request(app)
        .post(`/billing/payment-methods`)
        .set('Authorization', `Bearer ${authToken}`)
        .send({
          customerId,
          paymentMethodId: paymentMethod.id
        });
      
      expect(response.status).toBe(200);
    });
  });
  
  describe('SSO Integration', () => {
    it('should configure SAML provider', async () => {
      const response = await request(app)
        .post('/sso/providers')
        .set('Authorization', `Bearer ${authToken}`)

        signature,
        secret,
        timestamp
      );
      
      expect(isValid).toBe(true);
    });
    
    it('should handle webhook delivery failures', async () => {
      // Mock failed endpoint
      nock('https://test.example.com')
        .post('/webhooks')
        .reply(500);
      
      await webhookService.emit({
        id: 'test_event',
        type: 'asset.created',
        data: { test: true },
        timestamp: new Date(),
        version: '1.0'
      });
      
      // Wait for retry
      await new Promise(resolve => setTimeout(resolve, 6000));
      
      // Check delivery status
      const stats = await webhookService.getWebhookStats(endpointId);
      expect(stats.failed).toBeGreaterThan(0);
    });
  });
  
  describe('Rate Limiting', () => {
    it('should enforce API rate limits', async () => {
      const requests = Array(101).fill(null).map(() =>
        request(app)
          .get('/billing/subscriptions')
          .set('Authorization', `Bearer ${authToken}`)
      );
      
      const responses = await Promise.all(requests);
      const rateLimited = responses.filter(r => r.status === 429);
      
      expect(rateLimited.length).toBeGreaterThan(0);
    });
  });
  
  describe('Error Handling', () => {
    it('should handle Stripe webhook errors gracefully', async () => {
      const response = await request(app)
        .post('/webhooks/stripe')
        .send({
          type: 'invalid.event',
          data: {}
        })
        .set('stripe-signature', 'invalid');
      
      expect(response.status).toBe(400);
    });
    
    it('should handle SSO errors', async () => {

        .send({
          SAMLResponse: 'invalid_response'
        });
      
      expect(response.status).toBe(401);
    });
  });

// API Client SDK

export class NoahEnterpriseClient {
  private apiKey: string;
  private baseURL: string;
  
  constructor(config: { apiKey: string; baseURL?: string }) {
    this.apiKey = config.apiKey;
    this.baseURL = config.baseURL || 'https://api.noah.app/v1';
  }
  
  // Billing methods
  billing = {
    createSubscription: async (data: CreateSubscriptionRequest) => {
      return this.request('POST', '/billing/subscriptions', data);
    },
    
    updateSubscription: async (id: string, data: UpdateSubscriptionRequest) => {
      return this.request('PATCH', `/billing/subscriptions/${id}`, data);
    },
    
    cancelSubscription: async (id: string, immediately = false) => {
      return this.request('DELETE', `/billing/subscriptions/${id}`, null, {
        immediately: immediately.toString()
      });
    },
    
    recordUsage: async (data: RecordUsageRequest) => {
      return this.request('POST', '/billing/usage', data);
    },
    
    getUsageReport: async (params: UsageReportParams) => {
      return this.request('GET', '/billing/usage-report', null, params);
    }
  };
  
  // SSO methods
  sso = {
    configureProvider: async (data: ConfigureProviderRequest) => {
      return this.request('POST', '/sso/providers', data);
    },
    
    listProviders: async () => {
      return this.request('GET', '/sso/providers');
    },
    
    removeProvider: async (id: string) => {
      return this.request('DELETE', `/sso/providers/${id}`);
    }
  };
  
  // Webhook methods
  webhooks = {
    register: async (data: RegisterWebhookRequest) => {
      return this.request('POST', '/webhooks/endpoints', data);
    },
    
    list: async () => {
      return this.request('GET', '/webhooks/endpoints');
    },
    
    test: async (id: string) => {
      return this.request('POST', `/webhooks/endpoints/${id}/test`);
    },
    
    getDeliveries: async (id: string, params?: PaginationParams) => {
      return this.request('GET', `/webhooks/endpoints/${id}/deliveries`, null, params);
    }
  };
  
  private async request(
    method: string,
    path: string,
    data?: any,
    params?: any
  ): Promise<any> {
    const url = new URL(`${this.baseURL}${path}`);
    
    if (params) {
      Object.entries(params).forEach(([key, value]) => {
        url.searchParams.append(key, value);
      });
    }
    
    const response = await fetch(url.toString(), {
      method,
      headers: {
        'Content-Type': 'application/json',
        'X-API-Key': this.apiKey
      },
      body: data ? JSON.stringify(data) : undefined
    });
    
    if (!response.ok) {
      throw new Error(`API request failed: ${response.statusText}`);
    }
    
    return response.json();
  }
}
