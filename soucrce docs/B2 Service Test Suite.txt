import { describe, it, expect, jest, beforeEach, afterEach } from '@jest/globals';
import { B2Service, StorageTierManager, CostOptimizer } from './b2-service';
import { B2 } from '@backblaze/b2';
import { Logger } from 'pino';
import Redis from 'ioredis-mock';
import { EventEmitter } from 'events';
import { Readable } from 'stream';

// Mock dependencies
jest.mock('@backblaze/b2');
jest.mock('pino');

describe('B2Service', () => {
  let b2Service: B2Service;
  let mockB2: jest.Mocked<B2>;
  let mockLogger: jest.Mocked<Logger>;
  let mockRedis: Redis;

  beforeEach(() => {
    mockB2 = new B2() as jest.Mocked<B2>;
    mockLogger = {
      info: jest.fn(),
      error: jest.fn(),
      warn: jest.fn(),
      debug: jest.fn()
    } as any;
    mockRedis = new Redis();
    
    process.env.B2_BUCKET_ID = 'test-bucket-id';
    
    b2Service = new B2Service(mockB2, mockLogger, mockRedis);
  });

  afterEach(() => {
    jest.clearAllMocks();
  });

  describe('uploadFile', () => {
    it('should upload small files directly', async () => {
      const file = Buffer.from('test content');
      const fileName = 'test.txt';
      const metadata = { mimeType: 'text/plain' };
      
      mockB2.getUploadUrl.mockResolvedValue({
        uploadUrl: 'https://upload.url',
        authorizationToken: 'auth-token'
      });
      
      mockB2.uploadFile.mockResolvedValue({
        fileId: 'file-123',
        fileName,
        uploadTimestamp: Date.now()
      });
      
      const result = await b2Service.uploadFile(file, fileName, metadata);
      
      expect(result).toMatchObject({
        fileId: 'file-123',
        fileName,
        size: file.length
      });
      
      expect(mockB2.uploadFile).toHaveBeenCalledWith(
        expect.objectContaining({
          fileName,
          data: file
        })
      );
    });

    it('should use multi-part upload for large files', async () => {
      const largeFile = Buffer.alloc(200 * 1024 * 1024); // 200MB
      const fileName = 'large-file.bin';
      
      mockB2.startLargeFile.mockResolvedValue({
        fileId: 'large-file-123'
      });
      
      mockB2.getUploadUrl.mockResolvedValue({
        uploadUrl: 'https://upload.url',
        authorizationToken: 'auth-token'
      });
      
      mockB2.uploadPart.mockResolvedValue({});
      mockB2.finishLargeFile.mockResolvedValue({});
      
      const result = await b2Service.uploadFile(largeFile, fileName);
      
      expect(mockB2.startLargeFile).toHaveBeenCalled();
      expect(mockB2.uploadPart).toHaveBeenCalledTimes(2); // 200MB = 2x100MB chunks
      expect(mockB2.finishLargeFile).toHaveBeenCalled();
      
      expect(result).toMatchObject({
        fileId: 'large-file-123',
        fileName,
        size: largeFile.length
      });
    });

    it('should handle upload failures with circuit breaker', async () => {
      const file = Buffer.from('test');
      const fileName = 'test.txt';
      
      mockB2.getUploadUrl.mockRejectedValue(new Error('API Error'));
      
      // Trigger multiple failures to open circuit
      const attempts = 15;
      const promises = [];
      
      for (let i = 0; i < attempts; i++) {
        promises.push(
          b2Service.uploadFile(file, fileName).catch(() => {})
        );
      }
      
      await Promise.all(promises);
      
      // Circuit should be open now
      await expect(b2Service.uploadFile(file, fileName))
        .rejects.toThrow();
      
      expect(mockLogger.error).toHaveBeenCalledWith(
        expect.stringContaining('Circuit breaker opened')
      );
    });

    it('should emit progress events for large file uploads', async () => {
      const largeFile = Buffer.alloc(200 * 1024 * 1024);
      const fileName = 'large-file.bin';
      const progressEvents: any[] = [];
      
      b2Service.on('upload:progress', (event) => {
        progressEvents.push(event);
      });
      
      mockB2.startLargeFile.mockResolvedValue({ fileId: 'file-123' });
      mockB2.getUploadUrl.mockResolvedValue({
        uploadUrl: 'https://upload.url',
        authorizationToken: 'auth-token'
      });
      mockB2.uploadPart.mockResolvedValue({});
      mockB2.finishLargeFile.mockResolvedValue({});
      
      await b2Service.uploadFile(largeFile, fileName);
      
      expect(progressEvents).toHaveLength(2);
      expect(progressEvents[1]).toMatchObject({
        fileName,
        progress: 100,
        uploadedChunks: 2,
        totalChunks: 2
      });
    });

    it('should cancel large file upload on error', async () => {
      const largeFile = Buffer.alloc(200 * 1024 * 1024);
      const fileName = 'large-file.bin';
      
      mockB2.startLargeFile.mockResolvedValue({ fileId: 'file-123' });
      mockB2.uploadPart.mockRejectedValue(new Error('Upload part failed'));
      mockB2.cancelLargeFile.mockResolvedValue({});
      
      await expect(b2Service.uploadFile(largeFile, fileName))
        .rejects.toThrow('Upload part failed');
      
      expect(mockB2.cancelLargeFile).toHaveBeenCalledWith({
        fileId: 'file-123'
      });
    });
  });

  describe('Stream processing', () => {
    it('should handle stream uploads efficiently', async () => {
      const chunks = [
        Buffer.from('chunk1'),
        Buffer.from('chunk2'),
        Buffer.from('chunk3')
      ];
      
      const stream = new Readable({
        read() {
          this.push(chunks.shift() || null);
        }
      });
      
      const metadata = {
        mimeType: 'application/octet-stream',
        size: '18' // Total size of chunks
      };
      
      mockB2.getUploadUrl.mockResolvedValue({
        uploadUrl: 'https://upload.url',
        authorizationToken: 'auth-token'
      });
      
      mockB2.uploadFile.mockResolvedValue({
        fileId: 'stream-file-123',
        fileName: 'stream.dat'
      });
      
      const result = await b2Service.uploadFile(stream, 'stream.dat', metadata);
      
      expect(result).toMatchObject({
        fileId: 'stream-file-123',
        fileName: 'stream.dat'
      });
    });
  });
});

describe('StorageTierManager', () => {
  let tierManager: StorageTierManager;
  let mockB2: jest.Mocked<B2>;
  let mockRedis: Redis;
  let mockLogger: jest.Mocked<Logger>;

  beforeEach(() => {
    mockB2 = new B2() as jest.Mocked<B2>;
    mockRedis = new Redis();
    mockLogger = {
      info: jest.fn(),
      error: jest.fn(),
      warn: jest.fn()
    } as any;
    
    tierManager = new StorageTierManager(mockB2, mockRedis, mockLogger);
  });

  describe('evaluateAndMoveAsset', () => {
    it('should move old unused assets to cold storage', async () => {
      const assetId = 'asset-123';
      const oldDate = new Date();
      oldDate.setDate(oldDate.getDate() - 100); // 100 days ago
      
      await mockRedis.hset(`access:${assetId}`, {
        lastAccessed: oldDate.getTime().toString(),
        accessCount: '2',
        totalBandwidthUsed: '1000000',
        tier: 'hot'
      });
      
      mockB2.updateFileInfo.mockResolvedValue({});
      
      await tierManager.evaluateAndMoveAsset(assetId);
      
      expect(mockB2.updateFileInfo).toHaveBeenCalledWith({
        fileId: assetId,
        fileInfo: expect.objectContaining({
          tier: 'cold'
        })
      });
      
      const tier = await mockRedis.hget(`access:${assetId}`, 'tier');
      expect(tier).toBe('cold');
    });

    it('should move frequently accessed files to hot storage', async () => {
      const assetId = 'asset-456';
      const recentDate = new Date();
      recentDate.setDate(recentDate.getDate() - 5); // 5 days ago
      
      await mockRedis.hset(`access:${assetId}`, {
        lastAccessed: recentDate.getTime().toString(),
        accessCount: '50',
        totalBandwidthUsed: '50000000',
        tier: 'cold'
      });
      
      mockB2.updateFileInfo.mockResolvedValue({});
      
      await tierManager.evaluateAndMoveAsset(assetId);
      
      expect(mockB2.updateFileInfo).toHaveBeenCalledWith({
        fileId: assetId,
        fileInfo: expect.objectContaining({
          tier: 'hot'
        })
      });
    });

    it('should archive very old assets', async () => {
      const assetId = 'asset-789';
      const veryOldDate = new Date();
      veryOldDate.setDate(veryOldDate.getDate() - 400); // 400 days ago
      
      await mockRedis.hset(`access:${assetId}`, {
        lastAccessed: veryOldDate.getTime().toString(),
        accessCount: '1',
        totalBandwidthUsed: '100000',
        tier: 'cold'
      });
      
      mockB2.updateFileInfo.mockResolvedValue({});
      
      await tierManager.evaluateAndMoveAsset(assetId);
      
      expect(mockB2.updateFileInfo).toHaveBeenCalledWith({
        fileId: assetId,
        fileInfo: expect.objectContaining({
          tier: 'archive'
        })
      });
    });
  });

  describe('recordAccess', () => {
    it('should update access patterns correctly', async () => {
      const assetId = 'asset-999';
      const bytesTransferred = 1024 * 1024; // 1MB
      
      await tierManager.recordAccess(assetId, bytesTransferred);
      await tierManager.recordAccess(assetId, bytesTransferred);
      
      const accessPattern = await tierManager.getAccessPattern(assetId);
      
      expect(accessPattern.accessCount).toBe(2);
      expect(accessPattern.totalBandwidthUsed).toBe(bytesTransferred * 2);
      expect(accessPattern.lastAccessed.getTime()).toBeCloseTo(Date.now(), -2);
    });
  });
});

describe('CostOptimizer', () => {
  let costOptimizer: CostOptimizer;
  let mockRedis: Redis;
  let mockLogger: jest.Mocked<Logger>;

  beforeEach(() => {
    mockRedis = new Redis();
    mockLogger = {
      warn: jest.fn(),
      info: jest.fn()
    } as any;
    
    costOptimizer = new CostOptimizer(mockRedis, mockLogger);
  });

  describe('predictMonthlyCost', () => {
    it('should predict monthly costs based on usage trend', async () => {
      // Set up 30 days of usage history with increasing trend
      const baseUsage = 100;
      const dailyIncrease = 5;
      
      for (let i = 0; i < 30; i++) {
        const date = new Date();
        date.setDate(date.getDate() - (29 - i));
        const key = `usage:${date.toISOString().split('T')[0]}`;
        await mockRedis.set(key, (baseUsage + i * dailyIncrease).toString());
      }
      
      const predictedCost = await costOptimizer.predictMonthlyCost();
      
      // Should predict higher than current average due to upward trend
      expect(predictedCost).toBeGreaterThan(baseUsage);
    });
  });

  describe('checkBudgetAlerts', () => {
    it('should trigger alert when predicted cost exceeds threshold', async () => {
      // Set up usage data that will exceed budget
      const highUsage = 1000;
      const date = new Date();
      const key = `usage:${date.toISOString().split('T')[0]}`;
      await mockRedis.set(key, highUsage.toString());
      
      const publishSpy = jest.spyOn(mockRedis, 'publish');
      
      await costOptimizer.checkBudgetAlerts(500); // Low budget
      
      expect(mockLogger.warn).toHaveBeenCalledWith(
        expect.objectContaining({
          percentage: expect.any(Number)
        }),
        'Monthly budget alert triggered'
      );
      
      expect(publishSpy).toHaveBeenCalledWith(
        'budget:alert',
        expect.stringContaining('threshold_exceeded')
      );
    });
  });

  describe('getOptimizationRecommendations', () => {
    it('should provide relevant optimization recommendations', async () => {
      const recommendations = await costOptimizer.getOptimizationRecommendations();
      
      expect(recommendations).toBeInstanceOf(Array);
      expect(recommendations.length).toBeGreaterThan(0);
      
      // Should include specific actionable recommendations
      const hasStorageRecommendation = recommendations.some(r => 
        r.includes('cold data') || r.includes('archive tier')
      );
      expect(hasStorageRecommendation).toBe(true);
    });
  });
});

describe('Performance Benchmarks', () => {
  let b2Service: B2Service;
  let mockB2: jest.Mocked<B2>;
  let mockLogger: jest.Mocked<Logger>;
  let mockRedis: Redis;

  beforeEach(() => {
    mockB2 = new B2() as jest.Mocked<B2>;
    mockLogger = { info: jest.fn(), error: jest.fn() } as any;
    mockRedis = new Redis();
    
    b2Service = new B2Service(mockB2, mockLogger, mockRedis);
    
    // Mock successful responses
    mockB2.getUploadUrl.mockResolvedValue({
      uploadUrl: 'https://upload.url',
      authorizationToken: 'auth-token'
    });
    mockB2.uploadPart.mockResolvedValue({});
    mockB2.startLargeFile.mockResolvedValue({ fileId: 'file-123' });
    mockB2.finishLargeFile.mockResolvedValue({});
  });

  it('should handle 1GB file upload efficiently', async () => {
    const fileSize = 1024 * 1024 * 1024; // 1GB
    const file = Buffer.alloc(fileSize);
    
    const startTime = Date.now();
    await b2Service.uploadFile(file, 'large-file.bin');
    const duration = Date.now() - startTime;
    
    // Should complete within reasonable time (mocked, so should be fast)
    expect(duration).toBeLessThan(1000);
    
    // Should use optimal number of chunks (10 x 100MB)
    expect(mockB2.uploadPart).toHaveBeenCalledTimes(10);
  });

  it('should maintain throughput with concurrent uploads', async () => {
    const fileSize = 100 * 1024 * 1024; // 100MB
    const concurrentUploads = 10;
    
    const uploads = Array(concurrentUploads).fill(null).map((_, i) => 
      b2Service.uploadFile(
        Buffer.alloc(fileSize),
        `concurrent-${i}.bin`
      )
    );
    
    const startTime = Date.now();
    await Promise.all(uploads);
    const duration = Date.now() - startTime;
    
    // All uploads should complete efficiently
    expect(duration).toBeLessThan(2000);
  });
});
