// monitoring/dashboard.ts
import express from 'express';
import { register } from 'prom-client';
import { B2Service, StorageTierManager, CostOptimizer } from '../b2-service';

export class MonitoringDashboard {
  private app: express.Application;
  
  constructor(
    private b2Service: B2Service,
    private tierManager: StorageTierManager,
    private costOptimizer: CostOptimizer
  ) {
    this.app = express();
    this.setupRoutes();
  }
  
  private setupRoutes() {
    // Prometheus metrics endpoint
    this.app.get('/metrics', async (req, res) => {
      res.set('Content-Type', register.contentType);
      res.end(await register.metrics());
    });
    
    // Health check endpoint
    this.app.get('/health', async (req, res) => {
      const health = await this.getHealthStatus();
      res.status(health.status === 'healthy' ? 200 : 503).json(health);
    });
    
    // Real-time dashboard data
    this.app.get('/api/dashboard', async (req, res) => {
      const dashboard = await this.getDashboardData();
      res.json(dashboard);
    });
    
    // Cost analysis endpoint
    this.app.get('/api/costs', async (req, res) => {
      const costs = await this.getCostAnalysis();
      res.json(costs);
    });
    
    // Storage tier distribution
    this.app.get('/api/storage-tiers', async (req, res) => {
      const tiers = await this.getStorageTierDistribution();
      res.json(tiers);
    });
  }
  
  private async getHealthStatus() {
    const circuitBreakerState = (this.b2Service as any).breaker.state;
    const isHealthy = circuitBreakerState !== 'open';
    
    return {
      status: isHealthy ? 'healthy' : 'unhealthy',
      timestamp: new Date().toISOString(),
      components: {
        b2_api: {
          status: isHealthy ? 'up' : 'down',
          circuit_breaker: circuitBreakerState
        },
        redis: {
          status: 'up' // Would check Redis connection
        }
      }
    };
  }
  
  private async getDashboardData() {
    const metrics = await register.getMetricsAsJSON();
    const costMetrics = await (this.b2Service as any).getCostMetrics();
    const predictions = await this.costOptimizer.predictMonthlyCost();
    
    return {
      realtime: {
        concurrent_uploads: this.getMetricValue(metrics, 'b2_concurrent_uploads'),
        upload_rate: this.getUploadRate(metrics),
        error_rate: this.getErrorRate(metrics),
        circuit_breaker_state: (this.b2Service as any).breaker.state
      },
      costs: {
        current: costMetrics,
        predicted_monthly: predictions,
        daily_average: costMetrics.total / new Date().getDate()
      },
      performance: {
        avg_upload_duration: this.getAverageUploadDuration(metrics),
        throughput_mbps: this.getThroughput(metrics)
      }
    };
  }
  
  private async getCostAnalysis() {
    const costMetrics = await (this.b2Service as any).getCostMetrics();
    const recommendations = await this.costOptimizer.getOptimizationRecommendations();
    const prediction = await this.costOptimizer.predictMonthlyCost();
    
    return {
      current_costs: costMetrics,
      predicted_monthly: prediction,
      cost_breakdown: {
        storage_percentage: (costMetrics.storage / costMetrics.total) * 100,
        bandwidth_percentage: (costMetrics.bandwidth / costMetrics.total) * 100,
        transactions_percentage: (costMetrics.transactions / costMetrics.total) * 100
      },
      optimization_recommendations: recommendations,
      potential_savings: this.calculatePotentialSavings(recommendations)
    };
  }
  
  private async getStorageTierDistribution() {
    // Would query actual data - this is example distribution
    return {
      tiers: {
        hot: { size_gb: 2500, count: 15000, percentage: 40 },
        warm: { size_gb: 3000, count: 25000, percentage: 30 },
        cold: { size_gb: 4000, count: 50000, percentage: 25 },
        archive: { size_gb: 1500, count: 10000, percentage: 5 }
      },
      migration_candidates: 1250,
      last_evaluation: new Date().toISOString()
    };
  }
  
  private getMetricValue(metrics: any[], name: string): number {
    const metric = metrics.find(m => m.name === name);
    return metric?.values?.[0]?.value || 0;
  }
  
  private getUploadRate(metrics: any[]): number {
    const uploads = metrics.find(m => m.name === 'b2_uploads_total');
    // Calculate rate from counter increase
    return 0; // Placeholder
  }
  
  private getErrorRate(metrics: any[]): number {
    const uploads = metrics.find(m => m.name === 'b2_uploads_total');
    const failures = uploads?.values?.filter(v => v.labels.status === 'failure') || [];
    const total = uploads?.values?.length || 1;
    return (failures.length / total) * 100;
  }
  
  private getAverageUploadDuration(metrics: any[]): number {
    const histogram = metrics.find(m => m.name === 'b2_upload_duration_seconds');
    return histogram?.values?.[0]?.metricName || 0;
  }
  
  private getThroughput(metrics: any[]): number {
    // Calculate based on upload histogram and sizes
    return 850; // Placeholder MB/s
  }
  
  private calculatePotentialSavings(recommendations: string[]): number {
    // Parse recommendations for savings amounts
    let totalSavings = 0;
    recommendations.forEach(rec => {
      const match = rec.match(/~\$(\d+(?:\.\d+)?)/);
      if (match) {
        totalSavings += parseFloat(match[1]);
      }
    });
    return totalSavings;
  }
  
  listen(port: number) {
    this.app.listen(port, () => {
      console.log(`Monitoring dashboard listening on port ${port}`);
    });
  }
}

// docker/Dockerfile
const dockerfile = `
FROM node:18-alpine AS builder

WORKDIR /app

# Install dependencies
COPY package*.json ./
RUN npm ci --only=production

# Copy source
COPY . .

# Build TypeScript
RUN npm run build

# Production image
FROM node:18-alpine

WORKDIR /app

# Install production dependencies
COPY package*.json ./
RUN npm ci --only=production && npm cache clean --force

# Copy built application
COPY --from=builder /app/dist ./dist

# Create non-root user
RUN addgroup -g 1001 -S nodejs && \
    adduser -S nodejs -u 1001

USER nodejs

# Health check
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
  CMD node -e "require('http').get('http://localhost:9090/health', (r) => r.statusCode === 200 ? process.exit(0) : process.exit(1))"

EXPOSE 9090

CMD ["node", "dist/index.js"]
`;

// monitoring/alerts.yml
const alertsConfig = `
groups:
  - name: b2_alerts
    interval: 30s
    rules:
      - alert: HighErrorRate
        expr: rate(b2_uploads_total{status="failure"}[5m]) > 0.1
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "High B2 upload error rate"
          description: "Error rate is {{ $value }} errors per second"
      
      - alert: CircuitBreakerOpen
        expr: b2_circuit_breaker_state == 2
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "B2 circuit breaker is open"
          description: "B2 API circuit breaker has opened due to failures"
      
      - alert: BudgetThresholdExceeded
        expr: b2_cost_dollars{type="total"} > 800
        for: 10m
        labels:
          severity: warning
        annotations:
          summary: "Monthly budget threshold exceeded"
          description: "Current cost is ${{ $value }}"
      
      - alert: LowThroughput
        expr: rate(b2_upload_duration_seconds_sum[5m]) / rate(b2_upload_duration_seconds_count[5m]) > 30
        for: 10m
        labels:
          severity: warning
        annotations:
          summary: "B2 upload throughput is low"
          description: "Average upload duration is {{ $value }} seconds"
`;
