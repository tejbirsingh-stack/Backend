// Enterprise Billing Service with Stripe

import Stripe from 'stripe';
import { EventEmitter } from 'events';
import { Queue } from 'bull';
import { Logger } from 'winston';
import { Meter } from '@opentelemetry/api-metrics';

interface SubscriptionOptions {
  organizationId: string;
  trialDays?: number;
  addons?: string[];
  metadata?: Record<string, string>;
}

interface UsageMetrics {
  storage: {
    bytes: number;
    cost: number;
  };
  bandwidth: {
    bytes: number;
    cost: number;
  };
  processing: {
    minutes: number;
    cost: number;
  };
  customMetrics?: Record<string, number>;
}

interface PricingTier {
  id: string;
  name: string;
  basePrice: number;
  included: {
    storage: number;
    bandwidth: number;
    processing: number;
    users: number;
  };
  overage: {
    storage: number;
    bandwidth: number;
    processing: number;
    users: number;
  };
}

export class BillingService extends EventEmitter {
  private stripe: Stripe;
  private webhookEndpoint: string;
  private usageQueue: Queue;
  private logger: Logger;
  private metrics: Meter;
  
  constructor(config: BillingConfig) {
    super();
    this.stripe = new Stripe(config.stripeSecretKey, {
      apiVersion: '2023-10-16',
      typescript: true,
      telemetry: { enabled: false }
    });
    
    this.webhookEndpoint = config.webhookEndpoint;
    this.usageQueue = new Queue('usage-reporting', config.redisUrl);
    this.logger = config.logger;
    this.metrics = config.metrics;
    
    this.setupWebhooks();
    this.setupUsageReporting();
  }
  
  // Subscription Management
  
  async createSubscription(
    customerId: string,
    planId: string,
    options: SubscriptionOptions
  ): Promise<Stripe.Subscription> {
    try {
      // Get or create customer
      const customer = await this.getOrCreateCustomer(customerId, options.organizationId);
      
      // Build subscription items
      const items: Stripe.SubscriptionCreateParams.Item[] = [
        { price: planId } // Base plan
      ];
      
      // Add metered items
      const meteredPrices = await this.getMeteredPrices();
      items.push(
        { price: meteredPrices.storage },
        { price: meteredPrices.bandwidth },
        { price: meteredPrices.processing }
      );
      
      // Add optional addons
      if (options.addons) {
        for (const addon of options.addons) {
          items.push({ price: addon });
        }
      }
      
      // Create subscription
      const subscription = await this.stripe.subscriptions.create({
        customer: customer.id,
        items,
        payment_behavior: 'default_incomplete',
        payment_settings: { 
          save_default_payment_method: 'on_subscription',
          payment_method_types: ['card', 'sepa_debit', 'ach_debit']
        },
        expand: ['latest_invoice.payment_intent', 'pending_setup_intent'],
        trial_period_days: options.trialDays,
        metadata: {
          organizationId: options.organizationId,
          environment: process.env.NODE_ENV,
          ...options.metadata
        },
        collection_method: 'charge_automatically',
        proration_behavior: 'create_prorations'
      });
      
      // Set up usage recording
      await this.scheduleUsageReporting(subscription.id, options.organizationId);
      
      // Emit event
      this.emit('subscription:created', {
        subscriptionId: subscription.id,
        customerId: customer.id,
        organizationId: options.organizationId
      });
      
      return subscription;
      
    } catch (error) {
      this.logger.error('Failed to create subscription', { error, customerId, planId });
      throw error;
    }
  }
  
  async updateSubscription(
    subscriptionId: string,
    updates: Partial<Stripe.SubscriptionUpdateParams>
  ): Promise<Stripe.Subscription> {
    const subscription = await this.stripe.subscriptions.update(subscriptionId, {
      ...updates,
      proration_behavior: 'create_prorations'
    });
    
    this.emit('subscription:updated', {
      subscriptionId: subscription.id,
      changes: updates
    });
    
    return subscription;
  }
  
  async cancelSubscription(
    subscriptionId: string,
    immediately: boolean = false
  ): Promise<Stripe.Subscription> {
    const subscription = await this.stripe.subscriptions.update(subscriptionId, {
      cancel_at_period_end: !immediately,
      cancellation_details: {
        comment: 'Customer requested cancellation'
      }
    });
    
    if (immediately) {
      await this.stripe.subscriptions.cancel(subscriptionId);
    }
    
    this.emit('subscription:cancelled', {
      subscriptionId: subscription.id,
      immediate: immediately
    });
    
    return subscription;
  }
  
  // Usage Tracking and Reporting
  
  async recordUsage(
    subscriptionItemId: string,
    quantity: number,
    timestamp: number = Date.now(),
    metadata?: Record<string, string>
  ): Promise<void> {
    try {
      // Queue usage for batch processing
      await this.usageQueue.add('record-usage', {
        subscriptionItemId,
        quantity,
        timestamp,
        metadata
      }, {
        attempts: 3,
        backoff: {
          type: 'exponential',
          delay: 5000
        }
      });
      
      // Record metric
      this.metrics.createCounter('billing_usage_recorded').add(1, {
        subscription_item_id: subscriptionItemId
      });
      
    } catch (error) {
      this.logger.error('Failed to queue usage record', { 
        error, 
        subscriptionItemId, 
        quantity 
      });
      throw error;
    }
  }
  
  private async processUsageBatch(jobs: Job[]): Promise<void> {
    // Group by subscription item
    const grouped = jobs.reduce((acc, job) => {
      const { subscriptionItemId } = job.data;
      if (!acc[subscriptionItemId]) {
        acc[subscriptionItemId] = [];
      }
      acc[subscriptionItemId].push(job.data);
      return acc;
    }, {} as Record<string, any[]>);
    
    // Process each subscription item
    for (const [itemId, usages] of Object.entries(grouped)) {
      try {
        // Aggregate usage
        const totalQuantity = usages.reduce((sum, u) => sum + u.quantity, 0);
        const latestTimestamp = Math.max(...usages.map(u => u.timestamp));
        
        // Report to Stripe
        await this.stripe.subscriptionItems.createUsageRecord(itemId, {
          quantity: Math.ceil(totalQuantity),
          timestamp: Math.floor(latestTimestamp / 1000),
          action: 'set' // Use 'set' for idempotency
        });
        
        this.logger.info('Reported usage to Stripe', {
          itemId,
          quantity: totalQuantity,
          recordCount: usages.length
        });
        
      } catch (error) {
        this.logger.error('Failed to report usage batch', { error, itemId });
        throw error;
      }
    }
  }
  
  private async scheduleUsageReporting(
    subscriptionId: string,
    organizationId: string
  ): Promise<void> {
    // Schedule hourly usage collection
    await this.usageQueue.add('collect-usage', {
      subscriptionId,
      organizationId
    }, {
      repeat: {
        cron: '0 * * * *' // Every hour
      }
    });
  }
  
  async calculateUsageMetrics(orgId: string): Promise<UsageMetrics> {
    const [storage, bandwidth, processing] = await Promise.all([
      this.getStorageUsage(orgId),
      this.getBandwidthUsage(orgId),
      this.getProcessingMinutes(orgId)
    ]);
    
    // Calculate costs based on current pricing
    const pricing = await this.getCurrentPricing(orgId);
    
    return {
      storage: {
        bytes: storage,
        cost: this.calculateCost(storage, pricing.storage)
      },
      bandwidth: {
        bytes: bandwidth,
        cost: this.calculateCost(bandwidth, pricing.bandwidth)
      },
      processing: {
        minutes: processing,
        cost: this.calculateCost(processing, pricing.processing)
      }
    };
  }
  
  // Payment Methods
  
  async addPaymentMethod(
    customerId: string,
    paymentMethodId: string
  ): Promise<Stripe.PaymentMethod> {
    const paymentMethod = await this.stripe.paymentMethods.attach(
      paymentMethodId,
      { customer: customerId }
    );
    
    // Set as default if it's the first one
    const customer = await this.stripe.customers.retrieve(customerId) as Stripe.Customer;
    if (!customer.invoice_settings.default_payment_method) {
      await this.stripe.customers.update(customerId, {
        invoice_settings: {
          default_payment_method: paymentMethodId
        }
      });
    }
    
    return paymentMethod;
  }
  
  async removePaymentMethod(paymentMethodId: string): Promise<void> {
    await this.stripe.paymentMethods.detach(paymentMethodId);
  }
  
  // Invoicing
  
  async createInvoice(
    customerId: string,
    items: Stripe.InvoiceItemCreateParams[]
  ): Promise<Stripe.Invoice> {
    // Create invoice items
    for (const item of items) {
      await this.stripe.invoiceItems.create({
        customer: customerId,
        ...item
      });
    }
    
    // Create and finalize invoice
    const invoice = await this.stripe.invoices.create({
      customer: customerId,
      auto_advance: true,
      collection_method: 'charge_automatically'
    });
    
    await this.stripe.invoices.finalizeInvoice(invoice.id);
    
    return invoice;
  }
  
  // Webhook Handling
  
  private setupWebhooks(): void {
    // Verify webhook endpoint exists
    this.ensureWebhookEndpoint();
  }
  
  async handleWebhook(
    payload: string | Buffer,
    signature: string
  ): Promise<void> {
    let event: Stripe.Event;
    
    try {
      event = this.stripe.webhooks.constructEvent(
        payload,
        signature,
        process.env.STRIPE_WEBHOOK_SECRET!
      );
    } catch (error) {
      this.logger.error('Webhook signature verification failed', { error });
      throw new Error('Invalid webhook signature');
    }
    
    // Handle the event
    switch (event.type) {
      case 'customer.subscription.created':
      case 'customer.subscription.updated':
      case 'customer.subscription.deleted':
        await this.handleSubscriptionEvent(event);
        break;
        
      case 'invoice.paid':
      case 'invoice.payment_failed':
        await this.handleInvoiceEvent(event);
        break;
        
      case 'payment_method.attached':
      case 'payment_method.detached':
        await this.handlePaymentMethodEvent(event);
        break;
        
      default:
        this.logger.info('Unhandled webhook event type', { type: event.type });
    }
  }
  
  private async handleSubscriptionEvent(event: Stripe.Event): Promise<void> {
    const subscription = event.data.object as Stripe.Subscription;
    
    this.emit(`subscription:${event.type.split('.').pop()}`, {
      subscriptionId: subscription.id,
      customerId: subscription.customer,
      status: subscription.status
    });
  }
  
  // Reporting and Analytics
  
  async generateUsageReport(
    organizationId: string,
    startDate: Date,
    endDate: Date
  ): Promise<UsageReport> {
    const subscription = await this.getSubscriptionByOrgId(organizationId);
    
    // Fetch usage records from Stripe
    const usageRecords = await this.stripe.subscriptionItems.listUsageRecordSummaries(
      subscription.items.data[0].id,
      {
        from: Math.floor(startDate.getTime() / 1000),
        to: Math.floor(endDate.getTime() / 1000)
      }
    );
    
    // Fetch invoices
    const invoices = await this.stripe.invoices.list({
      customer: subscription.customer as string,
      created: {
        gte: Math.floor(startDate.getTime() / 1000),
        lte: Math.floor(endDate.getTime() / 1000)
      }
    });
    
    return {
      period: { start: startDate, end: endDate },
      usage: this.aggregateUsageRecords(usageRecords.data),
      invoices: invoices.data.map(i => ({
        id: i.id,
        amount: i.amount_paid,
        date: new Date(i.created * 1000),
        status: i.status
      })),
      summary: await this.calculateUsageSummary(organizationId, startDate, endDate)
    };
  }
}

// Enterprise SSO Service

import passport from 'passport';
import { Strategy as SamlStrategy } from '@node-saml/passport-saml';
import { Issuer, Strategy as OpenIDStrategy, generators } from 'openid-client';
import { OAuth2Strategy } from 'passport-oauth2';
import jwt from 'jsonwebtoken';
import { Redis } from 'ioredis';

interface SAMLConfig {
  entryPoint: string;
  certificate: string;
  issuer?: string;
  callbackUrl?: string;
  attributeMapping?: Record<string, string>;
}

interface OIDCConfig {
  discoveryUrl: string;
  clientId: string;
  clientSecret: string;
  scope?: string[];
}

interface SSOProvider {
  id: string;
  type: 'saml' | 'oidc' | 'oauth2';
  name: string;
  enabled: boolean;
  config: SAMLConfig | OIDCConfig;
  domains: string[];
  defaultRole?: string;
  attributeMapping: Record<string, string>;
}

export class SSOService {
  private providers: Map<string, SSOProvider> = new Map();
  private sessionStore: Redis;
  private logger: Logger;
  
  constructor(config: SSOConfig) {
    this.sessionStore = new Redis(config.redisUrl);
    this.logger = config.logger;
    
    this.setupPassport();
  }
  
  // SAML Configuration
  
  async configureSAML(providerId: string, config: SAMLConfig): Promise<void> {
    const strategy = new SamlStrategy({
      callbackUrl: config.callbackUrl || `${process.env.API_URL}/auth/saml/${providerId}/callback`,
      entryPoint: config.entryPoint,
      issuer: config.issuer || 'noah-mam',
      cert: config.certificate,
      identifierFormat: 'urn:oasis:names:tc:SAML:2.0:nameid-format:persistent',
      wantAssertionsSigned: true,
      signatureAlgorithm: 'sha256',
      acceptedClockSkewMs: 5000,
      attributeConsumingServiceIndex: false,
      disableRequestedAuthnContext: true,
      forceAuthn: false,
      skipRequestCompression: false,
      authnRequestBinding: 'HTTP-POST'
    }, async (profile: any, done: any) => {
      try {
        const user = await this.processSAMLProfile(providerId, profile);
        done(null, user);
      } catch (error) {
        done(error);
      }
    });
    
    passport.use(`saml-${providerId}`, strategy);
    
    // Store provider config
    this.providers.set(providerId, {
      id: providerId,
      type: 'saml',
      name: config.issuer || providerId,
      enabled: true,
      config,
      domains: [],
      attributeMapping: config.attributeMapping || this.getDefaultAttributeMapping()
    });
  }
  
  private async processSAMLProfile(providerId: string, profile: any): Promise<User> {
    const provider = this.providers.get(providerId);
    if (!provider) throw new Error('Provider not found');
    
    // Map attributes
    const attributes = this.mapAttributes(profile, provider.attributeMapping);
    
    // Find or create user
    const user = await this.findOrCreateUser({
      email: attributes.email,
      name: attributes.name,
      firstName: attributes.firstName,
      lastName: attributes.lastName,
      ssoProvider: providerId,
      ssoId: profile.nameID,
      attributes: profile
    });
    
    // Update user groups/roles if provided
    if (attributes.groups) {
      await this.syncUserGroups(user.id, attributes.groups);
    }
    
    // Create SSO session
    await this.createSSOSession(user.id, providerId, profile.sessionIndex);
    
    return user;
  }
  
  // OIDC Configuration
  
  async configureOIDC(providerId: string, config: OIDCConfig): Promise<void> {
    try {
      const issuer = await Issuer.discover(config.discoveryUrl);
      
      const client = new issuer.Client({
        client_id: config.clientId,
        client_secret: config.clientSecret,
        redirect_uris: [`${process.env.API_URL}/auth/oidc/${providerId}/callback`],
        response_types: ['code'],
        grant_types: ['authorization_code', 'refresh_token'],
        token_endpoint_auth_method: 'client_secret_post'
      });
      
      const strategy = new OpenIDStrategy({
        client,
        params: {
          scope: config.scope?.join(' ') || 'openid email profile'
        },
        passReqToCallback: true
      }, async (req: any, tokenSet: any, userinfo: any, done: any) => {
        try {
          const user = await this.processOIDCProfile(providerId, userinfo, tokenSet);
          done(null, user);
        } catch (error) {
          done(error);
        }
      });
      
      passport.use(`oidc-${providerId}`, strategy);
      
      // Store provider config
      this.providers.set(providerId, {
        id: providerId,
        type: 'oidc',
        name: issuer.metadata.issuer,
        enabled: true,
        config,
        domains: [],
        attributeMapping: this.getDefaultAttributeMapping()
      });
      
    } catch (error) {
      this.logger.error('Failed to configure OIDC provider', { error, providerId });
      throw error;
    }
  }
  
  private async processOIDCProfile(
    providerId: string,
    userinfo: any,
    tokenSet: any
  ): Promise<User> {
    const provider = this.providers.get(providerId);
    if (!provider) throw new Error('Provider not found');
    
    // Store tokens for later use
    await this.storeTokens(userinfo.sub, tokenSet);
    
    // Find or create user
    const user = await this.findOrCreateUser({
      email: userinfo.email,
      name: userinfo.name || `${userinfo.given_name} ${userinfo.family_name}`,
      firstName: userinfo.given_name,
      lastName: userinfo.family_name,
      ssoProvider: providerId,
      ssoId: userinfo.sub,
      attributes: userinfo
    });
    
    return user;
  }
  
  // Generic OAuth2
  
  async configureOAuth2(providerId: string, config: OAuth2Config): Promise<void> {
    const strategy = new OAuth2Strategy({
      authorizationURL: config.authorizationURL,
      tokenURL: config.tokenURL,
      clientID: config.clientId,
      clientSecret: config.clientSecret,
      callbackURL: `${process.env.API_URL}/auth/oauth2/${providerId}/callback`,
      scope: config.scope,
      passReqToCallback: true
    }, async (req: any, accessToken: string, refreshToken: string, profile: any, done: any) => {
      try {
        // Fetch user info if needed
        const userInfo = await this.fetchUserInfo(config.userInfoURL, accessToken);
        const user = await this.processOAuth2Profile(providerId, userInfo);
        done(null, user);
      } catch (error) {
        done(error);
      }
    });
    
    passport.use(`oauth2-${providerId}`, strategy);
  }
  
  // Session Management
  
  private async createSSOSession(
    userId: string,
    providerId: string,
    sessionIndex?: string
  ): Promise<string> {
    const sessionId = generators.random();
    
    const session = {
      id: sessionId,
      userId,
      providerId,
      sessionIndex,
      createdAt: new Date(),
      lastAccessed: new Date()
    };
    
    // Store in Redis with TTL
    await this.sessionStore.setex(
      `sso:session:${sessionId}`,
      86400, // 24 hours
      JSON.stringify(session)
    );
    
    // Map SAML session index to our session
    if (sessionIndex) {
      await this.sessionStore.setex(
        `sso:saml:${sessionIndex}`,
        86400,
        sessionId
      );
    }
    
    return sessionId;
  }
  
  async validateSSOSession(sessionId: string): Promise<SSOSession | null> {
    const data = await this.sessionStore.get(`sso:session:${sessionId}`);
    if (!data) return null;
    
    const session = JSON.parse(data);
    
    // Update last accessed
    session.lastAccessed = new Date();
    await this.sessionStore.setex(
      `sso:session:${sessionId}`,
      86400,
      JSON.stringify(session)
    );
    
    return session;
  }
  
  // Single Logout (SLO)
  
  async initiateSingleLogout(sessionId: string): Promise<string> {
    const session = await this.validateSSOSession(sessionId);
    if (!session) throw new Error('Session not found');
    
    const provider = this.providers.get(session.providerId);
    if (!provider) throw new Error('Provider not found');
    
    if (provider.type === 'saml' && session.sessionIndex) {
      // Generate SAML logout request
      const strategy = passport._strategies[`saml-${provider.id}`] as SamlStrategy;
      
      return new Promise((resolve, reject) => {
        strategy.logout({ 
          user: { 
            nameID: session.userId,
            sessionIndex: session.sessionIndex
          }
        }, (err, url) => {
          if (err) reject(err);
          else resolve(url);
        });
      });
    } else if (provider.type === 'oidc') {
      // OIDC logout
      const tokens = await this.getStoredTokens(session.userId);
      const client = passport._strategies[`oidc-${provider.id}`].client;
      
      return client.endSessionUrl({
        id_token_hint: tokens.id_token,
        post_logout_redirect_uri: `${process.env.APP_URL}/logout/complete`
      });
    }
    
    // Fallback: just clear session
    await this.clearSSOSession(sessionId);
    return `${process.env.APP_URL}/logout/complete`;
  }
  
  // Provider Management
  
  async listProviders(): Promise<SSOProvider[]> {
    return Array.from(this.providers.values())
      .filter(p => p.enabled)
      .map(p => ({
        ...p,
        config: undefined // Don't expose sensitive config
      }));
  }
  
  async getProviderByDomain(domain: string): Promise<SSOProvider | null> {
    for (const provider of this.providers.values()) {
      if (provider.domains.includes(domain)) {
        return provider;
      }
    }
    return null;
  }
  
  // Utilities
  
  private mapAttributes(
    profile: any,
    mapping: Record<string, string>
  ): Record<string, any> {
    const result: Record<string, any> = {};
    
    for (const [key, path] of Object.entries(mapping)) {
      result[key] = this.getNestedValue(profile, path);
    }
    
    return result;
  }
  
  private getNestedValue(obj: any, path: string): any {
    return path.split('.').reduce((curr, prop) => curr?.[prop], obj);
  }
  
  private getDefaultAttributeMapping(): Record<string, string> {
    return {
      email: 'email',
      name: 'displayName',
      firstName: 'givenName',
      lastName: 'surname',
      groups: 'memberOf'
    };
  }
  
  generateSSOToken(user: User, provider: SSOProvider): string {
    return jwt.sign({
      sub: user.id,
      email: user.email,
      name: user.name,
      provider: provider.id,
      iat: Math.floor(Date.now() / 1000),
      exp: Math.floor(Date.now() / 1000) + 3600 // 1 hour
    }, process.env.JWT_SECRET!, {
      algorithm: 'RS256'
    });
  }
}

// Webhook Service with Reliability

import { Queue, Job, Worker } from 'bullmq';
import crypto from 'crypto';
import { RateLimiter } from 'limiter';
import CircuitBreaker from 'opossum';

interface WebhookEvent {
  id: string;
  type: string;
  data: any;
  timestamp: Date;
  version: string;
}

interface WebhookEndpoint {
  id: string;
  url: string;
  secret: string;
  events: string[];
  active: boolean;
  metadata?: Record<string, any>;
  headers?: Record<string, string>;
  rateLimit?: {
    requests: number;
    window: number;
  };
}

interface WebhookDelivery {
  id: string;
  endpointId: string;
  eventId: string;
  attempts: number;
  lastAttemptAt?: Date;
  nextRetryAt?: Date;
  status: 'pending' | 'delivered' | 'failed';
  response?: {
    status: number;
    headers: Record<string, string>;
    body: string;
  };
  error?: string;
}

export class WebhookService {
  private queue: Queue;
  private worker: Worker;
  private endpoints: Map<string, WebhookEndpoint> = new Map();
  private rateLimiters: Map<string, RateLimiter> = new Map();
  private circuitBreakers: Map<string, CircuitBreaker> = new Map();
  private logger: Logger;
  private metrics: Meter;
  
  constructor(config: WebhookConfig) {
    this.queue = new Queue('webhook-delivery', {
      connection: config.redis,
      defaultJobOptions: {
        attempts: 5,
        backoff: {
          type: 'exponential',
          delay: 5000
        },
        removeOnComplete: {
          count: 100,
          age: 24 * 3600 // 24 hours
        },
        removeOnFail: false
      }
    });
    
    this.logger = config.logger;
    this.metrics = config.metrics;
    
    this.setupWorker();
    this.setupMonitoring();
  }
  
  // Webhook Registration
  
  async registerWebhook(endpoint: WebhookEndpoint): Promise<void> {
    // Validate endpoint
    await this.validateEndpoint(endpoint);
    
    // Store endpoint
    this.endpoints.set(endpoint.id, endpoint);
    await this.persistEndpoint(endpoint);
    
    // Set up rate limiter if configured
    if (endpoint.rateLimit) {
      this.rateLimiters.set(endpoint.id, new RateLimiter({
        tokensPerInterval: endpoint.rateLimit.requests,
        interval: endpoint.rateLimit.window * 1000
      }));
    }
    
    // Set up circuit breaker
    this.circuitBreakers.set(endpoint.id, new CircuitBreaker(
      this.deliverWebhook.bind(this),
      {
        timeout: 30000, // 30 seconds
        errorThresholdPercentage: 50,
        resetTimeout: 60000 // 1 minute
      }
    ));
    
    this.logger.info('Webhook endpoint registered', { endpointId: endpoint.id });
  }
  
  async unregisterWebhook(endpointId: string): Promise<void> {
    this.endpoints.delete(endpointId);
    this.rateLimiters.delete(endpointId);
    this.circuitBreakers.delete(endpointId);
    
    await this.removePersistedEndpoint(endpointId);
  }
  
  // Event Emission
  
  async emit(event: WebhookEvent): Promise<void> {
    // Find matching endpoints
    const endpoints = Array.from(this.endpoints.values())
      .filter(ep => ep.active && ep.events.includes(event.type));
    
    // Queue delivery to each endpoint
    for (const endpoint of endpoints) {
      await this.queueDelivery(endpoint, event);
    }
    
    // Record metric
    this.metrics.createCounter('webhooks_emitted').add(1, {
      event_type: event.type,
      endpoint_count: endpoints.length
    });
  }
  
  private async queueDelivery(
    endpoint: WebhookEndpoint,
    event: WebhookEvent
  ): Promise<void> {
    const delivery: WebhookDelivery = {
      id: crypto.randomUUID(),
      endpointId: endpoint.id,
      eventId: event.id,
      attempts: 0,
      status: 'pending'
    };
    
    // Check rate limit
    const rateLimiter = this.rateLimiters.get(endpoint.id);
    if (rateLimiter) {
      await rateLimiter.removeTokens(1);
    }
    
    // Generate signature
    const signature = this.generateSignature(endpoint.secret, event);
    
    // Queue job
    await this.queue.add('deliver', {
      delivery,
      endpoint,
      event,
      signature
    }, {
      delay: this.calculateDelay(endpoint),
      priority: this.calculatePriority(event.type)
    });
  }
  
  // Delivery Processing
  
  private setupWorker(): void {
    this.worker = new Worker('webhook-delivery', async (job: Job) => {
      const { delivery, endpoint, event, signature } = job.data;
      
      try {
        // Use circuit breaker
        const breaker = this.circuitBreakers.get(endpoint.id);
        if (!breaker) throw new Error('Circuit breaker not found');
        
        const response = await breaker.fire(endpoint, event, signature);
        
        // Update delivery record
        delivery.status = 'delivered';
        delivery.response = response;
        await this.updateDelivery(delivery);
        
        // Record success metric
        this.metrics.createCounter('webhooks_delivered').add(1, {
          endpoint_id: endpoint.id,
          event_type: event.type
        });
        
      } catch (error) {
        delivery.attempts++;
        delivery.lastAttemptAt = new Date();
        delivery.error = error.message;
        
        if (delivery.attempts >= 5) {
          delivery.status = 'failed';
          await this.handleFailedDelivery(delivery, endpoint, event);
        } else {
          delivery.nextRetryAt = this.calculateNextRetry(delivery.attempts);
        }
        
        await this.updateDelivery(delivery);
        
        // Record failure metric
        this.metrics.createCounter('webhooks_failed').add(1, {
          endpoint_id: endpoint.id,
          event_type: event.type,
          error: error.message
        });
        
        throw error; // Let Bull handle retry
      }
    }, {
      connection: this.queue.connection,
      concurrency: 10
    });
    
    this.worker.on('failed', (job, err) => {
      this.logger.error('Webhook delivery failed', {
        jobId: job.id,
        error: err.message,
        data: job.data
      });
    });
  }
  
  private async deliverWebhook(
    endpoint: WebhookEndpoint,
    event: WebhookEvent,
    signature: string
  ): Promise<any> {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 30000);
    
    try {
      const response = await fetch(endpoint.url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Noah-Event': event.type,
          'X-Noah-Signature': signature,
          'X-Noah-Timestamp': event.timestamp.toISOString(),
          'X-Noah-Event-Id': event.id,
          'User-Agent': 'Noah-Webhook/1.0',
          ...endpoint.headers
        },
        body: JSON.stringify(event),
        signal: controller.signal
      });
      
      clearTimeout(timeout);
      
      if (!response.ok && response.status >= 500) {
        throw new Error(`Server error: ${response.status}`);
      }
      
      return {
        status: response.status,
        headers: Object.fromEntries(response.headers.entries()),
        body: await response.text()
      };
      
    } catch (error) {
      clearTimeout(timeout);
      
      if (error.name === 'AbortError') {
        throw new Error('Request timeout');
      }
      
      throw error;
    }
  }
  
  // Signature Generation and Verification
  
  private generateSignature(secret: string, event: WebhookEvent): string {
    const timestamp = Math.floor(event.timestamp.getTime() / 1000);
    const payload = `${timestamp}.${JSON.stringify(event)}`;
    
    return crypto
      .createHmac('sha256', secret)
      .update(payload)
      .digest('hex');
  }
  
  verifySignature(
    payload: string | Buffer,
    signature: string,
    secret: string,
    timestamp: string
  ): boolean {
    const expectedSignature = crypto
      .createHmac('sha256', secret)
      .update(`${timestamp}.${payload}`)
      .digest('hex');
    
    return crypto.timingSafeEqual(
      Buffer.from(signature),
      Buffer.from(expectedSignature)
    );
  }
  
  // Retry Logic
  
  private calculateNextRetry(attempts: number): Date {
    // Exponential backoff with jitter
    const baseDelay = 5000; // 5 seconds
    const maxDelay = 3600000; // 1 hour
    
    const delay = Math.min(
      baseDelay * Math.pow(2, attempts - 1) + Math.random() * 1000,
      maxDelay
    );
    
    return new Date(Date.now() + delay);
  }
  
  private calculateDelay(endpoint: WebhookEndpoint): number {
    // Add small random delay to prevent thundering herd
    return Math.floor(Math.random() * 1000);
  }
  
  private calculatePriority(eventType: string): number {
    // Higher priority for critical events
    const priorities: Record<string, number> = {
      'payment.failed': 1,
      'subscription.cancelled': 2,
      'user.deleted': 2,
      'default': 5
    };
    
    return priorities[eventType] || priorities.default;
  }
  
  // Failed Delivery Handling
  
  private async handleFailedDelivery(
    delivery: WebhookDelivery,
    endpoint: WebhookEndpoint,
    event: WebhookEvent
  ): Promise<void> {
    // Notify webhook owner
    await this.notifyWebhookFailure(endpoint, delivery, event);
    
    // Disable endpoint after repeated failures
    const recentFailures = await this.getRecentFailures(endpoint.id);
    if (recentFailures > 10) {
      await this.disableEndpoint(endpoint.id);
    }
    
    // Store for later retry via UI
    await this.storeFailedDelivery(delivery, event);
  }
  
  // Monitoring and Analytics
  
  async getWebhookStats(endpointId?: string): Promise<WebhookStats> {
    const filter = endpointId ? { endpointId } : {};
    
    const [total, delivered, failed, pending] = await Promise.all([
      this.countDeliveries({ ...filter }),
      this.countDeliveries({ ...filter, status: 'delivered' }),
      this.countDeliveries({ ...filter, status: 'failed' }),
      this.countDeliveries({ ...filter, status: 'pending' })
    ]);
    
    const successRate = total > 0 ? (delivered / total) * 100 : 0;
    const avgDeliveryTime = await this.calculateAvgDeliveryTime(filter);
    
    return {
      total,
      delivered,
      failed,
      pending,
      successRate,
      avgDeliveryTime
    };
  }
  
  async getDeliveryHistory(
    endpointId: string,
    options: PaginationOptions
  ): Promise<PaginatedResult<WebhookDelivery>> {
    return this.queryDeliveries({
      endpointId,
      ...options
    });
  }
  
  // Manual Retry
  
  async retryDelivery(deliveryId: string): Promise<void> {
    const delivery = await this.getDelivery(deliveryId);
    if (!delivery) throw new Error('Delivery not found');
    
    const endpoint = this.endpoints.get(delivery.endpointId);
    if (!endpoint) throw new Error('Endpoint not found');
    
    const event = await this.getEvent(delivery.eventId);
    if (!event) throw new Error('Event not found');
    
    // Reset delivery status
    delivery.status = 'pending';
    delivery.attempts = 0;
    
    // Queue for immediate delivery
    await this.queueDelivery(endpoint, event);
  }
  
  // Testing Support
  
  async sendTestWebhook(endpointId: string): Promise<WebhookDelivery> {
    const endpoint = this.endpoints.get(endpointId);
    if (!endpoint) throw new Error('Endpoint not found');
    
    const testEvent: WebhookEvent = {
      id: `test_${crypto.randomUUID()}`,
      type: 'test.webhook',
      data: {
        message: 'This is a test webhook from Noah',
        timestamp: new Date()
      },
      timestamp: new Date(),
      version: '1.0'
    };
    
    // Deliver immediately
    const signature = this.generateSignature(endpoint.secret, testEvent);
    const response = await this.deliverWebhook(endpoint, testEvent, signature);
    
    return {
      id: crypto.randomUUID(),
      endpointId: endpoint.id,
      eventId: testEvent.id,
      attempts: 1,
      status: 'delivered',
      response
    };
  }
