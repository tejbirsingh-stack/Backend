typescriptimport { Authenticator } from '@otplib/authenticator';
import { SignJWT, jwtVerify, importPKCS8, importSPKI, KeyLike } from 'jose';
import * as argon2 from 'argon2';
import { createHash, randomBytes, createHmac } from 'crypto';

// Types and Interfaces
export interface LoginCredentials {
  email: string;
  password: string;
  mfaCode?: string;
  webauthnResponse?: AuthenticatorAssertionResponse;
}

export interface AuthContext {
  ip: string;
  userAgent: string;
  deviceFingerprint: string;
  sessionId?: string;
  requestId: string;
  timestamp: Date;
}

export interface AuthResult {
  accessToken: string;
  refreshToken: string;
  sessionToken: string;
  riskScore: number;
  requiresMfa: boolean;
  requiresWebAuthn: boolean;
  sessionId: string;
  expiresAt: Date;
}

export interface RiskFactors {
  ipRisk: number;
  deviceRisk: number;
  behaviorRisk: number;
  velocityRisk: number;
  geoRisk: number;
  timeRisk: number;
}

export class AuthService {
  private readonly riskEngine: RiskEngine;
  private readonly sessionStore: SessionStore;
  private readonly auditLogger: AuditLogger;
  private readonly webAuthnService: WebAuthnService;
  private readonly rateLimiter: RateLimiter;
  private readonly privateKey: KeyLike;
  private readonly publicKey: KeyLike;
  
  constructor(
    private readonly redis: Redis,
    private readonly logger: Logger,
    private readonly config: SecurityConfig
  ) {
    this.riskEngine = new RiskEngine(redis, logger);
    this.sessionStore = new SessionStore(redis);
    this.auditLogger = new AuditLogger(logger);
    this.webAuthnService = new WebAuthnService(redis);
    
    this.rateLimiter = new RateLimiter({
      storeClient: redis,
      keyPrefix: 'rl:auth',
      points: 5, // attempts
      duration: 900, // 15 minutes
      blockDuration: 900, // block for 15 minutes
    });
  }

  async authenticate(
    credentials: LoginCredentials,
    context: AuthContext
  ): Promise<AuthResult> {
    const startTime = Date.now();
    
    try {
      // Rate limiting check
      await this.checkRateLimit(credentials.email, context.ip);
      
      // Device fingerprint validation
      const deviceTrust = await this.validateDeviceFingerprint(
        credentials.email,
        context.deviceFingerprint
      );
      
      // Risk assessment with ML
      const riskScore = await this.riskEngine.assess({
        ip: context.ip,
        userAgent: context.userAgent,
        geoLocation: geoip.lookup(context.ip),
        loginHistory: await this.getLoginHistory(credentials.email),
        deviceTrust,
        timestamp: context.timestamp,
        behaviorProfile: await this.getBehaviorProfile(credentials.email)
      });
      
      // Log authentication attempt
      await this.auditLogger.log({
        event: 'auth.attempt',
        userId: null,
        email: credentials.email,
        ip: context.ip,
        riskScore,
        context
      });
      
      // High-risk authentication flow
      if (riskScore > 0.7) {
        await this.handleHighRiskAuth(credentials.email, context, riskScore);
      }
      
      // Verify credentials with timing-safe comparison
      const user = await this.getUserByEmail(credentials.email);
      if (!user) {
        // Prevent user enumeration - still perform password hash
        await argon2.hash('dummy-password');
        throw new InvalidCredentialsError();
      }
      
      const passwordValid = await this.verifyPassword(
        credentials.password,
        user.passwordHash
      );
      
      if (!passwordValid) {
        await this.handleFailedAuth(user.id, context);
        throw new InvalidCredentialsError();
      }
      
      // Check if MFA is required
      const mfaRequired = user.mfaEnabled || riskScore > 0.5;
      if (mfaRequired && !credentials.mfaCode && !credentials.webauthnResponse) {
        return {
          requiresMfa: true,
          requiresWebAuthn: user.webAuthnEnabled,
          sessionId: await this.createPendingSession(user.id, context),
          riskScore
        } as AuthResult;
      }
      
      // Verify MFA if provided
      if (credentials.mfaCode) {
        const mfaValid = await this.verifyMFA(user.id, credentials.mfaCode);
        if (!mfaValid) {
          await this.handleFailedAuth(user.id, context);
          throw new InvalidMFAError();
        }
      }
      
      // Verify WebAuthn if provided
      if (credentials.webauthnResponse) {
        const webauthnValid = await this.webAuthnService.verify(
          user.id,
          credentials.webauthnResponse,
          context
        );
        if (!webauthnValid) {
          await this.handleFailedAuth(user.id, context);
          throw new InvalidWebAuthnError();
        }
      }
      
      // Create session
      const session = await this.createSession(user, context, riskScore);
      
      // Generate tokens
      const tokens = await this.generateTokens(user, session);
      
      // Record successful authentication
      await this.recordSuccessfulAuth(user.id, context, riskScore);
      
      // Update behavior profile
      await this.updateBehaviorProfile(user.id, context);
      
      return {
        ...tokens,
        sessionId: session.id,
        riskScore,
        requiresMfa: false,
        requiresWebAuthn: false,
        expiresAt: new Date(Date.now() + 15 * 60 * 1000) // 15 minutes
      };
      
    } catch (error) {
      await this.auditLogger.log({
        event: 'auth.failed',
        email: credentials.email,
        error: error.message,
        ip: context.ip,
        duration: Date.now() - startTime
      });
      throw error;
    }
  }
  
  private async verifyPassword(
    password: string,
    hash: string
  ): Promise<boolean> {
    try {
      return await argon2.verify(hash, password, {
        type: argon2.argon2id,
        memoryCost: 65536, // 64 MB
        timeCost: 3,
        parallelism: 4,
      });
    } catch (error) {
      // Timing-safe false return
      await new Promise(resolve => setTimeout(resolve, 100));
      return false;
    }
  }
  
  private async generateTokens(
    user: User,
    session: Session
  ): Promise<{ accessToken: string; refreshToken: string; sessionToken: string }> {
    const permissions = await this.getEffectivePermissions(user.id);
    
    // Access token - short-lived
    const accessToken = await new SignJWT({
      sub: user.id,
      org: user.organizationId,
      email: user.email,
      permissions,
      sessionId: session.id,
      jti: randomBytes(16).toString('hex'),
    })
      .setProtectedHeader({ alg: 'RS256', typ: 'JWT' })
      .setIssuer(this.config.issuer)
      .setAudience(this.config.audience)
      .setExpirationTime('15m')
      .setNotBefore('0s')
      .setIssuedAt()
      .sign(this.privateKey);
    
    // Refresh token - longer-lived, bound to session
    const refreshToken = await new SignJWT({
      sub: user.id,
      sessionId: session.id,
      jti: randomBytes(16).toString('hex'),
      tokenFamily: session.tokenFamily,
    })
      .setProtectedHeader({ alg: 'RS256', typ: 'JWT' })
      .setIssuer(this.config.issuer)
      .setAudience(this.config.audience)
      .setExpirationTime('7d')
      .setNotBefore('0s')
      .setIssuedAt()
      .sign(this.privateKey);
    
    // Session token - for session management
    const sessionToken = this.generateSessionToken(session);
    
    // Store tokens for revocation checking
    await this.storeTokenMetadata(accessToken, refreshToken, session.id);
    
    return { accessToken, refreshToken, sessionToken };
  }
  
  async refreshAccessToken(
    refreshToken: string,
    context: AuthContext
  ): Promise<AuthResult> {
    try {
      // Verify refresh token
      const payload = await jwtVerify(refreshToken, this.publicKey, {
        issuer: this.config.issuer,
        audience: this.config.audience,
      });
      
      // Check if token is revoked
      const isRevoked = await this.isTokenRevoked(payload.jti as string);
      if (isRevoked) {
        throw new TokenRevokedError();
      }
      
      // Validate session
      const session = await this.sessionStore.get(payload.sessionId as string);
      if (!session || session.status !== 'active') {
        throw new InvalidSessionError();
      }
      
      // Check token family for refresh token rotation
      if (session.tokenFamily !== payload.tokenFamily) {
        // Possible token theft - revoke all tokens in family
        await this.revokeTokenFamily(session.tokenFamily);
        throw new TokenTheftDetectedError();
      }
      
      // Risk assessment for refresh
      const riskScore = await this.riskEngine.assessRefresh({
        userId: payload.sub,
        sessionId: session.id,
        ip: context.ip,
        userAgent: context.userAgent,
        lastActivity: session.lastActivity,
      });
      
      if (riskScore > 0.8) {
        await this.sessionStore.revoke(session.id);
        throw new HighRiskRefreshError();
      }
      
      // Get user and generate new tokens
      const user = await this.getUserById(payload.sub as string);
      
      // Rotate refresh token
      const newTokens = await this.generateTokens(user, session);
      
      // Revoke old refresh token
      await this.revokeToken(payload.jti as string);
      
      // Update session
      await this.sessionStore.update(session.id, {
        lastActivity: new Date(),
        tokenFamily: randomBytes(16).toString('hex'), // New family
      });
      
      return {
        ...newTokens,
        sessionId: session.id,
        riskScore,
        requiresMfa: false,
        requiresWebAuthn: false,
        expiresAt: new Date(Date.now() + 15 * 60 * 1000)
      };
      
    } catch (error) {
      await this.auditLogger.log({
        event: 'token.refresh.failed',
        error: error.message,
        ip: context.ip,
      });
      throw error;
    }
  }
  
  private async checkRateLimit(
    identifier: string,
    ip: string
  ): Promise<void> {
    try {
      await this.rateLimiter.consume(identifier);
      await this.rateLimiter.consume(ip);
    } catch (error) {
      throw new RateLimitExceededError();
    }
  }
  
  private async handleHighRiskAuth(
    email: string,
    context: AuthContext,
    riskScore: number
  ): Promise<void> {
    // Send alert
    await this.sendSecurityAlert(email, {
      type: 'high_risk_login',
      riskScore,
      ip: context.ip,
      location: geoip.lookup(context.ip),
      timestamp: new Date(),
    });
    
    // Require additional verification
    await this.triggerAdditionalVerification(email, context);
    
    // Log high-risk attempt
    await this.auditLogger.logSecurityEvent({
      severity: 'high',
      event: 'auth.high_risk',
      email,
      riskScore,
      context,
    });
    
    throw new AdditionalVerificationRequiredError();
  }
}
Risk Engine with ML
risk-engine.ts
typescriptimport * as tf from '@tensorflow/tfjs-node';
import { Redis } from 'ioredis';
import { Logger } from 'winston';
import * as geoip from 'geoip-lite';
import { distance } from 'fastest-levenshtein';

interface RiskAssessmentInput {
  ip: string;
  userAgent: string;
  geoLocation: geoip.Lookup | null;
  loginHistory: LoginAttempt[];
  deviceTrust: number;
  timestamp: Date;
  behaviorProfile: BehaviorProfile;
}

interface LoginAttempt {
  timestamp: Date;
  ip: string;
  success: boolean;
  riskScore: number;
  location?: geoip.Lookup;
}

interface BehaviorProfile {
  typicalLoginTimes: number[];
  typicalLocations: string[];
  typicalDevices: string[];
  averageSessionDuration: number;
  failedAttemptRate: number;
  lastPasswordChange: Date;
  accountAge: number;
}

export class RiskEngine {
  private model: tf.LayersModel;
  private readonly featureExtractor: FeatureExtractor;
  private readonly anomalyDetector: AnomalyDetector;
  
  constructor(
    private readonly redis: Redis,
    private readonly logger: Logger
  ) {
    this.featureExtractor = new FeatureExtractor();
    this.anomalyDetector = new AnomalyDetector();
    this.loadModel();
  }
  
  private async loadModel(): Promise<void> {
    try {
      // Load pre-trained fraud detection model
      this.model = await tf.loadLayersModel('file://./models/fraud-detection/model.json');
    } catch (error) {
      this.logger.error('Failed to load ML model', error);
      // Fallback to rule-based scoring
    }
  }
  
  async assess(input: RiskAssessmentInput): Promise<number> {
    try {
      // Extract features
      const features = await this.extractFeatures(input);
      
      // ML-based risk scoring if model is available
      if (this.model) {
        const mlScore = await this.mlRiskScore(features);
        const ruleScore = await this.ruleBasedScore(input);
        
        // Weighted combination
        return mlScore * 0.7 + ruleScore * 0.3;
      }
      
      // Fallback to rule-based scoring
      return this.ruleBasedScore(input);
      
    } catch (error) {
      this.logger.error('Risk assessment failed', error);
      // Conservative approach - return high risk on error
      return 0.8;
    }
  }
  
  private async extractFeatures(input: RiskAssessmentInput): Promise<number[]> {
    const features: number[] = [];
    
    // IP-based features
    features.push(...this.extractIpFeatures(input.ip, input.loginHistory));
    
    // Geographic features
    features.push(...this.extractGeoFeatures(input.geoLocation, input.behaviorProfile));
    
    // Temporal features
    features.push(...this.extractTemporalFeatures(input.timestamp, input.behaviorProfile));
    
    // Device features
    features.push(...this.extractDeviceFeatures(input.userAgent, input.deviceTrust));
    
    // Behavioral features
    features.push(...this.extractBehavioralFeatures(input.loginHistory, input.behaviorProfile));
    
    // Velocity features
    features.push(...this.extractVelocityFeatures(input.loginHistory));
    
    return features;
  }
  
  private extractIpFeatures(ip: string, history: LoginAttempt[]): number[] {
    const features: number[] = [];
    
    // IP reputation score (from threat intelligence)
    const ipReputation = await this.getIpReputation(ip);
    features.push(ipReputation);
    
    // IP type (residential, datacenter, VPN, Tor)
    const ipType = await this.getIpType(ip);
    features.push(
      ipType === 'residential' ? 0 : 
      ipType === 'datacenter' ? 0.3 :
      ipType === 'vpn' ? 0.6 :
      ipType === 'tor' ? 0.9 : 0.5
    );
    
    // Historical IP usage
    const ipUsageCount = history.filter(h => h.ip === ip).length;
    const totalAttempts = history.length;
    features.push(ipUsageCount / (totalAttempts || 1));
    
    // Failed attempts from this IP
    const failedFromIp = history.filter(h => h.ip === ip && !h.success).length;
    features.push(failedFromIp / (ipUsageCount || 1));
    
    return features;
  }
  
  private extractGeoFeatures(
    location: geoip.Lookup | null,
    profile: BehaviorProfile
  ): number[] {
    const features: number[] = [];
    
    if (!location) {
      features.push(1); // Unknown location is high risk
      return features;
    }
    
    // Distance from typical locations
    const minDistance = Math.min(
      ...profile.typicalLocations.map(loc => {
        const typical = geoip.lookup(loc);
        if (!typical) return Infinity;
        
        return this.calculateGeoDistance(
          location.ll[0], location.ll[1],
          typical.ll[0], typical.ll[1]
        );
      })
    );
    
    // Normalize distance (0-1)
    features.push(Math.min(minDistance / 10000, 1)); // 10,000 km max
    
    // Country risk score
    const countryRisk = this.getCountryRiskScore(location.country);
    features.push(countryRisk);
    
    // New country indicator
    const isNewCountry = !profile.typicalLocations.some(loc => {
      const typical = geoip.lookup(loc);
      return typical?.country === location.country;
    });
    features.push(isNewCountry ? 1 : 0);
    
    return features;
  }
  
  private extractTemporalFeatures(
    timestamp: Date,
    profile: BehaviorProfile
  ): number[] {
    const features: number[] = [];
    
    const hour = timestamp.getHours();
    const dayOfWeek = timestamp.getDay();
    
    // Time of day risk (normalized)
    const timeDeviation = Math.min(
      ...profile.typicalLoginTimes.map(typical => 
        Math.abs(hour - typical) / 12
      )
    );
    features.push(timeDeviation);
    
    // Weekend/weekday pattern
    const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
    const typicallyWeekend = profile.typicalLoginTimes.some(t => {
      const d = new Date();
      d.setHours(t);
      return d.getDay() === 0 || d.getDay() === 6;
    });
    features.push(isWeekend !== typicallyWeekend ? 0.3 : 0);
    
    // Night-time login (22:00 - 06:00)
    const isNightTime = hour >= 22 || hour <= 6;
    features.push(isNightTime ? 0.4 : 0);
    
    return features;
  }
  
  private extractDeviceFeatures(
    userAgent: string,
    deviceTrust: number
  ): number[] {
    const features: number[] = [];
    
    // Device trust score
    features.push(1 - deviceTrust);
    
    // Parse user agent
    const parser = new UAParser(userAgent);
    const device = parser.getResult();
    
    // Bot detection
    const botScore = this.detectBot(userAgent);
    features.push(botScore);
    
    // Suspicious user agent patterns
    const suspiciousUA = this.detectSuspiciousUserAgent(userAgent);
    features.push(suspiciousUA ? 0.7 : 0);
    
    // Device type risk
    const deviceTypeRisk = 
      device.device.type === 'mobile' ? 0.1 :
      device.device.type === 'tablet' ? 0.2 :
      !device.device.type ? 0.3 : 0; // Desktop
    features.push(deviceTypeRisk);
    
    return features;
  }
  
  private async mlRiskScore(features: number[]): Promise<number> {
    try {
      const input = tf.tensor2d([features]);
      const prediction = this.model.predict(input) as tf.Tensor;
      const score = await prediction.data();
      
      input.dispose();
      prediction.dispose();
      
      return score[0];
    } catch (error) {
      this.logger.error('ML prediction failed', error);
      return 0.5; // Neutral score on error
    }
  }
  
  private async ruleBasedScore(input: RiskAssessmentInput): Promise<number> {
    const factors: RiskFactors = {
      ipRisk: await this.calculateIpRisk(input.ip, input.loginHistory),
      deviceRisk: this.calculateDeviceRisk(input.userAgent, input.deviceTrust),
      behaviorRisk: this.calculateBehaviorRisk(input.behaviorProfile, input.timestamp),
      velocityRisk: this.calculateVelocityRisk(input.loginHistory),
      geoRisk: this.calculateGeoRisk(input.geoLocation, input.behaviorProfile),
      timeRisk: this.calculateTimeRisk(input.timestamp, input.behaviorProfile)
    };
    
    // Weighted average with emphasis on critical factors
    const weights = {
      ipRisk: 0.25,
      deviceRisk: 0.20,
      behaviorRisk: 0.20,
      velocityRisk: 0.15,
      geoRisk: 0.15,
      timeRisk: 0.05
    };
    
    let totalRisk = 0;
    for (const [factor, weight] of Object.entries(weights)) {
      totalRisk += factors[factor as keyof RiskFactors] * weight;
    }
    
    // Apply non-linear transformation for extreme cases
    if (totalRisk > 0.8) {
      totalRisk = 0.8 + (totalRisk - 0.8) * 0.5;
    }
    
    return Math.min(totalRisk, 1);
  }
  
  private calculateGeoDistance(
    lat1: number, lon1: number,
    lat2: number, lon2: number
  ): number {
    const R = 6371; // Earth's radius in kilometers
    const dLat = this.toRadians(lat2 - lat1);
    const dLon = this.toRadians(lon2 - lon1);
    
    const a = 
      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
      Math.cos(this.toRadians(lat1)) * Math.cos(this.toRadians(lat2)) *
      Math.sin(dLon / 2) * Math.sin(dLon / 2);
    
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    return R * c;
  }
  
  private toRadians(degrees: number): number {
    return degrees * (Math.PI / 180);
  }
  
  private getCountryRiskScore(country: string): number {
    // High-risk countries based on fraud statistics
    const highRiskCountries = ['XX', 'YY', 'ZZ']; // Placeholder
    const mediumRiskCountries = ['AA', 'BB', 'CC']; // Placeholder
    
    if (highRiskCountries.includes(country)) return 0.8;
    if (mediumRiskCountries.includes(country)) return 0.5;
    return 0.2;
  }
  
  private detectBot(userAgent: string): number {
    const botPatterns = [
      /bot/i, /crawler/i, /spider/i, /scraper/i,
      /curl/i, /wget/i, /python/i, /java/i
    ];
    
    for (const pattern of botPatterns) {
      if (pattern.test(userAgent)) return 0.9;
    }
    
    // Check for missing expected headers
    if (!userAgent || userAgent.length < 20) return 0.7;
    
    return 0;
  }
}

typescriptimport {
  generateRegistrationOptions,
  verifyRegistrationResponse,
  generateAuthenticationOptions,
  verifyAuthenticationResponse,
  RegistrationResponseJSON,
  AuthenticationResponseJSON,
} from '@simplewebauthn/server';
import { Redis } from 'ioredis';
import { randomBytes } from 'crypto';

interface WebAuthnCredential {
  credentialId: string;
  publicKey: string;
  counter: number;
  deviceName: string;
  createdAt: Date;
  lastUsedAt: Date;
  aaguid?: string;
  userVerified: boolean;
}

export class WebAuthnService {
  private readonly rpName = 'Noah Media Platform';
  private readonly rpId = process.env.WEBAUTHN_RP_ID || 'noah.io';
  private readonly origin = process.env.WEBAUTHN_ORIGIN || 'https://app.noah.io';
  
  constructor(private readonly redis: Redis) {}
  
  async generateRegistrationOptions(
    userId: string,
    userEmail: string,
    userName: string
  ): Promise<PublicKeyCredentialCreationOptions> {
    // Get existing credentials to exclude
    const existingCredentials = await this.getUserCredentials(userId);
    
    const options = generateRegistrationOptions({
      rpName: this.rpName,
      rpID: this.rpId,
      userID: userId,
      userName: userEmail,
      userDisplayName: userName,
      attestationType: 'direct', // Request attestation for high security
      authenticatorSelection: {
        authenticatorAttachment: 'cross-platform', // Security keys only
        userVerification: 'required',
        residentKey: 'required', // Discoverable credentials
      },
      excludeCredentials: existingCredentials.map(cred => ({
        id: Buffer.from(cred.credentialId, 'base64'),
        type: 'public-key',
      })),
      challenge: randomBytes(32),
      timeout: 300000, // 5 minutes
    });
    
    // Store challenge for verification
    await this.storeChallenge(userId, 'registration', options.challenge);
    
    return options;
  }
  
  async verifyRegistration(
    userId: string,
    response: RegistrationResponseJSON,
    deviceName: string
  ): Promise<boolean> {
    try {
      // Retrieve stored challenge
      const expectedChallenge = await this.getChallenge(userId, 'registration');
      if (!expectedChallenge) {
        throw new Error('Registration challenge not found or expired');
      }
      
      const verification = await verifyRegistrationResponse({
        response,
        expectedChallenge,
        expectedOrigin: this.origin,
        expectedRPID: this.rpId,
        requireUserVerification: true,
      });
      
      if (!verification.verified || !verification.registrationInfo) {
        return false;
      }
      
      const { credentialPublicKey, credentialID, counter, aaguid } = 
        verification.registrationInfo;
      
      // Check if authenticator is from approved vendor
      if (!this.isApprovedAuthenticator(aaguid)) {
        throw new Error('Authenticator not from approved vendor');
      }
      
      // Store credential
      const credential: WebAuthnCredential = {
        credentialId: Buffer.from(credentialID).toString('base64'),
        publicKey: Buffer.from(credentialPublicKey).toString('base64'),
        counter,
        deviceName,
        createdAt: new Date(),
        lastUsedAt: new Date(),
        aaguid: Buffer.from(aaguid).toString('hex'),
        userVerified: true,
      };
      
      await this.storeCredential(userId, credential);
      
      // Audit log
      await this.auditLog({
        event: 'webauthn.registered',
        userId,
        credentialId: credential.credentialId,
        deviceName,
        aaguid: credential.aaguid,
      });
      
      return true;
    } catch (error) {
      console.error('WebAuthn registration failed:', error);
      return false;
    }
  }
  
  async generateAuthenticationOptions(
    userId?: string
  ): Promise<PublicKeyCredentialRequestOptions> {
    const allowCredentials = userId 
      ? await this.getUserCredentials(userId)
      : [];
    
    const options = generateAuthenticationOptions({
      rpID: this.rpId,
      allowCredentials: allowCredentials.map(cred => ({
        id: Buffer.from(cred.credentialId, 'base64'),
        type: 'public-key',
      })),
      userVerification: 'required',
      challenge: randomBytes(32),
      timeout: 300000, // 5 minutes
    });
    
    // Store challenge

        userId,
        response.rawId
      );

        throw new Error('Credential not found');
      }

        userId,
        'authentication'
      );
      
      if (!expectedChallenge) {
        throw new Error('Authentication challenge not found or expired');
      }
      
      const verification = await verifyAuthenticationResponse({
        response,
        expectedChallenge,
        expectedOrigin: this.origin,
        expectedRPID: this.rpId,
        authenticator: {
          credentialPublicKey: Buffer.from(credential.publicKey, 'base64'),
          credentialID: Buffer.from(credential.credentialId, 'base64'),
          counter: credential.counter,
        },
        requireUserVerification: true,
      });
      
      if (!verification.verified) {
        return false;
      }
      
      // Update counter to prevent replay attacks
      if (verification.authenticationInfo.newCounter <= credential.counter) {
        // Possible cloned authenticator
        await this.handleClonedAuthenticator(userId, credential);
        return false;
      }
      
      // Update credential
      await this.updateCredential(userId, credential.credentialId, {
        counter: verification.authenticationInfo.newCounter,
        lastUsedAt: new Date(),
      });
      
      // Audit log
      await this.auditLog({
        event: 'webauthn.authenticated',
        userId,
        credentialId: credential.credentialId,
        deviceName: credential.deviceName,
        context,
      });
      
      return true;
    } catch (error) {
      console.error('WebAuthn verification failed:', error);
      return false;
    }

  evaluate(policy: PolicyDefinition, context: PolicyContext): PolicyEvaluation {

      const result = jsonLogic.apply(policy.conditions, context);
      
      return {
        policyId: policy.id,
        applicable: true,
        result: !!result,
        trace: this.generateTrace(policy.conditions, context)
      };
    } catch (error) {
      this.logger.error('Policy evaluation error', { 
        policyId: policy.id, 
        error: error.message 
      });
      
      return {
        policyId: policy.id,
        applicable: false,
        result: false,
        error: error.message
      };
    }
  }
  
  private generateTrace(conditions: any, context: any): any {
    // Generate evaluation trace for debugging
    return {
      conditions,
      context,
      evaluated: new Date()
    };
  }

typescriptimport { Redis } from 'ioredis';
import { randomBytes, createHash } from 'crypto';
import { Logger } from 'winston';

interface Session {
  id: string;
  userId: string;
  organizationId: string;
  deviceFingerprint: string;
  userAgent: string;
  ip: string;
  location?: {
    country: string;
    city: string;
    coordinates: [number, number];
  };
  tokenFamily: string;
  createdAt: Date;
  lastActivity: Date;
  expiresAt: Date;
  status: 'active' | 'expired' | 'revoked';
  mfaVerified: boolean;
  riskScore: number;
  metadata?: Record<string, any>;
}

export class SessionStore {
  private readonly ttl = 7 * 24 * 60 * 60; // 7 days
  private readonly maxConcurrentSessions = 5;
  
  constructor(
    private readonly redis: Redis,
    private readonly logger: Logger
  ) {}
  
  async create(
    userId: string,
    context: AuthContext,
    riskScore: number
  ): Promise<Session> {
    // Check concurrent session limit
    await this.enforceSessionLimit(userId);
    
    const session: Session = {
      id: this.generateSessionId(),
      userId,
      organizationId: await this.getUserOrganization(userId),
      deviceFingerprint: context.deviceFingerprint,
      userAgent: context.userAgent,
      ip: context.ip,
      location: await this.getLocation(context.ip),
      tokenFamily: randomBytes(16).toString('hex'),
      createdAt: new Date(),
      lastActivity: new Date(),
      expiresAt: new Date(Date.now() + this.ttl * 1000),
      status: 'active',
      mfaVerified: false,
      riskScore,
      metadata: {}
    };
    
    // Store session
    await this.store(session);
    
    // Add to user's session index
    await this.indexSession(userId, session.id);
    
    // Audit log
    await this.auditLog({
      event: 'session.created',
      sessionId: session.id,
      userId,
      ip: context.ip,
      riskScore
    });
    
    return session;
  }
  
  async get(sessionId: string): Promise<Session | null> {
    const key = `session:${sessionId}`;
    const data = await this.redis.get(key);
    
    if (!data) return null;
    
    const session = JSON.parse(data) as Session;
    
    // Check if expired
    if (new Date(session.expiresAt) < new Date()) {
      session.status = 'expired';
      await this.update(sessionId, { status: 'expired' });
    }
    
    return session;
  }
  
  async update(
    sessionId: string,
    updates: Partial<Session>
  ): Promise<void> {
    const session = await this.get(sessionId);
    if (!session) {
      throw new Error('Session not found');
    }
    
    const updated = { ...session, ...updates };
    await this.store(updated);
    
    // Update activity tracking
    if (updates.lastActivity) {
      await this.updateActivityTracking(session.userId, sessionId);
    }
  }
  
  async revoke(sessionId: string): Promise<void> {
    const session = await this.get(sessionId);
    if (!session) return;
    
    await this.update(sessionId, { status: 'revoked' });
    
    // Remove from user's active sessions
    await this.removeFromIndex(session.userId, sessionId);
    
    // Revoke all tokens in the family
    await this.revokeTokenFamily(session.tokenFamily);
    
    // Audit log
    await this.auditLog({
      event: 'session.revoked',
      sessionId,
      userId: session.userId
    });
  }
  
  async revokeAllUserSessions(userId: string): Promise<void> {
    const sessions = await this.getUserSessions(userId);
    
    for (const session of sessions) {
      await this.revoke(session.id);
    }
    
    // Audit log
    await this.auditLog({
      event: 'session.revoked_all',
      userId,
      count: sessions.length
    });
  }
  
  async validateSession(
    sessionId: string,
    context: AuthContext
  ): Promise<SessionValidation> {
    const session = await this.get(sessionId);
    
    if (!session || session.status !== 'active') {
      return { valid: false, reason: 'invalid_session' };
    }
    
    // Check expiration
    if (new Date(session.expiresAt) < new Date()) {
      await this.revoke(sessionId);
      return { valid: false, reason: 'expired' };
    }
    
    // Validate device fingerprint
    if (session.deviceFingerprint !== context.deviceFingerprint) {
      // Possible session hijacking
      await this.handleSuspiciousActivity(session, context);
      return { valid: false, reason: 'device_mismatch' };
    }
    
    // Check for significant changes
    const riskFactors = await this.assessSessionRisk(session, context);
    
    if (riskFactors.risk > 0.8) {
      await this.handleHighRiskSession(session, context, riskFactors);
      return { valid: false, reason: 'high_risk' };
    }
    
    // Update last activity
    await this.update(sessionId, { 
      lastActivity: new Date(),
      ip: context.ip 
    });
    
    return { valid: true, session, riskFactors };
  }
  
  private async enforceSessionLimit(userId: string): Promise<void> {
    const sessions = await this.getUserSessions(userId);
    
    if (sessions.length >= this.maxConcurrentSessions) {
      // Sort by last activity and revoke oldest
      sessions.sort((a, b) => 
        new Date(a.lastActivity).getTime() - new Date(b.lastActivity).getTime()
      );
      
      const toRevoke = sessions.slice(0, sessions.length - this.maxConcurrentSessions + 1);
      
      for (const session of toRevoke) {
        await this.revoke(session.id);
      }
    }
  }
  
  private async assessSessionRisk(
    session: Session,
    context: AuthContext
  ): Promise<SessionRiskFactors> {
    const factors: SessionRiskFactors = {
      ipChange: false,
      locationChange: false,
      deviceChange: false,
      unusualTime: false,
      risk: 0
    };
    
    // IP change detection
    if (session.ip !== context.ip) {
      factors.ipChange = true;
      factors.risk += 0.3;
      
      // Check if IP is in same network
      if (!this.isSameNetwork(session.ip, context.ip)) {
        factors.risk += 0.2;
      }
    }
    
    // Location change detection
    const newLocation = await this.getLocation(context.ip);
    if (session.location && newLocation) {
      const distance = this.calculateDistance(
        session.location.coordinates,
        newLocation.coordinates
      );
      
      if (distance > 100) { // 100km
        factors.locationChange = true;
        
        // Impossible travel detection
        const timeDiff = Date.now() - new Date(session.lastActivity).getTime();
        const possibleDistance = (timeDiff / 3600000) * 900; // 900km/h max
        
        if (distance > possibleDistance) {
          factors.risk += 0.5;
        } else {
          factors.risk += 0.2;
        }
      }
    }
    
    // User agent change
    if (session.userAgent !== context.userAgent) {
      const similarity = this.calculateUASimilarity(
        session.userAgent,
        context.userAgent
      );
      
      if (similarity < 0.8) {
        factors.deviceChange = true;
        factors.risk += 0.3;
      }
    }
    
    // Time-based anomaly
    const hour = new Date().getHours();
    const isUnusualTime = hour >= 2 && hour <= 5;
    if (isUnusualTime) {
      factors.unusualTime = true;
      factors.risk += 0.1;
    }
    
    return factors;
  }
  
  private generateSessionId(): string {
    return randomBytes(32).toString('base64url');
  }
  
  private async store(session: Session): Promise<void> {
    const key = `session:${session.id}`;
    const ttl = Math.floor(
      (new Date(session.expiresAt).getTime() - Date.now()) / 1000
    );
    
    await this.redis.setex(
      key,
      ttl,
      JSON.stringify(session)
    );
  }
  
  private async indexSession(userId: string, sessionId: string): Promise<void> {
    const key = `user:sessions:${userId}`;
    await this.redis.sadd(key, sessionId);
    await this.redis.expire(key, this.ttl);
  }
  
  private async getUserSessions(userId: string): Promise<Session[]> {
    const key = `user:sessions:${userId}`;
    const sessionIds = await this.redis.smembers(key);
    
    const sessions: Session[] = [];
    for (const id of sessionIds) {
      const session = await this.get(id);
      if (session && session.status === 'active') {
        sessions.push(session);
      } else {
        // Clean up invalid session from index
        await this.redis.srem(key, id);
      }
    }
    
    return sessions;
  }
  
  private calculateDistance(
    coords1: [number, number],
    coords2: [number, number]
  ): number {
    // Haversine formula for distance calculation
    const R = 6371; // Earth's radius in km
    const lat1 = coords1[0] * Math.PI / 180;
    const lat2 = coords2[0] * Math.PI / 180;
    const deltaLat = (coords2[0] - coords1[0]) * Math.PI / 180;
    const deltaLon = (coords2[1] - coords1[1]) * Math.PI / 180;
    
    const a = Math.sin(deltaLat / 2) * Math.sin(deltaLat / 2) +
              Math.cos(lat1) * Math.cos(lat2) *
              Math.sin(deltaLon / 2) * Math.sin(deltaLon / 2);
    
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    
    return R * c;
  }
  
  private isSameNetwork(ip1: string, ip2: string): boolean {
    // Simple check for same /24 network
    const parts1 = ip1.split('.');
    const parts2 = ip2.split('.');
    
    return parts1[0] === parts2[0] && 
           parts1[1] === parts2[1] && 
           parts1[2] === parts2[2];
  }
  
  private calculateUASimilarity(ua1: string, ua2: string): number {
    // Parse and compare user agents
    const parser = new UAParser();
    const parsed1 = parser.setUA(ua1).getResult();
    const parsed2 = parser.setUA(ua2).getResult();
    
    let similarity = 0;
    
    if (parsed1.browser.name === parsed2.browser.name) similarity += 0.3;
    if (parsed1.os.name === parsed2.os.name) similarity += 0.3;
    if (parsed1.device.type === parsed2.device.type) similarity += 0.2;
    if (parsed1.device.vendor === parsed2.device.vendor) similarity += 0.2;
    
    return similarity;
  }

typescriptimport { Logger } from 'winston';
import { createHash, sign, verify } from 'crypto';
import { Kafka, Producer } from 'kafkajs';
import { Client as ElasticsearchClient } from '@elastic/elasticsearch';

interface AuditEntry {
  id: string;
  timestamp: Date;
  event: string;
  severity: 'info' | 'warning' | 'error' | 'critical';
  userId?: string;
  organizationId?: string;
  sessionId?: string;
  ip?: string;
  userAgent?: string;
  resource?: {
    type: string;
    id: string;
    action: string;
  };
  result: 'success' | 'failure';
  metadata?: Record<string, any>;
  hash?: string;
  previousHash?: string;
  signature?: string;
}

export class AuditLogger {
  private readonly producer: Producer;
  private readonly elasticsearch: ElasticsearchClient;
  private previousHash: string = '0';
  private readonly privateKey: Buffer;
  
  constructor(
    private readonly logger: Logger,
    kafka: Kafka,
    elasticsearchUrl: string
  ) {
    this.producer = kafka.producer({
      idempotent: true,
      transactionalId: 'audit-logger',
      maxInFlightRequests: 1
    });
    
    this.elasticsearch = new ElasticsearchClient({
      node: elasticsearchUrl,
      auth: {
        apiKey: process.env.ELASTICSEARCH_API_KEY!
      }
    });
    
    this.privateKey = Buffer.from(process.env.AUDIT_SIGNING_KEY!, 'base64');
  }
  
  async log(entry: Partial<AuditEntry>): Promise<void> {
    const auditEntry: AuditEntry = {
      id: this.generateId(),
      timestamp: new Date(),
      event: entry.event!,
      severity: entry.severity || 'info',
      result: entry.result || 'success',
      previousHash: this.previousHash,
      ...entry
    };
    
    // Create cryptographic hash chain
    auditEntry.hash = this.generateHash(auditEntry);
    
    // Sign the entry
    auditEntry.signature = this.signEntry(auditEntry);
    
    // Update chain
    this.previousHash = auditEntry.hash!;
    
    try {
      // Write to multiple destinations for redundancy
      await Promise.all([
        this.writeToKafka(auditEntry),
        this.writeToElasticsearch(auditEntry),
        this.writeToS3(auditEntry), // Long-term archive
        this.writeToSIEM(auditEntry) // Real-time security monitoring
      ]);
      
    } catch (error) {
      // Critical failure - write to local emergency log
      this.logger.error('Audit log write failed', { error, entry: auditEntry });
      await this.writeToEmergencyLog(auditEntry);
      throw error;
    }
  }
  
  async logSecurityEvent(event: SecurityAuditEvent): Promise<void> {
    await this.log({
      ...event,
      severity: event.severity || 'warning',
      metadata: {
        ...event.metadata,
        securityRelevant: true,
        alertRequired: event.severity === 'critical'
      }
    });
    
    // Alert security team for critical events
    if (event.severity === 'critical') {
      await this.alertSecurityTeam(event);
    }
  }
  
  private generateHash(entry: AuditEntry): string {
    const content = JSON.stringify({
      id: entry.id,
      timestamp: entry.timestamp,
      event: entry.event,
      userId: entry.userId,
      data: entry.metadata,
      previousHash: entry.previousHash
    });
    
    return createHash('sha256').update(content).digest('hex');
  }
  
  private signEntry(entry: AuditEntry): string {
    const content = JSON.stringify({
      id: entry.id,
      hash: entry.hash,
      timestamp: entry.timestamp
    });
    
    return sign('sha256', Buffer.from(content), this.privateKey).toString('base64');
  }
  
  async verifyIntegrity(
    startDate: Date,
    endDate: Date
  ): Promise<IntegrityVerificationResult> {
    const entries = await this.getEntries(startDate, endDate);
    
    let valid = true;
    const issues: string[] = [];
    let previousHash = '0';
    
    for (const entry of entries) {
      // Verify hash chain
      if (entry.previousHash !== previousHash) {
        valid = false;
        issues.push(`Chain broken at ${entry.id}`);
      }
      
      // Verify entry hash
      const expectedHash = this.generateHash(entry);
      if (entry.hash !== expectedHash) {
        valid = false;
        issues.push(`Hash mismatch at ${entry.id}`);
      }
      
      // Verify signature
      const signatureValid = this.verifySignature(entry);
      if (!signatureValid) {
        valid = false;
        issues.push(`Invalid signature at ${entry.id}`);
      }
      
      previousHash = entry.hash!;
    }
    
    return { valid, issues, entriesChecked: entries.length };
  }
  
  private async writeToKafka(entry: AuditEntry): Promise<void> {
    await this.producer.send({
      topic: 'audit-log',
      messages: [{
        key: entry.id,
        value: JSON.stringify(entry),
        headers: {
          'content-type': 'application/json',
          'entry-type': entry.event,
          'severity': entry.severity
        }
      }],
      acks: -1, // Wait for all replicas
      timeout: 30000
    });
  }
  
  private async writeToElasticsearch(entry: AuditEntry): Promise<void> {
    await this.elasticsearch.index({
      index: `audit-log-${new Date().toISOString().slice(0, 7)}`, // Monthly indices
      id: entry.id,
      body: entry,
      refresh: 'wait_for' // Ensure it's searchable immediately
    });
  }
  
  private async writeToS3(entry: AuditEntry): Promise<void> {
    // Write to S3 with server-side encryption and versioning
    const key = `audit-log/${new Date().toISOString().slice(0, 10)}/${entry.id}.json`;
    
    await this.s3.putObject({
      Bucket: process.env.AUDIT_LOG_BUCKET!,
      Key: key,
      Body: JSON.stringify(entry),
      ServerSideEncryption: 'aws:kms',
      SSEKMSKeyId: process.env.AUDIT_LOG_KMS_KEY!,
      StorageClass: 'GLACIER_IR' // Immediate retrieval glacier for compliance
    }).promise();
  }
  
  private async writeToSIEM(entry: AuditEntry): Promise<void> {
    // Send to SIEM system (e.g., Splunk, Datadog)
    if (entry.severity === 'critical' || entry.metadata?.securityRelevant) {
      await this.siemClient.sendEvent({
        source: 'noah-audit',
        sourcetype: 'json',
        event: entry
      });
    }
  }
  
  async generateComplianceReport(
    organizationId: string,
    startDate: Date,
    endDate: Date,
    reportType: 'SOC2' | 'ISO27001' | 'GDPR'
  ): Promise<ComplianceReport> {
    const relevantEvents = await this.getComplianceEvents(
      organizationId,
      startDate,
      endDate,
      reportType
    );
    
    const report: ComplianceReport = {
      organizationId,
      reportType,
      period: { start: startDate, end: endDate },
      generatedAt: new Date(),
      summary: await this.generateSummary(relevantEvents),
      events: relevantEvents,
      controls: await this.evaluateControls(relevantEvents, reportType),
      recommendations: await this.generateRecommendations(relevantEvents)
    };
    
    // Sign the report
    report.signature = this.signReport(report);
    
    return report;
  }
  
  private async evaluateControls(
    events: AuditEntry[],
    reportType: string
  ): Promise<ControlEvaluation[]> {
    const controls = this.getControlsForReportType(reportType);
    const evaluations: ControlEvaluation[] = [];
    
    for (const control of controls) {
      const relevantEvents = events.filter(e => 
        control.eventTypes.includes(e.event)
      );
      
      const evaluation: ControlEvaluation = {
        controlId: control.id,
        controlName: control.name,
        status: this.evaluateControlStatus(relevantEvents, control),
        evidence: relevantEvents.map(e => e.id),
        findings: this.generateFindings(relevantEvents, control)
      };
      
      evaluations.push(evaluation);
    }
    
    return evaluations;
  }
  
  private generateId(): string {
    return `${Date.now()}-${randomBytes(16).toString('hex')}`;
  }
}
