CREATE OR REPLACE FUNCTION create_monthly_partitions()
RETURNS void AS $$
DECLARE
    start_date date;
    end_date date;
    partition_name text;
BEGIN
    -- Create partitions for next 3 months
    FOR i IN 0..3 LOOP
        start_date := date_trunc('month', CURRENT_DATE + (i || ' months')::interval);
        end_date := date_trunc('month', start_date + '1 month'::interval);
        partition_name := 'media_assets_' || to_char(start_date, 'YYYY_MM');
        
        -- Check if partition exists
        IF NOT EXISTS (
            SELECT 1 FROM pg_tables 
            WHERE tablename = partition_name
        ) THEN
            EXECUTE format(
                'CREATE TABLE %I PARTITION OF media_assets 
                FOR VALUES FROM (%L) TO (%L)',
                partition_name, start_date, end_date
            );
            
            -- Create partition-specific indexes
            EXECUTE format(
                'CREATE INDEX CONCURRENTLY %I ON %I (org_id, created_at DESC) 
                WHERE deleted_at IS NULL',
                partition_name || '_org_created_idx', partition_name
            );
            
            EXECUTE format(
                'CREATE INDEX CONCURRENTLY %I ON %I USING btree (created_at) 
                WHERE status = %L',
                partition_name || '_pending_idx', partition_name, 'pending'
            );
        END IF;
    END LOOP;
    
    -- Drop old partitions (keep 12 months)
    FOR partition_name IN 
        SELECT tablename 
        FROM pg_tables 
        WHERE tablename LIKE 'media_assets_%'
        AND tablename < 'media_assets_' || to_char(CURRENT_DATE - interval '12 months', 'YYYY_MM')
    LOOP
        -- Archive to cold storage first
        PERFORM archive_partition(partition_name);
        EXECUTE format('DROP TABLE %I', partition_name);
    END LOOP;
END;
$$ LANGUAGE plpgsql;

CREATE MATERIALIZED VIEW mv_org_daily_stats AS
WITH asset_stats AS (
    SELECT 
        org_id,
        DATE_TRUNC('day', created_at) as date,
        COUNT(*) as upload_count,
        SUM(file_size) as total_bytes,
        SUM(file_size * (1 - compression_ratio)) as bytes_saved,
        AVG(compression_ratio)::DECIMAL(4,2) as avg_compression,
        PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY file_size) as median_size,
        JSONB_OBJECT_AGG(
            mime_type, 
            COUNT(*) 
            ORDER BY COUNT(*) DESC
        ) FILTER (WHERE mime_type IS NOT NULL) as files_by_type
    FROM media_assets
    WHERE deleted_at IS NULL
    GROUP BY org_id, DATE_TRUNC('day', created_at)
),
cumulative_stats AS (
    SELECT 
        org_id,
        date,
        upload_count,
        total_bytes,
        bytes_saved,
        avg_compression,
        median_size,
        files_by_type,
        SUM(total_bytes) OVER (
            PARTITION BY org_id 
            ORDER BY date 
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) as cumulative_bytes,
        LAG(upload_count, 7) OVER (
            PARTITION BY org_id 
            ORDER BY date
        ) as upload_count_week_ago
    FROM asset_stats
)
SELECT 
    org_id,
    date,
    upload_count,
    total_bytes,
    bytes_saved,
    avg_compression,
    median_size,
    files_by_type,
    cumulative_bytes,
    CASE 
        WHEN upload_count_week_ago > 0 
        THEN ((upload_count::FLOAT - upload_count_week_ago) / upload_count_week_ago * 100)::DECIMAL(5,2)
        ELSE NULL 
    END as weekly_growth_rate
FROM cumulative_stats;

CREATE OR REPLACE FUNCTION refresh_org_stats_incremental(p_date DATE DEFAULT CURRENT_DATE - 1)
RETURNS void AS $$
BEGIN
    -- Delete existing data for the date
    DELETE FROM mv_org_daily_stats WHERE date = p_date;
    
    -- Insert new data
    INSERT INTO mv_org_daily_stats
    SELECT * FROM generate_org_daily_stats(p_date);
    
    -- Update cumulative values
    UPDATE mv_org_daily_stats curr
    SET cumulative_bytes = prev.cumulative_bytes + curr.total_bytes
    FROM mv_org_daily_stats prev
    WHERE curr.org_id = prev.org_id
    AND curr.date = p_date
    AND prev.date = p_date - 1;
END;
$$ LANGUAGE plpgsql;

typescriptimport { LRUCache } from 'lru-cache';

interface CacheLayer {
  name: string;
  get<T>(key: string): Promise<T | undefined>;
  set<T>(key: string, value: T, options?: CacheOptions): Promise<void>;
  del(key: string): Promise<void>;
  clear(): Promise<void>;
  stats(): Promise<CacheStats>;
}

interface CacheOptions {
  ttl?: number;
  tags?: string[];
  priority?: 'low' | 'medium' | 'high';
  compress?: boolean;
}

interface CacheStats {
  hits: number;
  misses: number;
  evictions: number;
  size: number;
  hitRate: number;
}

// L1: In-memory cache with smart eviction
class MemoryCache implements CacheLayer {
  name = 'memory';
  private cache: LRUCache<string, any>;
  private stats = { hits: 0, misses: 0, evictions: 0 };
  
  constructor(options: { maxSize: number; maxAge: number }) {
    this.cache = new LRUCache({
      max: options.maxSize,
      ttl: options.maxAge * 1000,
      updateAgeOnGet: true,
      updateAgeOnHas: true,
      sizeCalculation: (value) => {
        return JSON.stringify(value).length;
      },
      dispose: (value, key, reason) => {
        if (reason === 'evict') {
          this.stats.evictions++;
        }
      }
    });
  }
  
  async get<T>(key: string): Promise<T | undefined> {
    const value = this.cache.get(key);
    if (value !== undefined) {
      this.stats.hits++;
      return value;
    }
    this.stats.misses++;
    return undefined;
  }
  
  async set<T>(key: string, value: T, options?: CacheOptions): Promise<void> {
    const ttl = options?.ttl ? options.ttl * 1000 : undefined;
    this.cache.set(key, value, { ttl });
  }
  
  async del(key: string): Promise<void> {
    this.cache.delete(key);
  }
  
  async clear(): Promise<void> {
    this.cache.clear();
  }
  
  async stats(): Promise<CacheStats> {
    const total = this.stats.hits + this.stats.misses;
    return {
      ...this.stats,
      size: this.cache.size,
      hitRate: total > 0 ? this.stats.hits / total : 0
    };
  }
}

// L2: Redis cache with clustering and compression
class RedisCache implements CacheLayer {
  name = 'redis';
  private cluster: Redis.Cluster;
  private readonly compression = true;
  
  constructor(nodes: string[]) {
    this.cluster = new Redis.Cluster(nodes, {
      enableReadyCheck: true,
      maxRetriesPerRequest: 3,
      retryDelayOnClusterDown: 300,
      retryDelayOnTryAgain: 100,
      slotsRefreshTimeout: 2000,
      clusterRetryStrategy: (times) => Math.min(100 * times, 2000),
      redisOptions: {
        family: 4,
        connectTimeout: 10000,
        keepAlive: 10000,
        noDelay: true,
        enableOfflineQueue: true
      }
    });
  }
  
  async get<T>(key: string): Promise<T | undefined> {
    try {
      const value = await this.cluster.get(key);
      if (!value) return undefined;
      
      if (this.compression) {
        const decompressed = await this.decompress(value);
        return JSON.parse(decompressed);
      }
      
      return JSON.parse(value);
    } catch (error) {
      console.error('Redis get error:', error);
      return undefined;
    }
  }
  
  async set<T>(key: string, value: T, options?: CacheOptions): Promise<void> {
    try {
      let data = JSON.stringify(value);
      
      if (this.compression && data.length > 1024) {
        data = await this.compress(data);
      }
      
      const ttl = options?.ttl || 300;
      await this.cluster.setex(key, ttl, data);
      
      // Tag-based invalidation
      if (options?.tags) {
        for (const tag of options.tags) {
          await this.cluster.sadd(`tag:${tag}`, key);
          await this.cluster.expire(`tag:${tag}`, ttl);
        }
      }
    } catch (error) {
      console.error('Redis set error:', error);
    }
  }
  
  async invalidateByTag(tag: string): Promise<void> {
    const keys = await this.cluster.smembers(`tag:${tag}`);
    if (keys.length > 0) {
      await this.cluster.del(...keys);
      await this.cluster.del(`tag:${tag}`);
    }
  }
  
  private async compress(data: string): Promise<string> {
    const { promisify } = require('util');
    const zlib = require('zlib');
    const gzip = promisify(zlib.gzip);
    const compressed = await gzip(data);
    return compressed.toString('base64');
  }
  
  private async decompress(data: string): Promise<string> {
    const { promisify } = require('util');
    const zlib = require('zlib');
    const gunzip = promisify(zlib.gunzip);
    const buffer = Buffer.from(data, 'base64');
    const decompressed = await gunzip(buffer);
    return decompressed.toString();
  }
  
  async del(key: string): Promise<void> {
    await this.cluster.del(key);
  }
  
  async clear(): Promise<void> {
    // Clear with pattern matching
    const keys = await this.cluster.keys('*');
    if (keys.length > 0) {
      await this.cluster.del(...keys);
    }
  }
  
  async stats(): Promise<CacheStats> {
    const info = await this.cluster.info('stats');
    // Parse Redis INFO stats
    const hits = parseInt(info.match(/keyspace_hits:(\d+)/)?.[1] || '0');
    const misses = parseInt(info.match(/keyspace_misses:(\d+)/)?.[1] || '0');
    const evictions = parseInt(info.match(/evicted_keys:(\d+)/)?.[1] || '0');
    
    return {
      hits,
      misses,
      evictions,
      size: await this.cluster.dbsize(),
      hitRate: hits / (hits + misses) || 0
    };
  }
}

// Advanced Cache Manager with intelligent routing
export class CacheManager extends EventEmitter {
  private layers: CacheLayer[] = [];
  private keyHasher = new XXHash64();
  private hotKeys = new Set<string>();
  private accessCounts = new Map<string, number>();
  
  constructor() {
    super();
    
    // Initialize cache layers
    this.layers = [
      new MemoryCache({ maxSize: 10000, maxAge: 60 }),
      new RedisCache([
        'redis-1.noah.internal:6379',
        'redis-2.noah.internal:6379',
        'redis-3.noah.internal:6379'
      ]),
      new CDNCache('E1234567890ABC')
    ];
    
    // Start background tasks
    this.startHotKeyDetection();
    this.startCacheWarming();
    this.startMetricsCollection();
  }
  
  async get<T>(
    key: string,
    factory: () => Promise<T>,
    options: CacheOptions = {}
  ): Promise<T> {
    // Track access for hot key detection
    this.trackAccess(key);
    
    // Generate consistent hash for sharding
    const hash = this.keyHasher.hash(Buffer.from(key));
    
    // Try each cache layer
    for (let i = 0; i < this.layers.length; i++) {
      const layer = this.layers[i];
      
      try {
        const value = await layer.get<T>(key);
        if (value !== undefined) {
          // Promote to higher layers if hot key
          if (this.isHotKey(key) && i > 0) {
            await this.promote(key, value, i);
          }
          
          this.emit('cache:hit', { layer: layer.name, key });
          return value;
        }
      } catch (error) {
        console.error(`Cache layer ${layer.name} error:`, error);
        // Continue to next layer
      }
    }
    
    // Cache miss - generate value
    this.emit('cache:miss', { key });
    
    try {
      const value = await factory();
      
      // Store in appropriate layers based on priority
      await this.store(key, value, options);
      
      return value;
    } catch (error) {
      this.emit('cache:error', { key, error });
      throw error;
    }
  }
  
  async set<T>(
    key: string,
    value: T,
    options: CacheOptions = {}
  ): Promise<void> {
    await this.store(key, value, options);
  }
  
  async invalidate(key: string): Promise<void> {
    await Promise.all(
      this.layers.map(layer => layer.del(key))
    );
    
    this.emit('cache:invalidated', { key });
  }
  
  async invalidateByTag(tag: string): Promise<void> {
    // Only Redis supports tag-based invalidation
    const redisLayer = this.layers.find(l => l.name === 'redis') as RedisCache;
    if (redisLayer) {
      await redisLayer.invalidateByTag(tag);
    }
    
    this.emit('cache:invalidated:tag', { tag });
  }
  
  private async store<T>(
    key: string,
    value: T,
    options: CacheOptions
  ): Promise<void> {
    const promises: Promise<void>[] = [];
    
    // Determine which layers to store in based on priority
    const priority = options.priority || 'medium';
    
    if (priority === 'high' || this.isHotKey(key)) {
      // Store in all layers for high priority
      promises.push(...this.layers.map(layer => layer.set(key, value, options)));
    } else if (priority === 'medium') {
      // Skip L1 for medium priority
      promises.push(...this.layers.slice(1).map(layer => layer.set(key, value, options)));
    } else {
      // Only L3 for low priority
      promises.push(this.layers[2].set(key, value, options));
    }
    
    await Promise.all(promises);
  }
  
  private async promote<T>(
    key: string,
    value: T,
    fromIndex: number
  ): Promise<void> {
    // Promote to all higher layers
    const promises: Promise<void>[] = [];
    
    for (let i = 0; i < fromIndex; i++) {
      promises.push(this.layers[i].set(key, value));
    }
    
    await Promise.all(promises);
  }
  
  private trackAccess(key: string): void {
    const count = (this.accessCounts.get(key) || 0) + 1;
    this.accessCounts.set(key, count);
  }
  
  private isHotKey(key: string): boolean {
    return this.hotKeys.has(key);
  }
  
  private startHotKeyDetection(): void {
    setInterval(() => {
      // Find keys accessed more than threshold
      const threshold = 100; // accesses per minute
      const newHotKeys = new Set<string>();
      
      for (const [key, count] of this.accessCounts) {
        if (count > threshold) {
          newHotKeys.add(key);
        }
      }
      
      // Update hot keys
      this.hotKeys = newHotKeys;
      
      // Reset counts
      this.accessCounts.clear();
      
      this.emit('hotkeys:updated', { count: this.hotKeys.size });
    }, 60000); // Every minute
  }
  
  private async startCacheWarming(): Promise<void> {
    // Warm cache with predicted hot keys
    setInterval(async () => {
      try {
        const predictedKeys = await this.predictHotKeys();
        
        for (const key of predictedKeys) {
          // Pre-warm in L1 and L2
          const value = await this.layers[2].get(key); // Get from L3
          if (value) {
            await this.layers[0].set(key, value, { ttl: 300 });
            await this.layers[1].set(key, value, { ttl: 3600 });
          }
        }
        
        this.emit('cache:warmed', { count: predictedKeys.length });
      } catch (error) {
        console.error('Cache warming error:', error);
      }
    }, 300000); // Every 5 minutes
  }
  
  private async predictHotKeys(): Promise<string[]> {
    // ML-based prediction or time-based patterns
    // For now, return top accessed keys from previous period
    const sorted = Array.from(this.accessCounts.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, 100)
      .map(([key]) => key);
    
    return sorted;
  }
  
  private startMetricsCollection(): void {
    setInterval(async () => {
      const metrics = await this.getMetrics();
      
      // Export to monitoring system
      this.emit('metrics:collected', metrics);
      
      // Auto-tune cache sizes based on hit rates
      if (metrics.memory.hitRate < 0.5 && metrics.redis.hitRate > 0.8) {
        // Consider increasing memory cache size
        this.emit('cache:tune', {
          recommendation: 'increase_memory_cache',
          current: metrics
        });
      }
    }, 60000); // Every minute
  }
  
  async getMetrics(): Promise<Record<string, CacheStats>> {
    const metrics: Record<string, CacheStats> = {};
    
    for (const layer of this.layers) {
      metrics[layer.name] = await layer.stats();
    }
    
    return metrics;
  }
}

typescriptimport { CloudFrontClient } from '@aws-sdk/client-cloudfront';

interface OptimizationOptions {
  format?: 'auto' | 'webp' | 'avif' | 'jpeg' | 'png';
  quality?: number;
  width?: number;
  height?: number;
  fit?: 'cover' | 'contain' | 'fill' | 'inside' | 'outside';
  dpr?: number;
  progressive?: boolean;
}

export class CDNOptimizer {
  private readonly supportedFormats = {
    'image/webp': 'webp',
    'image/avif': 'avif',
    'image/jpeg': 'jpeg',
    'image/png': 'png'
  };
  
  async optimizeDelivery(
    assetId: string,
    request: Request
  ): Promise<OptimizedResponse> {
    const asset = await this.getAsset(assetId);
    
    // Parse optimization parameters from request
    const params = this.parseOptimizationParams(request);
    
    // Check if optimized variant exists
    const variantKey = this.generateVariantKey(asset.id, params);
    const cachedVariant = await this.cache.get(variantKey);
    
    if (cachedVariant) {
      return this.createResponse(cachedVariant, params);
    }
    
    // Generate optimized variant
    const variant = await this.generateVariant(asset, params);
    
    // Store variant
    await this.storeVariant(variantKey, variant);
    
    return this.createResponse(variant, params);
  }
  
  private parseOptimizationParams(request: Request): OptimizationOptions {
    const url = new URL(request.url);
    const headers = request.headers;
    
    // Auto-detect best format based on Accept header
    const acceptHeader = headers.get('accept') || '';
    let format: string = 'auto';
    
    if (acceptHeader.includes('image/avif')) {
      format = 'avif';
    } else if (acceptHeader.includes('image/webp')) {
      format = 'webp';
    }
    
    // Parse query parameters
    const params: OptimizationOptions = {
      format: (url.searchParams.get('f') as any) || format,
      quality: parseInt(url.searchParams.get('q') || '85'),
      width: parseInt(url.searchParams.get('w') || '0') || undefined,
      height: parseInt(url.searchParams.get('h') || '0') || undefined,
      fit: (url.searchParams.get('fit') as any) || 'cover',
      dpr: parseFloat(headers.get('dpr') || url.searchParams.get('dpr') || '1'),
      progressive: url.searchParams.get('progressive') !== 'false'
    };
    
    // Apply DPR to dimensions
    if (params.dpr > 1) {
      if (params.width) params.width *= params.dpr;
      if (params.height) params.height *= params.dpr;
    }
    
    // Security limits
    params.width = Math.min(params.width || 4096, 4096);
    params.height = Math.min(params.height || 4096, 4096);
    params.quality = Math.max(1, Math.min(params.quality, 100));
    
    return params;
  }
  
  private async generateVariant(
    asset: MediaAsset,
    params: OptimizationOptions
  ): Promise<OptimizedVariant> {
    const startTime = Date.now();
    
    // Download original
    const original = await this.downloadAsset(asset);
    
    // Create sharp instance
    let pipeline = sharp(original, {
      failOnError: false,
      sequentialRead: true,
      density: 300 // For SVG
    });
    
    // Get metadata
    const metadata = await pipeline.metadata();
    
    // Resize if needed
    if (params.width || params.height) {
      pipeline = pipeline.resize({
        width: params.width,

          aqMode: 3, // Auto-variance AQ

        }
      };

    // Default balanced approach
    return {
      type: 'standard',
      name: 'Balanced Compression',
      reason: 'Standard system load',
      params: {
        preset: 'medium',
        crf: 23,
        threads: Math.ceil(os.cpus().length * 0.7)

          index,

        }
      });

      worker.on('error', reject);

        }
      });

export class MemoryProfiler extends EventEmitter {
  private profiles: MemoryProfile[] = [];

  private isMonitoring = false;
  private gcStats = {
    count: 0,
    duration: 0,

  };
  
  startProfiling(interval: number = 10000): void {
    if (this.isMonitoring) return;
    
    this.isMonitoring = true;
    
    // Enable GC tracking
    if (global.gc) {
      const obs = new PerformanceObserver((list) => {
        const entries = list.getEntries();
        entries.forEach((entry) => {
          if (entry.name === 'gc') {
            this.gcStats.count++;
            this.gcStats.duration += entry.duration;
            this.gcStats.lastGC = Date.now();
          }
        });
      });
      obs.observe({ entryTypes: ['gc'] });
    }
    
    // Periodic memory profiling
    const profileInterval = setInterval(() => {
      if (!this.isMonitoring) {
        clearInterval(profileInterval);
        return;
      }
      
      this.collectProfile();
    }, interval);
    
    // Heap snapshot on demand
    process.on('SIGUSR2', () => {
      this.takeHeapSnapshot();
    });
  }
  
  private collectProfile(): void {
    const memUsage = process.memoryUsage();
    
    const profile: MemoryProfile = {
      timestamp: Date.now(),
      heapUsed: memUsage.heapUsed,
      heapTotal: memUsage.heapTotal,
      external: memUsage.external,
      arrayBuffers: memUsage.arrayBuffers || 0,
      pid: process.pid
    };
    
    this.profiles.push(profile);
    
    // Keep only last hour of profiles
    const hourAgo = Date.now() - 3600000;
    this.profiles = this.profiles.filter(p => p.timestamp > hourAgo);
    
    // Check for memory leaks
    if (this.profiles.length > 10) {
      const leakResult = this.detectMemoryLeak();
      if (leakResult.hasLeak) {
        this.emit('memory:leak:detected', leakResult);
        
        // Auto-remediation
        if (leakResult.confidence > 0.8) {
          this.attemptLeakMitigation();
        }
      }
    }
    
    this.emit('memory:profile', profile);
  }
  
  private detectMemoryLeak(): LeakDetectionResult {
    if (this.profiles.length < 10) {
      return { hasLeak: false, confidence: 0, growthRate: 0, suspectObjects: [] };
    }
    
    // Linear regression on heap usage
    const regression = this.linearRegression(
      this.profiles.map(p => p.timestamp),
      this.profiles.map(p => p.heapUsed)
    );
    
    // Calculate growth rate (bytes per second)
    const growthRate = regression.slope;
    
    // Leak detection criteria
    const hasLeak = growthRate > 1024 * 1024; // 1MB/s growth
    const confidence = Math.min(regression.r2, 1.0);
    
    // Identify suspect objects if leak detected
    let suspectObjects: string[] = [];
    if (hasLeak && confidence > 0.7) {
      suspectObjects = this.identifySuspectObjects();
    }
    
    return {
      hasLeak,
      confidence,
      growthRate,
      suspectObjects
    };
  }
  
  private linearRegression(x: number[], y: number[]): { slope: number; r2: number } {
    const n = x.length;
    const sumX = x.reduce((a, b) => a + b, 0);
    const sumY = y.reduce((a, b) => a + b, 0);
    const sumXY = x.reduce((sum, xi, i) => sum + xi * y[i], 0);
    const sumX2 = x.reduce((sum, xi) => sum + xi * xi, 0);
    const sumY2 = y.reduce((sum, yi) => sum + yi * yi, 0);
    
    const slope = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
    const intercept = (sumY - slope * sumX) / n;
    
    // Calculate R-squared
    const yMean = sumY / n;
    const ssTotal = y.reduce((sum, yi) => sum + Math.pow(yi - yMean, 2), 0);
    const ssResidual = y.reduce((sum, yi, i) => {
      const yPred = slope * x[i] + intercept;
      return sum + Math.pow(yi - yPred, 2);
    }, 0);
    
    const r2 = 1 - (ssResidual / ssTotal);
    
    return { slope, r2 };
  }
  
  private identifySuspectObjects(): string[] {

    const heapStats = v8.getHeapStatistics();
    const heapSpaces = v8.getHeapSpaceStatistics();

      const avgGCDuration = this.gcStats.duration / this.gcStats.count;
      if (avgGCDuration > 100) {
        suspects.push('Inefficient garbage collection');

    }

  }

  async takeHeapSnapshot(): Promise<string> {

    }

    return filepath;
  }

export class PerformanceProfiler {
  private marks = new Map<string, number>();
  private measures = new Map<string, number[]>();
  private asyncHooks = require('async_hooks');
  private asyncOperations = new Map();
  
  constructor() {
    this.setupAsyncHooks();
  }
  
  mark(name: string): void {
    this.marks.set(name, performance.now());
  }
  
  measure(name: string, startMark: string, endMark?: string): number {
    const start = this.marks.get(startMark);
    if (!start) throw new Error(`Start mark ${startMark} not found`);
    
    const end = endMark ? this.marks.get(endMark) : performance.now();
    if (!end) throw new Error(`End mark ${endMark} not found`);
    
    const duration = end - start;
    
    if (!this.measures.has(name)) {
      this.measures.set(name, []);
    }
    
    this.measures.get(name)!.push(duration);
    
    // Keep only last 1000 measurements
    const measurements = this.measures.get(name)!;
    if (measurements.length > 1000) {
      measurements.shift();
    }
    
    return duration;
  }
  
  getStats(measureName: string): PerformanceStats {
    const measurements = this.measures.get(measureName);
    if (!measurements || measurements.length === 0) {
      return {
        count: 0,
        mean: 0,
        median: 0,
        p95: 0,
        p99: 0,
        min: 0,
        max: 0
      };
    }
    
    const sorted = [...measurements].sort((a, b) => a - b);
    const count = sorted.length;
    
    return {
      count,
      mean: sorted.reduce((a, b) => a + b, 0) / count,
      median: sorted[Math.floor(count / 2)],
      p95: sorted[Math.floor(count * 0.95)],
      p99: sorted[Math.floor(count * 0.99)],
      min: sorted[0],
      max: sorted[count - 1]
    };
  }
  
  private setupAsyncHooks(): void {
    const hook = this.asyncHooks.createHook({
      init: (asyncId, type, triggerAsyncId) => {
        if (type === 'PROMISE' || type === 'TIMERWRAP') {
          this.asyncOperations.set(asyncId, {
            type,
            triggerAsyncId,
            startTime: performance.now()
          });
        }
      },
      destroy: (asyncId) => {
        const op = this.asyncOperations.get(asyncId);
        if (op) {
          const duration = performance.now() - op.startTime;
          if (duration > 100) { // Log slow async operations
            console.log(`Slow async operation: ${op.type} took ${duration}ms`);
          }
          this.asyncOperations.delete(asyncId);
        }
      }
    });
    
    hook.enable();
  }
  
  profileFunction<T extends (...args: any[]) => any>(
    fn: T,
    name: string
  ): T {
    return ((...args: Parameters<T>) => {
      const start = performance.now();
      
      try {
        const result = fn(...args);
        
        if (result instanceof Promise) {
          return result.finally(() => {
            this.recordExecution(name, performance.now() - start);
          });
        }
        
        this.recordExecution(name, performance.now() - start);
        return result;
      } catch (error) {
        this.recordExecution(name, performance.now() - start);
        throw error;
      }
    }) as T;
  }
  
  private recordExecution(name: string, duration: number): void {
    if (!this.measures.has(name)) {
      this.measures.set(name, []);
    }
    
    this.measures.get(name)!.push(duration);
    
    // Alert on slow operations
    if (duration > 1000) {
      console.warn(`Slow operation detected: ${name} took ${duration}ms`);
    }
  }
  
  generateReport(): PerformanceReport {
    const report: PerformanceReport = {
      timestamp: Date.now(),
      measurements: {},
      slowOperations: [],
      recommendations: []
    };
    
    for (const [name, measurements] of this.measures) {
      const stats = this.getStats(name);
      report.measurements[name] = stats;
      
      // Identify slow operations
      if (stats.p95 > 100) {
        report.slowOperations.push({
          name,
          p95: stats.p95,
          p99: stats.p99,
          impact: 'high'
        });
      }
    }
    
    // Generate optimization recommendations
    report.recommendations = this.generateOptimizationRecommendations(report);
    
    return report;
  }
  
  private generateOptimizationRecommendations(report: PerformanceReport): string[] {
    const recommendations: string[] = [];
    
    for (const op of report.slowOperations) {
      if (op.name.includes('database')) {
        recommendations.push(`Optimize database queries in ${op.name} - consider adding indexes`);
      } else if (op.name.includes('api')) {
        recommendations.push(`API call ${op.name} is slow - implement caching or batch requests`);
      } else if (op.p99 > op.p95 * 2) {
        recommendations.push(`High variance in ${op.name} - investigate outliers`);
      }
    }
    
    return recommendations;
  }
}

  private prioritizeRecommendations(
    recommendations: OptimizationRecommendation[]
  ): OptimizationRecommendation[] {
    // Score each recommendation
    const scored = recommendations.map(rec => ({
      ...rec,
      score: this.calculateRecommendationScore(rec)
    }));
    
    // Sort by score descending
    return scored.sort((a, b) => b.score - a.score);
  }
  
  private calculateRecommendationScore(rec: OptimizationRecommendation): number {
    const effortScore = { low: 3, medium: 2, high: 1 }[rec.effort];
    const impactScore = { low: 1, medium: 2, high: 3 }[rec.impact];
    const savingsScore = Math.min(rec.potentialSavings / 1000, 3); // Normalize to 0-3
    
    return effortScore + impactScore + savingsScore;
  }
  
  private async autoImplementRecommendations(
    recommendations: OptimizationRecommendation[]
  ): Promise<string[]> {
    const implemented: string[] = [];
    
    for (const rec of recommendations) {
      if (rec.effort === 'low' && rec.impact !== 'low') {
        try {
          await this.implementRecommendation(rec);
          implemented.push(rec.type);
