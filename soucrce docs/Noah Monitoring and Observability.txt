# prometheus-rules.yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: noah-platform-rules
  namespace: noah-production
  labels:
    prometheus: noah
spec:
  groups:
  - name: noah-api
    interval: 30s
    rules:
    # API Health Rules
    - alert: NoahAPIDown
      expr: up{job="noah-api"} == 0
      for: 1m
      labels:
        severity: critical
        service: noah-api
        team: platform
      annotations:
        summary: "Noah API is down"
        description: "Noah API instance {{ $labels.instance }} has been down for more than 1 minute"
        runbook_url: "https://runbooks.noah.app/api-down"
        dashboard_url: "https://grafana.noah.app/d/noah-api/noah-api-performance"
    
    - alert: NoahAPIHighErrorRate
      expr: |
        (
          sum(rate(noah_http_requests_total{status=~"5.."}[5m])) by (job)
          /
          sum(rate(noah_http_requests_total[5m])) by (job)
        ) > 0.05
      for: 5m
      labels:
        severity: critical
        service: noah-api
        team: platform
      annotations:
        summary: "High error rate on Noah API"
        description: "Error rate is {{ $value | humanizePercentage }} for job {{ $labels.job }}"
        runbook_url: "https://runbooks.noah.app/api-high-error-rate"
    
    - alert: NoahAPIHighLatency
      expr: |
        histogram_quantile(0.95,
          sum(rate(noah_http_request_duration_seconds_bucket[5m])) by (le, job)
        ) > 1
      for: 5m
      labels:
        severity: warning
        service: noah-api
        team: platform
      annotations:
        summary: "High latency on Noah API"
        description: "95th percentile latency is {{ $value }}s for job {{ $labels.job }}"
        runbook_url: "https://runbooks.noah.app/api-high-latency"
    
    # Resource Usage Rules
    - alert: NoahAPIPodHighMemory
      expr: |
        (
          container_memory_working_set_bytes{namespace="noah-production",pod=~"noah-api-.*"}
          /
          container_spec_memory_limit_bytes{namespace="noah-production",pod=~"noah-api-.*"}
        ) > 0.9
      for: 5m
      labels:
        severity: warning
        service: noah-api
        team: platform
      annotations:
        summary: "High memory usage in Noah API pod"
        description: "Pod {{ $labels.pod }} memory usage is at {{ $value | humanizePercentage }}"
        runbook_url: "https://runbooks.noah.app/pod-high-memory"
    
    - alert: NoahAPIPodCPUThrottling
      expr: |
        rate(container_cpu_cfs_throttled_seconds_total{namespace="noah-production",pod=~"noah-api-.*"}[5m]) > 0.1
      for: 5m
      labels:
        severity: warning
        service: noah-api
        team: platform
      annotations:
        summary: "CPU throttling detected in Noah API pod"
        description: "Pod {{ $labels.pod }} is experiencing CPU throttling"
        runbook_url: "https://runbooks.noah.app/pod-cpu-throttling"
    
    # Database Rules
    - alert: NoahDatabaseConnectionsHigh
      expr: |
        noah_database_connections_active / noah_database_connections_max > 0.8
      for: 5m
      labels:
        severity: warning
        service: database
        team: database
      annotations:
        summary: "Database connection pool usage high"
        description: "Database connection usage is at {{ $value | humanizePercentage }}"
        runbook_url: "https://runbooks.noah.app/database-connections-high"
    
    - alert: NoahDatabaseReplicationLag
      expr: |
        noah_database_replication_lag_seconds > 60
      for: 5m
      labels:
        severity: critical
        service: database
        team: database
      annotations:
        summary: "Database replication lag detected"
        description: "Replication lag is {{ $value }}s for replica {{ $labels.replica }}"
        runbook_url: "https://runbooks.noah.app/database-replication-lag"
    
    # Queue and Worker Rules
    - alert: NoahCompressionQueueBacklog
      expr: |
        noah_compression_queue_size > 1000
      for: 15m
      labels:
        severity: warning
        service: noah-worker
        team: platform
      annotations:
        summary: "Compression queue backlog detected"
        description: "Compression queue has {{ $value }} items pending"
        runbook_url: "https://runbooks.noah.app/compression-queue-backlog"
    
    - alert: NoahWorkerProcessingError
      expr: |
        rate(noah_worker_processing_errors_total[5m]) > 0.1
      for: 5m
      labels:
        severity: warning
        service: noah-worker
        team: platform
      annotations:
        summary: "Worker processing errors detected"
        description: "Worker {{ $labels.worker_type }} is experiencing {{ $value }} errors per second"
        runbook_url: "https://runbooks.noah.app/worker-processing-errors"
    
    # Storage Rules
    - alert: NoahStorageUsageHigh
      expr: |
        (noah_storage_bytes_used / noah_storage_bytes_quota) > 0.9
      for: 30m
      labels:
        severity: warning
        service: noah-storage
        team: platform
      annotations:
        summary: "Storage usage is high"
        description: "Storage usage is at {{ $value | humanizePercentage }} of quota"
        runbook_url: "https://runbooks.noah.app/storage-usage-high"
    
    - alert: NoahB2UploadErrors
      expr: |
        rate(noah_b2_upload_errors_total[5m]) > 0.05
      for: 5m
      labels:
        severity: warning
        service: noah-storage
        team: platform
      annotations:
        summary: "B2 upload errors detected"
        description: "B2 upload error rate is {{ $value }} per second"
        runbook_url: "https://runbooks.noah.app/b2-upload-errors"
    
    # SLO Rules
    - alert: NoahSLOBudgetBurn
      expr: |
        (
          1 - (
            sum(rate(noah_http_requests_total{status!~"5.."}[30m]))
            /
            sum(rate(noah_http_requests_total[30m]))
          )
        ) > 0.001
      for: 5m
      labels:
        severity: warning
        service: noah-api
        team: platform
        slo: availability
      annotations:
        summary: "SLO budget burn rate high"
        description: "Error budget burn rate is {{ $value | humanizePercentage }} over the last 30 minutes"
        runbook_url: "https://runbooks.noah.app/slo-budget-burn"
