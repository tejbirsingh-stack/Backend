// hooks/useOptimizedQuery.ts
import { useQuery, UseQueryOptions } from '@tanstack/react-query';
import { useEffect, useRef } from 'react';
import { InteractionManager } from 'react-native';
import NetInfo from '@react-native-community/netinfo';
import { storage } from '../App';

interface OptimizedQueryOptions<T> extends UseQueryOptions<T> {
  cacheKey?: string;
  enableOfflineSupport?: boolean;
  lowDataMode?: boolean;
}

export function useOptimizedQuery<T>(
  queryKey: any[],
  queryFn: () => Promise<T>,
  options?: OptimizedQueryOptions<T>
) {
  const interactionHandle = useRef<any>();
  
  const modifiedOptions: UseQueryOptions<T> = {
    ...options,
    queryFn: async () => {
      // Wait for interactions to complete
      await new Promise((resolve) => {
        interactionHandle.current = InteractionManager.runAfterInteractions(() => {
          resolve(undefined);
        });
      });
      
      // Check network status
      const netInfo = await NetInfo.fetch();
      
      if (!netInfo.isConnected && options?.enableOfflineSupport) {
        // Try to get from cache
        const cacheKey = options.cacheKey || queryKey.join(':');
        const cachedData = storage.getString(`cache:${cacheKey}`);
        if (cachedData) {
          return JSON.parse(cachedData);
        }
      }
      
      // Low data mode - reduce quality/quantity
      if (options?.lowDataMode && netInfo.type === 'cellular') {
        // Modify request for low data usage
        // This is context-specific
      }
      
      const data = await queryFn();
      
      // Cache the data
      if (options?.enableOfflineSupport) {
        const cacheKey = options.cacheKey || queryKey.join(':');
        storage.set(`cache:${cacheKey}`, JSON.stringify(data));
      }
      
      return data;
    },
  };
  
  useEffect(() => {
    return () => {
      if (interactionHandle.current) {
        interactionHandle.current.cancel();
      }
    };
  }, []);
  
  return useQuery(queryKey, modifiedOptions);
}

// utils/imageOptimizer.ts
import { Image } from 'react-native';
import FastImage from 'react-native-fast-image';

interface ImageDimensions {
  width: number;
  height: number;
}

class ImageOptimizer {
  private dimensionsCache = new Map<string, ImageDimensions>();
  
  async getImageDimensions(uri: string): Promise<ImageDimensions> {
    if (this.dimensionsCache.has(uri)) {
      return this.dimensionsCache.get(uri)!;
    }
    
    return new Promise((resolve, reject) => {
      Image.getSize(
        uri,
        (width, height) => {
          const dimensions = { width, height };
          this.dimensionsCache.set(uri, dimensions);
          resolve(dimensions);
        },
        reject
      );
    });
  }
  
  getOptimizedUri(uri: string, targetWidth: number): string {
    // For services that support dynamic resizing
    if (uri.includes('cloudinary.com')) {
      return uri.replace('/upload/', `/upload/w_${targetWidth}/`);
    }
    if (uri.includes('imgix.net')) {
      return `${uri}?w=${targetWidth}&auto=format&q=75`;
    }
    return uri;
  }
  
  preloadImages(urls: string[]): void {
    FastImage.preload(
      urls.map(uri => ({
        uri,
        priority: FastImage.priority.normal,
        cache: FastImage.cacheControl.immutable,
      }))
    );
  }
  
  clearCache(): void {
    this.dimensionsCache.clear();
    // FastImage doesn't expose cache clearing on iOS
    // On Android, we can clear disk cache
    if (Platform.OS === 'android') {
      FastImage.clearDiskCache();
    }
  }
}

export const imageOptimizer = new ImageOptimizer();

// utils/performanceMonitor.ts
import { 
  InteractionManager,
  NativeModules,
  Platform,
  DeviceEventEmitter
} from 'react-native';
import analytics from '@react-native-firebase/analytics';

interface PerformanceMetric {
  name: string;
  value: number;
  unit: string;
  metadata?: Record<string, any>;
}

class PerformanceMonitor {
  private metrics: PerformanceMetric[] = [];
  private frameCallbackId?: number;
  private lastFrameTime = 0;
  private frameCount = 0;
  private fps = 60;
  
  startFPSMonitoring() {
    if (this.frameCallbackId) return;
    
    const measureFPS = () => {
      const now = performance.now();
      if (this.lastFrameTime !== 0) {
        const delta = now - this.lastFrameTime;
        const currentFPS = 1000 / delta;
        
        // Moving average
        this.fps = this.fps * 0.9 + currentFPS * 0.1;
        
        if (this.frameCount % 60 === 0) {
          this.reportMetric({
            name: 'fps',
            value: Math.round(this.fps),
            unit: 'frames_per_second',
          });
          
          if (this.fps < 55) {
            console.warn(`Low FPS detected: ${this.fps.toFixed(1)}`);
          }
        }
      }
      
      this.lastFrameTime = now;
      this.frameCount++;
      this.frameCallbackId = requestAnimationFrame(measureFPS);
    };
    
    measureFPS();
  }
  
  stopFPSMonitoring() {
    if (this.frameCallbackId) {
      cancelAnimationFrame(this.frameCallbackId);
      this.frameCallbackId = undefined;
    }
  }
  
  measureInteraction(name: string, fn: () => Promise<void>) {
    const startTime = performance.now();
    
    return InteractionManager.runAfterInteractions(async () => {
      try {
        await fn();
      } finally {
        const duration = performance.now() - startTime;
        this.reportMetric({
          name: `interaction_${name}`,
          value: duration,
          unit: 'milliseconds',
        });
      }
    });
  }
  
  measureRender(componentName: string): () => void {
    const startTime = performance.now();
    
    return () => {
      const duration = performance.now() - startTime;
      this.reportMetric({
        name: `render_${componentName}`,
        value: duration,
        unit: 'milliseconds',
      });
    };
  }
  
  private reportMetric(metric: PerformanceMetric) {
    this.metrics.push(metric);
    
    // Report to analytics
    analytics().logEvent('performance_metric', {
      metric_name: metric.name,
      metric_value: metric.value,
      metric_unit: metric.unit,
      ...metric.metadata,
    });
    
    // Log in development
    if (__DEV__) {
      console.log(`[Performance] ${metric.name}: ${metric.value}${metric.unit}`);
    }
  }
  
  getMetrics(): PerformanceMetric[] {
    return [...this.metrics];
  }
  
  clearMetrics() {
    this.metrics = [];
  }
}

export const performanceMonitor = new PerformanceMonitor();

// utils/haptics.ts
import { Platform } from 'react-native';
import ReactNativeHapticFeedback from 'react-native-haptic-feedback';

const hapticOptions = {
  enableVibrateFallback: true,
  ignoreAndroidSystemSettings: false,
};

export function hapticFeedback(type: 'light' | 'medium' | 'heavy' | 'success' | 'warning' | 'error') {
  if (Platform.OS === 'ios') {
    switch (type) {
      case 'light':
        ReactNativeHapticFeedback.trigger('impactLight', hapticOptions);
        break;
      case 'medium':
        ReactNativeHapticFeedback.trigger('impactMedium', hapticOptions);
        break;
      case 'heavy':
        ReactNativeHapticFeedback.trigger('impactHeavy', hapticOptions);
        break;
      case 'success':
        ReactNativeHapticFeedback.trigger('notificationSuccess', hapticOptions);
        break;
      case 'warning':
        ReactNativeHapticFeedback.trigger('notificationWarning', hapticOptions);
        break;
      case 'error':
        ReactNativeHapticFeedback.trigger('notificationError', hapticOptions);
        break;
    }
  } else {
    // Android haptic feedback
    ReactNativeHapticFeedback.trigger('impactMedium', hapticOptions);
  }
}
