import React, { useState, useEffect, useCallback } from 'react';
import { FixedSizeGrid as Grid } from 'react-window';
import InfiniteLoader from 'react-window-infinite-loader';
import { useDebounce } from '../hooks/useDebounce';
import { MediaThumbnail } from './MediaThumbnail';
import { FilterPanel } from './FilterPanel';
import { NoahAsset } from '../types';

interface MediaBrowserProps {
  panel: NoahPanel;
}

export const MediaBrowser: React.FC<MediaBrowserProps> = ({ panel }) => {
  const [assets, setAssets] = useState<NoahAsset[]>([]);
  const [searchQuery, setSearchQuery] = useState('');
  const [filters, setFilters] = useState({
    mediaType: 'all',
    dateRange: null,
    tags: [],
    collection: null
  });
  const [hasNextPage, setHasNextPage] = useState(true);
  const [isLoading, setIsLoading] = useState(false);
  
  const debouncedSearch = useDebounce(searchQuery, 300);
  
  const loadMoreAssets = useCallback(async (startIndex: number, stopIndex: number) => {
    if (isLoading) return;
    
    setIsLoading(true);
    try {
      const response = await panel.api.searchAssets({
        query: debouncedSearch,
        filters,
        offset: startIndex,
        limit: stopIndex - startIndex + 1
      });
      
      setAssets(prev => {
        const newAssets = [...prev];
        response.items.forEach((asset, index) => {
          newAssets[startIndex + index] = asset;
        });
        return newAssets;
      });
      
      setHasNextPage(response.hasMore);
    } catch (error) {
      console.error('Failed to load assets:', error);
      panel.showError('Failed to load media assets');
    } finally {
      setIsLoading(false);
    }
  }, [debouncedSearch, filters, panel]);
  
  const handleDragStart = (event: React.DragEvent, asset: NoahAsset) => {
    // Set up drag data for Premiere Pro
    const dragData = {
      effectType: 'noah-import',
      assetId: asset.id,
      assetUrl: asset.url,
      metadata: asset.metadata
    };
    
    event.dataTransfer.setData('application/json', JSON.stringify(dragData));
    event.dataTransfer.effectAllowed = 'copy';
    
    // Visual feedback
    const dragImage = new Image();
    dragImage.src = asset.thumbnailUrl;
    event.dataTransfer.setDragImage(dragImage, 50, 50);
  };
  
  const isItemLoaded = (index: number) => !!assets[index];
  
  const Cell = ({ columnIndex, rowIndex, style }) => {
    const index = rowIndex * 3 + columnIndex;
    const asset = assets[index];
    
    if (!asset) {
      return <div style={style} className="loading-placeholder" />;
    }
    
    return (
      <div style={style} className="media-cell">
        <MediaThumbnail
          asset={asset}
          onDragStart={(e) => handleDragStart(e, asset)}
          onDoubleClick={() => panel.importAsset(asset)}
          onContextMenu={(e) => showContextMenu(e, asset)}
        />
      </div>
    );
  };
  
  return (
    <div className="media-browser">
      <div className="search-bar">
        <input
          type="text"
          placeholder="Search media..."
          value={searchQuery}
          onChange={(e) => setSearchQuery(e.target.value)}
          className="search-input"
        />
        <FilterPanel 
          filters={filters}
          onChange={setFilters}
        />
      </div>
      
      <div className="media-grid-container">
        <InfiniteLoader
          isItemLoaded={isItemLoaded}
          itemCount={hasNextPage ? assets.length + 20 : assets.length}
          loadMoreItems={loadMoreAssets}
        >
          {({ onItemsRendered, ref }) => (
            <Grid
              ref={ref}
              columnCount={3}
              columnWidth={120}
              height={500}
              rowCount={Math.ceil(assets.length / 3)}
              rowHeight={120}
              width={380}
              onItemsRendered={({
                visibleRowStartIndex,
                visibleRowStopIndex,
                overscanRowStopIndex,
                overscanRowStartIndex,
              }) => {
                onItemsRendered({
                  overscanStartIndex: overscanRowStartIndex * 3,
                  overscanStopIndex: overscanRowStopIndex * 3,
                  visibleStartIndex: visibleRowStartIndex * 3,
                  visibleStopIndex: visibleRowStopIndex * 3,
                });
              }}
            >
              {Cell}
            </Grid>
          )}
        </InfiniteLoader>
      </div>
    </div>
  );
};

// components/TimelineSync.tsx
export class TimelineSync {
  private ws: WebSocket | null = null;
  private syncInterval: number | null = null;
  private lastSequenceState: any = null;
  
  constructor(private panel: NoahPanel) {}
  
  async startSync(projectId: string) {
    // Connect to WebSocket for real-time updates
    this.ws = new WebSocket(`wss://api.noah.app/v1/projects/${projectId}/sync`);
    
    this.ws.onopen = () => {
      console.log('Timeline sync connected');
      this.sendCurrentState();
    };
    
    this.ws.onmessage = (event) => {
      const update = JSON.parse(event.data);
      this.handleRemoteUpdate(update);
    };
    
    // Start monitoring local changes
    this.syncInterval = window.setInterval(() => {
      this.checkForLocalChanges();
    }, 1000);
  }
  
  private async checkForLocalChanges() {
    const jsxCode = `
      (function() {
        var sequence = app.project.activeSequence;
        if (!sequence) return null;
        
        var state = {
          id: sequence.id,
          name: sequence.name,
          clips: [],
          markers: []
        };
        
        // Get all clips
        for (var i = 0; i < sequence.videoTracks.numTracks; i++) {
          var track = sequence.videoTracks[i];
          for (var j = 0; j < track.clips.numItems; j++) {
            var clip = track.clips[j];
            state.clips.push({
              name: clip.name,
              start: clip.start.seconds,
              end: clip.end.seconds,
              trackIndex: i,
              mediaPath: clip.projectItem.getMediaPath()
            });
          }
        }
        
        // Get markers
        for (var i = 0; i < sequence.markers.numMarkers; i++) {
          var marker = sequence.markers[i];
          state.markers.push({
            name: marker.name,
            time: marker.start.seconds,
            comment: marker.comments,
            color: marker.color
          });
        }
        
        return JSON.stringify(state);
      })()
    `;
    
    this.panel.csInterface.evalScript(jsxCode, (result) => {
      if (result) {
        const currentState = JSON.parse(result);
        if (this.hasStateChanged(currentState)) {
          this.sendUpdate(currentState);
          this.lastSequenceState = currentState;
        }
      }
    });
  }
  
  private hasStateChanged(newState: any): boolean {
    if (!this.lastSequenceState) return true;
    return JSON.stringify(newState) !== JSON.stringify(this.lastSequenceState);
  }
  
  private sendUpdate(state: any) {
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify({
        type: 'sequence_update',
        data: state,
        timestamp: new Date().toISOString()
      }));
    }
  }
  
  private handleRemoteUpdate(update: any) {
    switch (update.type) {
      case 'marker_added':
        this.addMarker(update.data);
        break;
      case 'clip_moved':
        this.moveClip(update.data);
        break;
      case 'effect_applied':
        this.applyEffect(update.data);
        break;
    }
  }
  
  stopSync() {
    if (this.ws) {
      this.ws.close();
      this.ws = null;
    }
    if (this.syncInterval) {
      clearInterval(this.syncInterval);
      this.syncInterval = null;
    }
  }
}

// ProxyManager.ts - Intelligent proxy workflow
export class ProxyManager {
  private proxyQueue: Map<string, ProxyJob> = new Map();
  private activeJobs = 0;
  private maxConcurrentJobs = 3;
  
  constructor(private panel: NoahPanel) {}
  
  async generateProxiesForSequence() {
    // Get all clips in active sequence
    const clips = await this.getSequenceClips();
    
    // Filter clips that need proxies
    const clipsNeedingProxies = clips.filter(clip => {
      return !clip.hasProxy && clip.fileSize > 100 * 1024 * 1024; // > 100MB
    });
    
    // Queue proxy generation
    for (const clip of clipsNeedingProxies) {
      await this.queueProxyGeneration(clip);
    }
    
    // Start processing queue
    this.processQueue();
  }
  
  private async queueProxyGeneration(clip: any) {
    const job: ProxyJob = {
      id: crypto.randomUUID(),
      clipId: clip.id,
      sourcePath: clip.mediaPath,
      status: 'queued',
      progress: 0
    };
    
    this.proxyQueue.set(job.id, job);
    
    // Request proxy generation from Noah server
    const response = await this.panel.api.generateProxy({
      assetId: clip.assetId,
      preset: this.getProxyPreset(clip),
      priority: clip.isInActiveSequence ? 'high' : 'normal'
    });
    
    job.serverJobId = response.jobId;
  }
  
  private getProxyPreset(clip: any): string {
    // Intelligent preset selection based on source
    if (clip.width > 3840) {
      return 'proxy_4k_to_1080p';
    } else if (clip.width > 1920) {
      return 'proxy_2k_to_720p';
    } else {
      return 'proxy_1080p_to_540p';
    }
  }
  
  private async processQueue() {
    if (this.activeJobs >= this.maxConcurrentJobs) return;
    
    const nextJob = Array.from(this.proxyQueue.values())
      .find(job => job.status === 'queued');
    
    if (!nextJob) return;
    
    this.activeJobs++;
    nextJob.status = 'processing';
    
    // Monitor proxy generation
    const checkInterval = setInterval(async () => {
      const status = await this.panel.api.getProxyStatus(nextJob.serverJobId);
      
      nextJob.progress = status.progress;
      this.updateUI(nextJob);
      
      if (status.status === 'complete') {
        clearInterval(checkInterval);
        await this.attachProxy(nextJob, status.proxyUrl);
        this.activeJobs--;
        this.processQueue();
      } else if (status.status === 'failed') {
        clearInterval(checkInterval);
        nextJob.status = 'failed';
        this.activeJobs--;
        this.processQueue();
      }
    }, 2000);
  }
  
  private async attachProxy(job: ProxyJob, proxyUrl: string) {
    // Download proxy to local cache
    const localPath = await this.downloadProxy(proxyUrl, job.clipId);
    
    // Attach proxy in Premiere
    const jsxCode = `
      (function() {
        var item = app.project.getItemByID("${job.clipId}");
        if (item && item.canChangeMediaPath()) {
          return item.attachProxy("${localPath}", 0);
        }
        return false;
      })()
    `;
    
    this.panel.csInterface.evalScript(jsxCode, (result) => {
      if (result === 'true') {
        job.status = 'complete';
        this.panel.showNotification('Proxy attached successfully');
      } else {
        job.status = 'failed';
        this.panel.showError('Failed to attach proxy');
      }
    });
  }
}

// AIAssistant.ts
export class AIAssistant {
  constructor(private panel: NoahPanel) {}
  
  async suggestCuts(options: CutSuggestionOptions) {
    const sequence = await this.getActiveSequence();
    
    // Analyze sequence content
    const analysis = await this.panel.api.analyzeSequence({
      clips: sequence.clips,
      audio: sequence.audioData,
      style: options.style || 'dynamic',
      targetDuration: options.targetDuration
    });
    
    // Generate cut suggestions
    const suggestions = analysis.suggestions.map(s => ({
      timestamp: s.timestamp,
      type: s.type,
      confidence: s.confidence,
      reason: s.reason,
      preview: this.generatePreview(s)
    }));
    
    // Show suggestions UI
    this.showSuggestionsPanel(suggestions);
  }
  
  async autoColorGrade(style: string = 'cinematic') {
    const selectedClips = await this.getSelectedClips();
    
    for (const clip of selectedClips) {
      // Analyze clip characteristics
      const analysis = await this.panel.api.analyzeClip({
        clipId: clip.id,
        frameSamples: 10
      });
      
      // Get color grading suggestions
      const colorData = await this.panel.api.getColorGrade({
        analysis,
        style,
        referenceImage: this.getReferenceImage()
      });
      
      // Apply Lumetri effect
      await this.applyLumetriEffect(clip, colorData);
    }
  }
  
  private async applyLumetriEffect(clip: any, colorData: any) {
    const jsxCode = `
      (function() {
        var clip = app.project.getItemByID("${clip.id}");
        if (!clip) return false;
        
        // Find or add Lumetri effect
        var lumetri = null;
        for (var i = 0; i < clip.videoEffects.numItems; i++) {
          if (clip.videoEffects[i].matchName === "AE.ADBE Lumetri") {
            lumetri = clip.videoEffects[i];
            break;
          }
        }
        
        if (!lumetri) {
          lumetri = clip.videoEffects.addEffect(
            app.effects.getEffectByMatchName("AE.ADBE Lumetri")
          );
        }
        
        // Apply color values
        var props = lumetri.properties;
        props.getParamForDisplayName("Temperature").setValue(${colorData.temperature});
        props.getParamForDisplayName("Tint").setValue(${colorData.tint});
        props.getParamForDisplayName("Exposure").setValue(${colorData.exposure});
        props.getParamForDisplayName("Contrast").setValue(${colorData.contrast});
        props.getParamForDisplayName("Highlights").setValue(${colorData.highlights});
        props.getParamForDisplayName("Shadows").setValue(${colorData.shadows});
        props.getParamForDisplayName("Saturation").setValue(${colorData.saturation});
        
        return true;
      })()
    `;
    
    return new Promise((resolve) => {
      this.panel.csInterface.evalScript(jsxCode, resolve);
    });
  }
}

// CollaborationFeatures.ts
export class CollaborationManager {
  private presenceService: PresenceService;
  private conflictResolver: ConflictResolver;
  
  constructor(private panel: NoahPanel) {
    this.presenceService = new PresenceService(panel);
    this.conflictResolver = new ConflictResolver(panel);
  }
  
  async enableLiveCollaboration(projectId: string) {
    // Initialize presence
    await this.presenceService.connect(projectId);
    
    // Set up change tracking
    this.trackChanges();
    
    // Enable cursor sharing
    this.enableCursorSharing();
    
    // Set up voice chat
    this.initializeVoiceChat();
  }
  
  private trackChanges() {
    // Monitor all changes in Premiere
    const changeTypes = [
      'clip_added',
      'clip_removed', 
      'clip_moved',
      'effect_added',
      'effect_modified',
      'marker_added',
      'transition_added'
    ];
    
    changeTypes.forEach(type => {
      this.panel.csInterface.addEventListener(
        `com.adobe.csxs.events.${type}`,
        (event) => this.handleChange(type, event.data)
      );
    });
  }
  
  private async handleChange(type: string, data: any) {
    const change = {
      id: crypto.randomUUID(),
      type,
      data,
      userId: this.panel.currentUser.id,
      timestamp: new Date().toISOString()
    };
    
    // Check for conflicts
    const conflicts = await this.conflictResolver.checkConflicts(change);
    
    if (conflicts.length > 0) {
      const resolution = await this.conflictResolver.resolveConflicts(conflicts);
      if (resolution.action === 'abort') return;
    }
    
    // Broadcast change
    this.presenceService.broadcastChange(change);
    
    // Update timeline
    this.updateTimeline(change);
  }
  
  private enableCursorSharing() {
    // Track mouse position in timeline
    document.addEventListener('mousemove', (e) => {
      if (this.isOverTimeline(e)) {
        const timelinePos = this.getTimelinePosition(e);
        this.presenceService.updateCursor({
          x: timelinePos.x,
          y: timelinePos.y,
          time: timelinePos.time,
          track: timelinePos.track
        });
      }
    });
    
    // Display other users' cursors
    this.presenceService.onCursorUpdate((userId, cursor) => {
      this.displayUserCursor(userId, cursor);
    });
  }
}

// NoahTimelineController.swift
import Foundation
import FxPlug

public class NoahTimelineController {
    private let timeline: FxTimeline
    private var syncEngine: TimelineSyncEngine
    
    public init(timeline: FxTimeline) {
        self.timeline = timeline
        self.syncEngine = TimelineSyncEngine()
    }
    
    public func enableSmartSync() {
        // Monitor timeline changes
        timeline.addObserver(self, 
                           forKeyPath: "clips",
                           options: [.new, .old],
                           context: nil)
        
        // Start sync engine
        syncEngine.start(timelineId: timeline.identifier)
    }
    
    public func importNoahSequence(_ sequenceData: NoahSequence) async throws {
        // Create compound clip for complex edits
        let compoundClip = FxCompoundClip()
        
        // Build timeline structure
        for track in sequenceData.tracks {
            let fxTrack = compoundClip.addTrack(type: track.type)
            
            for clip in track.clips {
                // Download media if needed
                let mediaRef = try await resolveMediaReference(clip.mediaId)
                
                // Create clip with effects
                let fxClip = FxClip(media: mediaRef)
                fxClip.inPoint = CMTime(seconds: clip.inPoint, preferredTimescale: 600)
                fxClip.outPoint = CMTime(seconds: clip.outPoint, preferredTimescale: 600)
                
                // Apply effects
                for effect in clip.effects {
                    try applyEffect(effect, to: fxClip)
                }
                
                // Add to track
                fxTrack.insertClip(fxClip, at: CMTime(seconds: clip.startTime, preferredTimescale: 600))
            }
        }
        
        // Insert into timeline
        timeline.insertClip(compoundClip, at: timeline.currentTime)
    }
    
    private func applyEffect(_ effectData: NoahEffect, to clip: FxClip) throws {
        switch effectData.type {
        case .colorCorrection:
            let colorEffect = FxColorCorrectionEffect()
            if let params = effectData.parameters as? ColorCorrectionParams {
                colorEffect.temperature = params.temperature
                colorEffect.tint = params.tint
                colorEffect.exposure = params.exposure
                colorEffect.saturation = params.saturation
            }
            clip.addEffect(colorEffect)
            
        case .transform:
            let transformEffect = FxTransformEffect()
            if let params = effectData.parameters as? TransformParams {
                transformEffect.position = CGPoint(x: params.x, y: params.y)
                transformEffect.scale = params.scale
                transformEffect.rotation = params.rotation
            }
            clip.addEffect(transformEffect)
            
        case .custom:
            // Load custom Motion template
            if let templateURL = Bundle.main.url(forResource: effectData.templateName,
                                               withExtension: "moef") {
                let customEffect = try FxMotionEffect(contentsOf: templateURL)
                customEffect.parameters = effectData.parameters
                clip.addEffect(customEffect)
            }
        }
    }
}

// NoahMLFeatures.swift - Machine Learning Integration
import CoreML
import Vision

public class NoahMLFeatures {
    private let sceneClassifier: VNCoreMLModel
    private let objectDetector: VNCoreMLModel
    private let styleTransfer: VNCoreMLModel
    
    public init() throws {
        // Load ML models
        let sceneModel = try MLModel(contentsOf: Bundle.main.url(forResource: "NoahSceneClassifier", 
                                                                withExtension: "mlmodelc")!)
        self.sceneClassifier = try VNCoreMLModel(for: sceneModel)
        
        let objectModel = try MLModel(contentsOf: Bundle.main.url(forResource: "NoahObjectDetector",
                                                                 withExtension: "mlmodelc")!)
        self.objectDetector = try VNCoreMLModel(for: objectModel)
        
        let styleModel = try MLModel(contentsOf: Bundle.main.url(forResource: "NoahStyleTransfer",
                                                                withExtension: "mlmodelc")!)
        self.styleTransfer = try VNCoreMLModel(for: styleModel)
    }
    
    public func analyzeClip(_ clip: FxClip) async throws -> ClipAnalysis {
        var analysis = ClipAnalysis()
        
        // Sample frames from clip
        let frameSampler = FrameSampler(clip: clip)
        let frames = try await frameSampler.sampleFrames(count: 10)
        
        // Scene classification
        for (index, frame) in frames.enumerated() {
            let sceneRequest = VNCoreMLRequest(model: sceneClassifier) { request, error in
                if let results = request.results as? [VNClassificationObservation] {
                    let topResults = results.prefix(3).map { 
                        SceneClass(label: $0.identifier, confidence: $0.confidence)
                    }
                    analysis.scenes.append(Scene(timestamp: index, classes: topResults))
                }
            }
            
            try VNImageRequestHandler(cgImage: frame).perform([sceneRequest])
        }
        
        // Object detection
        let objectRequest = VNCoreMLRequest(model: objectDetector) { request, error in
            if let results = request.results as? [VNRecognizedObjectObservation] {
                analysis.objects = results.map { observation in
                    DetectedObject(
                        label: observation.labels.first?.identifier ?? "unknown",
                        confidence: observation.confidence,
                        boundingBox: observation.boundingBox
                    )
                }
            }
        }
        
        if let firstFrame = frames.first {
            try VNImageRequestHandler(cgImage: firstFrame).perform([objectRequest])
        }
        
        return analysis
    }
    
    public func suggestEdits(for timeline: FxTimeline) async throws -> [EditSuggestion] {
        var suggestions: [EditSuggestion] = []
        
        // Analyze pacing
        let pacingAnalysis = analyzePacing(timeline)
        if pacingAnalysis.averageShotLength > 5.0 {
            suggestions.append(EditSuggestion(
                type: .pacing,
                description: "Consider shorter cuts for more dynamic pacing",
                timestamp: pacingAnalysis.slowestSection,
                confidence: 0.8
            ))
        }
        
        // Detect jump cuts
        let jumpCuts = detectJumpCuts(timeline)
        for jumpCut in jumpCuts {
            suggestions.append(EditSuggestion(
                type: .continuity,
                description: "Potential jump cut detected",
                timestamp: jumpCut.timestamp,
                confidence: jumpCut.severity
            ))
        }
        
        // Color matching suggestions
        let colorAnalysis = await analyzeColorConsistency(timeline)
        for mismatch in colorAnalysis.mismatches {
            suggestions.append(EditSuggestion(
                type: .color,
                description: "Color mismatch with previous shot",
                timestamp: mismatch.timestamp,
                confidence: mismatch.severity,
                autoFix: {
                    self.autoMatchColor(clip: mismatch.clip, reference: mismatch.reference)
                }
            ))
        }
        
        return suggestions
    }
}
