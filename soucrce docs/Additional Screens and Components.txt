// screens/HomeScreen.tsx
import React, { useCallback, useRef } from 'react';
import {
  View,
  Text,
  StyleSheet,
  SafeAreaView,
  StatusBar,
  Platform,
  RefreshControl,
} from 'react-native';
import { FlashList } from '@shopify/flash-list';
import { useInfiniteQuery } from '@tanstack/react-query';
import Animated, {
  useSharedValue,
  useAnimatedScrollHandler,
  useAnimatedStyle,
  interpolate,
  Extrapolate,
} from 'react-native-reanimated';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import { FeedItem } from '../components/FeedItem';
import { StoryBar } from '../components/StoryBar';
import { useOptimizedQuery } from '../hooks/useOptimizedQuery';

const AnimatedFlashList = Animated.createAnimatedComponent(FlashList);

export const HomeScreen: React.FC = () => {
  const insets = useSafeAreaInsets();
  const scrollY = useSharedValue(0);
  const lastScrollY = useRef(0);
  
  const {
    data,
    fetchNextPage,
    hasNextPage,
    isFetchingNextPage,
    isRefetching,
    refetch,
  } = useInfiniteQuery({
    queryKey: ['feed'],
    queryFn: async ({ pageParam = 0 }) => {
      const response = await fetch(`/api/feed?page=${pageParam}`);
      return response.json();
    },
    getNextPageParam: (lastPage) => lastPage.nextPage,
  });
  
  const scrollHandler = useAnimatedScrollHandler({
    onScroll: (event) => {
      scrollY.value = event.contentOffset.y;
    },
  });
  
  const headerAnimatedStyle = useAnimatedStyle(() => {
    const translateY = interpolate(
      scrollY.value,
      [0, 100],
      [0, -100],
      Extrapolate.CLAMP
    );
    
    return {
      transform: [{ translateY }],
    };
  });
  
  const renderItem = useCallback(({ item }: { item: any }) => {
    return <FeedItem item={item} />;
  }, []);
  
  const feedData = data?.pages.flatMap(page => page.items) ?? [];
  
  return (
    <SafeAreaView style={styles.container}>
      <StatusBar
        barStyle="dark-content"
        backgroundColor="transparent"
        translucent
      />
      
      <Animated.View style={[styles.header, headerAnimatedStyle]}>
        <Text style={styles.logo}>Instagram</Text>
      </Animated.View>
      
      <AnimatedFlashList
        data={feedData}
        renderItem={renderItem}
        keyExtractor={(item) => item.id}
        estimatedItemSize={600}
        onScroll={scrollHandler}
        scrollEventThrottle={16}
        ListHeaderComponent={<StoryBar />}
        onEndReached={() => {
          if (hasNextPage && !isFetchingNextPage) {
            fetchNextPage();
          }
        }}
        onEndReachedThreshold={0.5}
        refreshControl={
          <RefreshControl
            refreshing={isRefetching}
            onRefresh={refetch}
            tintColor="#007AFF"
          />
        }
        contentContainerStyle={{
          paddingTop: insets.top + 44,
          paddingBottom: insets.bottom,
        }}
      />
    </SafeAreaView>
  );
};

// components/FeedItem.tsx
import React, { useState, useCallback } from 'react';
import {
  View,
  Text,
  StyleSheet,
  Pressable,
  Dimensions,
} from 'react-native';
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withSpring,
  withSequence,
  withDelay,
} from 'react-native-reanimated';
import { Image } from 'expo-image';
import { Heart, MessageCircle, Send, Bookmark } from 'lucide-react-native';
import { DoubleTapGesture } from '../components/DoubleTapGesture';
import { hapticFeedback } from '../utils/haptics';

const { width: screenWidth } = Dimensions.get('window');

interface FeedItemProps {
  item: {
    id: string;
    user: {
      id: string;
      username: string;
      avatar: string;
    };
    media: {
      url: string;
      type: 'image' | 'video';
      aspectRatio: number;
    };
    caption?: string;
    likes: number;
    comments: number;
    timestamp: string;
  };
}

export const FeedItem: React.FC<FeedItemProps> = ({ item }) => {
  const [liked, setLiked] = useState(false);
  const likeScale = useSharedValue(0);
  const heartScale = useSharedValue(1);
  
  const handleDoubleTap = useCallback(() => {
    if (!liked) {
      setLiked(true);
      hapticFeedback('success');
      
      // Animate the large heart
      likeScale.value = withSequence(
        withSpring(1, { damping: 6 }),
        withDelay(500, withSpring(0))
      );
      
      // Animate the heart button
      heartScale.value = withSequence(
        withSpring(1.3),
        withSpring(1)
      );
    }
  }, [liked]);
  
  const handleLikePress = useCallback(() => {
    setLiked(!liked);
    hapticFeedback('light');
    
    heartScale.value = withSequence(
      withSpring(0.8),
      withSpring(1.2),
      withSpring(1)
    );
  }, [liked]);
  
  const likeAnimatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: likeScale.value }],
    opacity: likeScale.value,
  }));
  
  const heartAnimatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: heartScale.value }],
  }));
  
  const imageHeight = screenWidth / item.media.aspectRatio;
  
  return (
    <View style={styles.container}>
      <View style={styles.header}>
        <Pressable style={styles.userInfo}>
          <Image
            source={{ uri: item.user.avatar }}
            style={styles.avatar}
            contentFit="cover"
          />
          <Text style={styles.username}>{item.user.username}</Text>
        </Pressable>
      </View>
      
      <DoubleTapGesture onDoubleTap={handleDoubleTap}>
        <View>
          <Image
            source={{ uri: item.media.url }}
            style={[styles.media, { height: imageHeight }]}
            contentFit="cover"
            transition={200}
          />
          
          <Animated.View style={[styles.bigHeart, likeAnimatedStyle]}>
            <Heart
              size={80}
              color="white"
              fill="white"
              style={styles.bigHeartIcon}
            />
          </Animated.View>
        </View>
      </DoubleTapGesture>
      
      <View style={styles.actions}>
        <View style={styles.leftActions}>
          <Pressable onPress={handleLikePress}>
            <Animated.View style={heartAnimatedStyle}>
              <Heart
                size={24}
                color={liked ? '#ED4956' : '#000'}
                fill={liked ? '#ED4956' : 'none'}
              />
            </Animated.View>
          </Pressable>
          
          <Pressable style={styles.actionButton}>
            <MessageCircle size={24} color="#000" />
          </Pressable>
          
          <Pressable style={styles.actionButton}>
            <Send size={24} color="#000" />
          </Pressable>
        </View>
        
        <Pressable>
          <Bookmark size={24} color="#000" />
        </Pressable>
      </View>
      
      <View style={styles.content}>
        <Text style={styles.likes}>{item.likes.toLocaleString()} likes</Text>
        {item.caption && (
          <Text style={styles.caption}>
            <Text style={styles.username}>{item.user.username}</Text> {item.caption}
          </Text>
        )}
        <Text style={styles.timestamp}>{item.timestamp}</Text>
      </View>
    </View>
  );
};

# Instagram-Level React Native App

A high-performance Instagram clone built with React Native, featuring native modules, 120fps animations, and production-ready optimizations.

## Features

- **60-120fps Animations**: Smooth interactions using Reanimated 3
- **Native Performance**: Custom native modules for video compression and image processing
- **Optimized Media Grid**: FlashList implementation handling 10,000+ items
- **Smart Caching**: MMKV storage with offline support
- **Advanced Camera**: Full-featured camera with filters and effects
- **Push Notifications**: Firebase Cloud Messaging integration
- **Deep Linking**: Universal links and app links support
- **CI/CD**: Automated deployment with GitHub Actions and Fastlane

## Tech Stack

- React Native 0.74 (New Architecture enabled)
- Expo SDK 50 (Custom Dev Client)
- TypeScript
- React Navigation 6
- React Query
- Reanimated 3
- MMKV Storage
- Native Modules (Swift/Kotlin)

## Prerequisites

- Node.js 18+
- Yarn
- Xcode 14+ (for iOS)
- Android Studio (for Android)
- CocoaPods
- Ruby (for Fastlane)
